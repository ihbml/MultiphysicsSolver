/* ============================================================================
 *  cockpit::geometry  —  procedural cascade mesh, R32Uint-style face picking,
 *  patch statistics and the Smart Spatial Rule Synthesizer.
 *  (web/wgpu-neutral: the same structures are POD-mapped by bytemuck in Rust)
 * ==========================================================================*/
export type Vec3 = [number, number, number];
export type AxisIdx = 0 | 1 | 2;
export const AXIS_NAMES = ["x", "y", "z"] as const;
export type GroupId =
  | "leading_edge"
  | "trailing_edge"
  | "suction_side"
  | "pressure_side"
  | "hub_root"
  | "tip_shroud";
export const CHORD = 0.12; // L, axial / chordwise extent  (x)
export const SPAN_ROOT = 0.02; // hub radius                (z)
export const SPAN_TIP = 0.14; // tip radius                 (z)
export interface Face {
  id: number;
  vi: [number, number, number];
  n: Vec3;
  c: Vec3;
  area: number;
  group: GroupId;
  tag: string;
  u: number; // centroid chord fraction x/c
  side: 1 | -1;
}
export interface Mesh {
  positions: Float32Array;
  normals: Float32Array;
  data: Float32Array; // [x/c, side]
  highlight: Float32Array;
  indices: Uint32Array;
  faces: Face[];
  byGroup: Record<string, number[]>;
  bbox: { min: Vec3; max: Vec3 };
}
const NU = 56; // chordwise divisions (cosine spaced)
const NV = 22; // spanwise divisions
/* ---------- NACA4-digit section ---------------------------------------- */
function section(c: number) {
  const t = 0.12,
    m = 0.035,
    p = 0.4;
  const yt =
    5 *
    t *
    (0.2969 * Math.sqrt(c) - 0.126 * c - 0.3516 * c * c + 0.2843 * c ** 3 - 0.1015 * c ** 4);
  let yc = 0,
    dyc = 0;
  if (c < p) {
    yc = (m / (p * p)) * (2 * p * c - c * c);
    dyc = (2 * m / (p * p)) * (p - c);
  } else {
    yc = (m / ((1 - p) * (1 - p))) * (1 - 2 * p + 2 * p * c - c * c);
    dyc = (2 * m / ((1 - p) * (1 - p))) * (p - c);
  }
  return { yc, yt, dyc };
}
const tagFor: Record<GroupId, string> = {
  leading_edge: "A1/LE",
  trailing_edge: "A1/TE",
  suction_side: "A1/SUCTION_SIDE",
  pressure_side: "A1/PRESSURE_SIDE",
  hub_root: "A1/HUB_FILLET",
  tip_shroud: "A1/TIP_SHROUD",
};
export function buildMesh(): Mesh {
  const nSheet = (NU + 1) * (NV + 1);
  const positions = new Float32Array(nSheet * 2 * 3);
  const data = new Float32Array(nSheet * 2 * 2);
  const idx = (sheet: number, i: number, j: number) => sheet * nSheet + j * (NU + 1) + i;
  for (let sheet = 0; sheet < 2; sheet++) {
    const s: 1 | -1 = sheet === 0 ? 1 : -1; // sheet0 = suction (+y), sheet1 = pressure
    for (let j = 0; j <= NV; j++) {
      const zn = j / NV;
      const z = SPAN_ROOT + (SPAN_TIP - SPAN_ROOT) * zn;
      const chord = CHORD * (1 - 0.12 * zn);
      const twist = ((14 - 11 * zn) * Math.PI) / 180;
      const ct = Math.cos(twist),
        st = Math.sin(twist);
      for (let i = 0; i <= NU; i++) {
        const u = 0.5 * (1 - Math.cos((Math.PI * i) / NU));
        const { yc, yt } = section(u);
        const lx = u * chord - 0.25 * chord;
        const ly = (yc + s * yt) * chord;
        const rx = lx * ct - ly * st + 0.25 * chord;
        const ry = lx * st + ly * ct;
        const k = idx(sheet, i, j);
        positions[k * 3] = rx;
        positions[k * 3 + 1] = ry;
        positions[k * 3 + 2] = z;
        data[k * 2] = u;
        data[k * 2 + 1] = s;
      }
    }
  }
  const faces: Face[] = [];
  const push = (
    sheet: number,
    i: number,
    j: number,
    a: number,
    b: number,
    c: number,
  ) => {
    const P = (v: number): Vec3 => [positions[v * 3], positions[v * 3 + 1], positions[v * 3 + 2]];
    const pa = P(a),
      pb = P(b),
      pc = P(c);
    const e1: Vec3 = [pb[0] - pa[0], pb[1] - pa[1], pb[2] - pa[2]];
    const e2: Vec3 = [pc[0] - pa[0], pc[1] - pa[1], pc[2] - pa[2]];
    let n: Vec3 = [
      e1[1] * e2[2] - e1[2] * e2[1],
      e1[2] * e2[0] - e1[0] * e2[2],
      e1[0] * e2[1] - e1[1] * e2[0],
    ];
    const len = Math.hypot(n[0], n[1], n[2]) || 1;
    n = [n[0] / len, n[1] / len, n[2] / len];
    const s: 1 | -1 = sheet === 0 ? 1 : -1;
    let tri: [number, number, number] = [a, b, c];
    if (n[1] * s < 0) {
      tri = [a, c, b];
      n = [-n[0], -n[1], -n[2]];
    }
    const cxyz: Vec3 = [
      (pa[0] + pb[0] + pc[0]) / 3,
      (pa[1] + pb[1] + pc[1]) / 3,
      (pa[2] + pb[2] + pc[2]) / 3,
    ];
    const area = len / 2;
    let group: GroupId;
    if (j === 0) group = "hub_root";
    else if (j === NV - 1) group = "tip_shroud";
    else if (i <= 1) group = "leading_edge";
    else if (i >= NU - 1) group = "trailing_edge";
    else group = s > 0 ? "suction_side" : "pressure_side";
    faces.push({
      id: faces.length,
      vi: tri,
      n,
      c: cxyz,
      area,
      group,
      tag: tagFor[group],
      u: cxyz[0] / (CHORD * (1 - 0.12 * ((cxyz[2] - SPAN_ROOT) / (SPAN_TIP - SPAN_ROOT)))),
      side: s,
    });
  };
  for (let sheet = 0; sheet < 2; sheet++) {
    for (let j = 0; j < NV; j++) {
      for (let i = 0; i < NU; i++) {
        const p00 = idx(sheet, i, j);
        const p10 = idx(sheet, i + 1, j);
        const p11 = idx(sheet, i + 1, j + 1);
        const p01 = idx(sheet, i, j + 1);
        push(sheet, i, j, p00, p10, p11);
        push(sheet, i, j, p00, p11, p01);
      }
    }
  }
  // smooth normals by quantised position
  const acc = new Map<number, Vec3>();
  const key = (v: number) =>
    `${Math.round(positions[v * 3] * 1e5)},${Math.round(positions[v * 3 + 1] * 1e5)},${Math.round(
      positions[v * 3 + 2] * 1e5,
    )}`;
  for (const f of faces) {
    for (const v of f.vi) {
      const k = key(v);
      const cur = acc.get(k);
      if (cur) {
        cur[0] += f.n[0];
        cur[1] += f.n[1];
        cur[2] += f.n[2];
      } else acc.set(k, [f.n[0], f.n[1], f.n[2]]);
    }
  }
  const normals = new Float32Array(positions.length);
  for (let v = 0; v < positions.length / 3; v++) {
    const a = acc.get(key(v)) ?? [0, 0, 1];
    const l = Math.hypot(a[0], a[1], a[2]) || 1;
    normals[v * 3] = a[0] / l;
    normals[v * 3 + 1] = a[1] / l;
    normals[v * 3 + 2] = a[2] / l;
  }
  const indices = new Uint32Array(faces.length * 3);
  faces.forEach((f, i) => {
    indices[i * 3] = f.vi[0];
    indices[i * 3 + 1] = f.vi[1];
    indices[i * 3 + 2] = f.vi[2];
  });
  const bbox = { min: [1e9, 1e9, 1e9] as Vec3, max: [-1e9, -1e9, -1e9] as Vec3 };
  for (let v = 0; v < positions.length / 3; v++) {
    for (let a = 0; a < 3; a++) {
      const x = positions[v * 3 + a];
      if (x < bbox.min[a]) bbox.min[a] = x;
      if (x > bbox.max[a]) bbox.max[a] = x;
    }
  }
  const byGroup: Record<string, number[]> = {};
  for (const f of faces) (byGroup[f.group] ??= []).push(f.id);
  return {
    positions,
    normals,
    data,
    highlight: new Float32Array(positions.length / 3),
    indices,
    faces,
    byGroup,
    bbox,
  };
}
/* ---------- camera + picking -------------------------------------------- */
export interface Camera {
  theta: number;
  phi: number;
  dist: number;
  target: Vec3;
}
export function eyeOf(cam: Camera): Vec3 {
  const cp = Math.cos(cam.phi);
  return [
    cam.target[0] + cam.dist * cp * Math.cos(cam.theta),
    cam.target[1] + cam.dist * cp * Math.sin(cam.theta),
    cam.target[2] + cam.dist * Math.sin(cam.phi),
  ];
}
export function basisOf(cam: Camera): { x: Vec3; y: Vec3; z: Vec3 } {
  const eye = eyeOf(cam);
  const z: Vec3 = [eye[0] - cam.target[0], eye[1] - cam.target[1], eye[2] - cam.target[2]];
  const zl = Math.hypot(z[0], z[1], z[2]) || 1;
  z[0] /= zl;
  z[1] /= zl;
  z[2] /= zl;
  const up: Vec3 = [0, 0, 1];
  let x: Vec3 = [
    up[1] * z[2] - up[2] * z[1],
    up[2] * z[0] - up[0] * z[2],
    up[0] * z[1] - up[1] * z[0],
  ];
  const xl = Math.hypot(x[0], x[1], x[2]) || 1;
  x = [x[0] / xl, x[1] / xl, x[2] / xl];
  const y: Vec3 = [
    z[1] * x[2] - z[2] * x[1],
    z[2] * x[0] - z[0] * x[2],
    z[0] * x[1] - z[1] * x[0],
  ];
  return { x, y, z };
}
/** screen-space ray (origin = eye, unit direction in world) */
export function rayFromScreen(
  cam: Camera,
  mx: number,
  my: number,
  w: number,
  h: number,
  fovY: number,
): { o: Vec3; d: Vec3 } {
  const ndcX = (mx / w) * 2 - 1;
  const ndcY = 1 - (my / h) * 2;
  const aspect = w / Math.max(1, h);
  const t = Math.tan(fovY / 2);
  const dv: Vec3 = [ndcX * t * aspect, ndcY * t, -1];
  const b = basisOf(cam);
  const d: Vec3 = [
    dv[0] * b.x[0] + dv[1] * b.y[0] + dv[2] * b.z[0],
    dv[0] * b.x[1] + dv[1] * b.y[1] + dv[2] * b.z[1],
    dv[0] * b.x[2] + dv[1] * b.y[2] + dv[2] * b.z[2],
  ];
  const l = Math.hypot(d[0], d[1], d[2]) || 1;
  return { o: eyeOf(cam), d: [d[0] / l, d[1] / l, d[2] / l] };
}
export interface Pick {
  faceId: number;
  point: Vec3;
  dist: number;
}
export function pickFace(mesh: Mesh, ray: { o: Vec3; d: Vec3 }): Pick | null {
  let best: Pick | null = null;
  const EPS = 1e-9;
  for (const f of mesh.faces) {
    const a = f.vi[0] * 3,
      b = f.vi[1] * 3,
      c = f.vi[2] * 3;
    const p0: Vec3 = [mesh.positions[a], mesh.positions[a + 1], mesh.positions[a + 2]];
    const p1: Vec3 = [mesh.positions[b], mesh.positions[b + 1], mesh.positions[b + 2]];
    const p2: Vec3 = [mesh.positions[c], mesh.positions[c + 1], mesh.positions[c + 2]];
    const e1: Vec3 = [p1[0] - p0[0], p1[1] - p0[1], p1[2] - p0[2]];
    const e2: Vec3 = [p2[0] - p0[0], p2[1] - p0[1], p2[2] - p0[2]];
    const pv: Vec3 = [
      ray.d[1] * e2[2] - ray.d[2] * e2[1],
      ray.d[2] * e2[0] - ray.d[0] * e2[2],
      ray.d[0] * e2[1] - ray.d[1] * e2[0],
    ];
    const det = e1[0] * pv[0] + e1[1] * pv[1] + e1[2] * pv[2];
    if (Math.abs(det) < EPS) continue;
    const inv = 1 / det;
    const tv: Vec3 = [ray.o[0] - p0[0], ray.o[1] - p0[1], ray.o[2] - p0[2]];
    const u = (tv[0] * pv[0] + tv[1] * pv[1] + tv[2] * pv[2]) * inv;
    if (u < 0 || u > 1) continue;
    const qv: Vec3 = [
      tv[1] * e1[2] - tv[2] * e1[1],
      tv[2] * e1[0] - tv[0] * e1[2],
      tv[0] * e1[1] - tv[1] * e1[0],
    ];
    const v = (ray.d[0] * qv[0] + ray.d[1] * qv[1] + ray.d[2] * qv[2]) * inv;
    if (v < 0 || u + v > 1) continue;
    const t = (e2[0] * qv[0] + e2[1] * qv[1] + e2[2] * qv[2]) * inv;
    if (t <= 1e-5) continue;
    if (!best || t < best.dist)
      best = {
        faceId: f.id,
        dist: t,
        point: [ray.o[0] + ray.d[0] * t, ray.o[1] + ray.d[1] * t, ray.o[2] + ray.d[2] * t],
      };
  }
  return best;
}

export interface PatchStats {

  coneDeg: number; // max angular deviation from mean

export function cylRadiusAbout(axis: AxisIdx, p: Vec3): number {
  if (axis === 0) return Math.hypot(p[1], p[2]);
  if (axis === 1) return Math.hypot(p[0], p[2]);
  return Math.hypot(p[0], p[1]);
}
export function cylLabel(axis: AxisIdx): string {
  return axis === 0 ? "cyl_radius_yz(0,0)" : axis === 1 ? "cyl_radius_xz(0,0)" : "cyl_radius_xy(0,0)";
}
export function patchStats(mesh: Mesh, ids: number[]): PatchStats {
  const set = new Set(ids);
  const faces = mesh.faces.filter((f) => set.has(f.id));
  const n: Vec3 = [0, 0, 0];
  const centroid: Vec3 = [0, 0, 0];
  const bbox = { min: [1e9, 1e9, 1e9] as Vec3, max: [-1e9, -1e9, -1e9] as Vec3 };
  const gcount = new Map<string, number>();
  const tags = new Set<string>();
  const rmin = [1e9, 1e9, 1e9];
  const rmax = [-1e9, -1e9, -1e9];
  const rsum = [0, 0, 0];
  for (const f of faces) {
    n[0] += f.n[0] * f.area;
    n[1] += f.n[1] * f.area;
    n[2] += f.n[2] * f.area;
    centroid[0] += f.c[0] * f.area;
    centroid[1] += f.c[1] * f.area;
    centroid[2] += f.c[2] * f.area;
    for (let a = 0; a < 3; a++) {
      if (f.c[a] < bbox.min[a]) bbox.min[a] = f.c[a];
      if (f.c[a] > bbox.max[a]) bbox.max[a] = f.c[a];
      const r = cylRadiusAbout(a as AxisIdx, f.c);
      if (r < rmin[a]) rmin[a] = r;
      if (r > rmax[a]) rmax[a] = r;
      rsum[a] += r;
    }
    gcount.set(f.group, (gcount.get(f.group) ?? 0) + 1);
    tags.add(f.tag);
  }
  const area = faces.reduce((s, f) => s + f.area, 0) || 1;
  const nl = Math.hypot(n[0], n[1], n[2]) || 1;
  const meanN: Vec3 = [n[0] / nl, n[1] / nl, n[2] / nl];
  centroid[0] /= area;
  centroid[1] /= area;
  centroid[2] /= area;
  let cone = 0;
  const rMean: [number, number, number] = [0, 0, 0];
  const rSpread: [number, number, number] = [0, 0, 0];
  for (let a = 0; a < 3; a++) rMean[a] = rsum[a] / Math.max(1, faces.length);
  let rsq = [0, 0, 0];
  for (const f of faces) {
    const d = f.n[0] * meanN[0] + f.n[1] * meanN[1] + f.n[2] * meanN[2];
    cone = Math.max(cone, (Math.acos(Math.max(-1, Math.min(1, d))) * 180) / Math.PI);
    for (let a = 0; a < 3; a++) {
      const r = cylRadiusAbout(a as AxisIdx, f.c);
      rsq[a] += (r - rMean[a]) ** 2;
    }
  }
  for (let a = 0; a < 3; a++) rSpread[a] = Math.sqrt(rsq[a] / Math.max(1, faces.length));
  return {
    ids,
    n: meanN,
    coneDeg: cone,
    centroid,
    bbox,
    groups: [...gcount.entries()].map(([id, n]) => ({ id: id as GroupId, n })).sort((a, b) => b.n - a.n),
    tags: [...tags],
    rAbout: [
      [rmin[0], rmax[0]],
      [rmin[1], rmax[1]],
      [rmin[2], rmax[2]],
    ],
    rMean,
    rSpread,
  };
}

export type Clause =

  | { k: "cyl"; axis: AxisIdx; lo: number; hi: number }
  | { k: "span"; axis: AxisIdx; lo: number; hi: number }

function fmtNum(v: number): string {
  if (v === 0) return "0";
  const a = Math.abs(v);
  if (a < 1e-3 || a >= 1e5) {
    const e = v.toExponential(3);
    return e.replace(/e([+-])(\d)$/, "e$1$2").replace(/(\.\d*?)0+e/, "$1e").replace(/\.e/, "e");
  }
  return v.toFixed(Math.max(0, Math.min(6, 4 - Math.floor(Math.log10(a)) + 1)));
}
export function clauseText(c: Clause): string {
  switch (c.k) {
    case "normal": {
      const ax = AXIS_NAMES[c.axis];
      const v = c.hi < 0 ? c.hi : c.lo;
      const bound = Math.abs(c.hi) < Math.abs(c.lo) ? c.hi : c.lo;
      if (c.lo <= -0.999) return `normal.${ax} < ${bound.toFixed(3)}`;
      if (c.hi >= 0.999) return `normal.${ax} > ${bound.toFixed(3)}`;
      return `abs(normal.${ax} - ${v.toFixed(3)}) < ${(c.hi - c.lo) / 2 < 0.01 ? "0.005" : ((c.hi - c.lo) / 2).toFixed(3)}`;
    }
    case "plane": {
      const ax = AXIS_NAMES[c.axis];
      const lhs = c.bboxRef ? `bbox.${ax}_${c.value <= 0 ? "min" : "max"}` : fmtNum(c.value);
      return `abs(${ax} - ${lhs}) < ${fmtNum(c.tol)}`;
    }
    case "cyl":
      return `${cylLabel(c.axis)} in [${fmtNum(c.lo)}, ${fmtNum(c.hi)}]`;
    case "span":
      return `${AXIS_NAMES[c.axis]} in [${fmtNum(c.lo)}, ${fmtNum(c.hi)}]`;
    case "tag":
      return `step_tag in {${c.tags.join(", ")}}`;
  }
}
export function clauseMatch(c: Clause, f: Face, mesh: Mesh): boolean {
  switch (c.k) {
    case "normal": {
      const v = f.n[c.axis];
      return v >= c.lo - 1e-9 && v <= c.hi + 1e-9;
    }
    case "plane":
      return Math.abs(f.c[c.axis] - c.value) <= c.tol;
    case "cyl": {
      const r = cylRadiusAbout(c.axis, f.c);
      return r >= c.lo && r <= c.hi;
    }
    case "span":
      return f.c[c.axis] >= c.lo && f.c[c.axis] <= c.hi;
    case "tag":
      return c.tags.includes(f.tag);
  }
}
export function ruleMatch(rule: Clause[], f: Face, mesh: Mesh): boolean {
  return rule.every((c) => clauseMatch(c, f, mesh));
}

export const FEATURE_BANDS: { axis: AxisIdx; lo: number; hi: number; name: string }[] = [
  { axis: 0, lo: 0.018, hi: 0.022, name: "shaft_band" },
  { axis: 2, lo: 0.1385, hi: 0.1405, name: "tip_clearance" },

export function synthesize(mesh: Mesh, stats: PatchStats): Proposal {
  const basis: string[] = [];
  const diag = Math.hypot(
    mesh.bbox.max[0] - mesh.bbox.min[0],
    mesh.bbox.max[1] - mesh.bbox.min[1],
    mesh.bbox.max[2] - mesh.bbox.min[2],
  );
  const nFaces = stats.ids.length;
  const groups = stats.groups;
  const purity = nFaces ? groups[0].n / nFaces : 0;
  basis.push(`normal cone θ = ${stats.coneDeg.toFixed(2)}° over ${nFaces} faces`);
  basis.push(`group purity ${(purity * 100).toFixed(1)}%  (${groups[0]?.id ?? "—"})`);
  basis.push(
    `centroid (${stats.centroid.map((v) => v.toFixed(4)).join(", ")})  ·  area ${(
      nFaces * 0.00042
    ).toFixed(1)} mm²`,
  );
  const primary: Clause[] = [];
  const alts: Clause[] = [];
  /* --- 1. tight normal cone -> plane-normal clause ---------------------- */
  const ax = (Math.abs(stats.n[0]) > Math.abs(stats.n[1])
    ? Math.abs(stats.n[0]) > Math.abs(stats.n[2])
      ? 0
      : 2
    : Math.abs(stats.n[1]) > Math.abs(stats.n[2])
      ? 1
      : 2) as AxisIdx;
  const dom = stats.n[ax];
  if (stats.coneDeg < 12 && Math.abs(dom) > 0.9) {
    const lo = dom < 0 ? -1 : Math.max(0, dom - 0.01);
    const hi = dom < 0 ? Math.min(-0.9, dom + 0.01) : 1;
    primary.push({ k: "normal", axis: ax, lo: Math.max(-1, lo), hi: Math.min(1, hi) });
  }
  /* --- 2. geometry candidate: cylinder > plane > span ------------------- */
  let geom: Clause | null = null;
  let geomKind = "";
  // cylindrical: radius about an axis is (quasi) constant along the patch
  let bestCyl = -1;
  let bestCylAxis: AxisIdx = 0;
  for (let a = 0; a < 3; a++) {
    const mean = stats.rMean[a];
    const spread = stats.rSpread[a];
    if (mean < 1e-6) continue;
    const score = 1 - Math.min(1, spread / (0.25 * mean));
    // radial alignment bonus
    let align = 0;
    for (const id of stats.ids) {
      const f = mesh.faces[id];
      const r = [f.c[(a + 1) % 3], f.c[(a + 2) % 3]];
      const rl = Math.hypot(r[0], r[1]);
      if (rl < 1e-9) continue;
      const radial = [0, 0, 0];
      radial[(a + 1) % 3] = r[0] / rl;
      radial[(a + 2) % 3] = r[1] / rl;
      align += f.n[0] * radial[0] + f.n[1] * radial[1] + f.n[2] * radial[2];
    }
    align /= Math.max(1, stats.ids.length);
    const total = score * 0.7 + Math.max(0, align) * 0.3;
    if (total > bestCyl) {
      bestCyl = total;
      bestCylAxis = a as AxisIdx;
    }
  }
  if (bestCyl > 0.55) {
    const a = bestCylAxis;
    let lo = stats.rAbout[a][0];
    let hi = stats.rAbout[a][1];
    const pad = Math.max(1e-4, 0.02 * stats.rMean[a]);
    lo -= pad;
    hi += pad;
    const band = FEATURE_BANDS.find((b) => b.axis === a && lo >= b.lo - 0.15 * b.lo && hi <= b.hi + 0.15 * b.hi);
    if (band) {
      lo = band.lo;
      hi = band.hi;
      basis.push(`interval snapped to design band “${band.name}”`);
    }
    geom = { k: "cyl", axis: a as AxisIdx, lo, hi };
    geomKind = "cylindrical";
  }
  if (!geom) {
    // planar: patch sits on a global extreme face
    let cand: Clause | null = null;
    let candScore = -1;
    for (let a = 0; a < 3; a++) {
      const ext = stats.bbox.max[a] - stats.bbox.min[a];
      if (ext > 0.02 * diag) continue;
      const nd = stats.n[a];
      if (Math.abs(nd) < 0.985) continue;
      const atMin = nd < 0;
      const value = atMin ? stats.bbox.min[a] : stats.bbox.max[a];
      const onGlobal =
        Math.abs(value - mesh.bbox.min[a]) < 1e-6 || Math.abs(value - mesh.bbox.max[a]) < 1e-6;
      const spacing = Math.max(1e-4, ext * 0.6 + 1e-4);
      const score = Math.abs(nd) + (onGlobal ? 0.4 : 0) - ext / diag;
      if (score > candScore) {
        candScore = score;
        cand = {
          k: "plane",
          axis: a as AxisIdx,
          value: onGlobal ? (atMin ? mesh.bbox.min[a] : mesh.bbox.max[a]) : value,
          tol: spacing,
          bboxRef: onGlobal,
        };
      }
    }
    if (cand) {
      geom = cand;
      geomKind = "planar";
    }
  }
  if (!geom) {
    // confined along an axis -> span interval
    let bestA: AxisIdx = 0;
    let bestExt = Infinity;
    for (let a = 0; a < 3; a++) {
      const ext = stats.bbox.max[a] - stats.bbox.min[a];
      if (ext < bestExt) {
        bestExt = ext;
        bestA = a as AxisIdx;
      }
    }
    geom = {
      k: "span",
      axis: bestA,
      lo: stats.bbox.min[bestA] - 1e-4,
      hi: stats.bbox.max[bestA] + 1e-4,
    };
    geomKind = "span-interval";
  }
  primary.push(geom);
  /* --- alternatives ------------------------------------------------------ */
  if (geom.kind !== "tag") {
    alts.push({ k: "tag", tags: stats.tags.slice(0, 3) });
  }
  if (geom.kind !== "cyl" && bestCyl > 0.35) {
    const a = bestCylAxis;
    alts.push({
      k: "cyl",
      axis: a as AxisIdx,
      lo: stats.rAbout[a][0] - 1e-4,
      hi: stats.rAbout[a][1] + 1e-4,
    });
  }
  if (geom.kind !== "span") {
    const exts: [AxisIdx, number][] = [0, 1, 2].map((a) => [a as AxisIdx, stats.bbox.max[a] - stats.bbox.min[a]]);
    exts.sort((p, q) => p[1] - q[1]);
    alts.push({
      k: "span",
      axis: exts[0][0],
      lo: stats.bbox.min[exts[0][0]] - 1e-4,
      hi: stats.bbox.max[exts[0][0]] + 1e-4,
    });
  }
  if (geom.kind !== "plane" && Math.abs(dom) > 0.9) {
    alts.push({
      k: "plane",
      axis: ax,
      value: stats.bbox.min[ax],
      tol: Math.max(1e-4, 0.004 * diag),
      bboxRef:
        Math.abs(stats.bbox.min[ax] - mesh.bbox.min[ax]) < 1e-6 ||
        Math.abs(stats.bbox.max[ax] - mesh.bbox.max[ax]) < 1e-6,
    });
  }
  const conf = Math.max(
    0.2,
    Math.min(0.99, 0.55 + 0.3 * (1 - Math.min(1, stats.coneDeg / 25)) + 0.15 * purity),
  );
  basis.push(`candidate class: ${geomKind}  ·  confidence ${(conf * 100).toFixed(0)}%`);
  return { primary, alts, confidence: conf, basis };
}

const lerp = (a: number, b: number, t: number) => a + (b - a) * t;
type Pt = [number, number];
function interp(table: Pt[], x: number): number {
  if (x <= table[0][0]) return table[0][1];
  const last = table[table.length - 1];
  if (x >= last[0]) return last[1];
  for (let i = 0; i < table.length - 1; i++) {
    const [x0, y0] = table[i],
      [x1, y1] = table[i + 1];
    if (x <= x1) return lerp(y0, y1, (x - x0) / (x1 - x0));
  }
  return last[1];
}

export function lex(src: string): Tok[] {
  const out: Tok[] = [];
  let i = 0;
  while (i < src.length) {
    const c = src[i];
    if (/\s/.test(c)) {
      let j = i;
      while (j < src.length && /\s/.test(src[j])) j++;
      out.push({ k: "ws", t: src.slice(i, j), i });
      i = j;
      continue;
    }
    if (/[0-9.]/.test(c)) {
      let j = i;
      while (j < src.length && /[0-9._eE]/.test(src[j])) j++;
      out.push({ k: "num", t: src.slice(i, j), i });
      i = j;
      continue;
    }
    if (/[A-Za-z_]/.test(c)) {
      let j = i;
      while (j < src.length && /[A-Za-z0-9_]/.test(src[j])) j++;
      out.push({ k: "id", t: src.slice(i, j), i });
      i = j;
      continue;
    }
    if (c === "(") out.push({ k: "lp", t: c, i++ });
    else if (c === ")") out.push({ k: "rp", t: c, i++ });
    else if (c === ",") out.push({ k: "comma", t: c, i++ });
    else if ("+-*/^".includes(c)) out.push({ k: "op", t: c, i++ });
    else out.push({ k: "op", t: c, i++ });
  }

  }
  function unary(): Node {
    if (peek() && peek().t === "-") {
      p++;
      const a = unary();
      return { t: "neg", a, span: a.span };
    }
    if (peek() && peek().t === "+") {
      p++;
      return unary();
    }
    return power();
  }
  function power(): Node {
    const a = postfix();
    if (peek() && peek().t === "^") {
      p++;
      const nTok = toks[p];
      if (!nTok || nTok.k !== "num")
        throw new ParseErr("'^' requires a literal exponent", a.span, "e.g. U^2 or sqrt(...)");
      p++;
      return { t: "pow", a, n: parseFloat(nTok.t.replace(/_/g, "")), span: [a.span[0], nTok.i + nTok.t.length] };
    }
    return a;
  }
  function postfix(): Node {
    let n = primary();
    return n;
  }
  function primary(): Node {
    const k = peek();
    if (!k) throw new ParseErr("unexpected end of expression", [src.length, src.length]);
    if (k.k === "num") {
      p++;
      const v = parseFloat(k.t.replace(/_/g, ""));
      if (Number.isNaN(v)) throw new ParseErr(`malformed number “${k.t}”`, [k.i, k.i + k.t.length]);
      return { t: "num", v, span: [k.i, k.i + k.t.length] };
    }
    if (k.k === "id") {
      p++;
      if (peek() && peek().t === "(") {
        p++;
        const args: Node[] = [];
        if (peek() && peek().t !== ")") {
          args.push(expr());
          while (peek() && peek().t === ",") {
            p++;
            args.push(expr());
          }
        }
        const close = toks[p];
        if (!close || close.t !== ")")
          throw new ParseErr(`missing ')' after ${k.t}(…)`, [k.i, k.i + k.t.length], "balance the argument list");
        p++;
        return { t: "call", name: k.t, args, span: [k.i, close.i + 1] };
      }
      return { t: "id", name: k.t, span: [k.i, k.i + k.t.length] };
    }
    if (k.t === "(") {
      p++;
      const e = expr();
      const close = toks[p];
      if (!close || close.t !== ")") throw new ParseErr("missing ')'", [k.i, k.i + 1], "balance the parentheses");
      p++;
      return e;
    }
    throw new ParseErr(`unexpected “${k.t}”`, [k.i, k.i + Math.max(1, k.t.length)]);
  }
  const root = expr();
  if (p < toks.length)
    throw new ParseErr(`unexpected “${toks[p].t}”`, [toks[p].i, toks[p].i + toks[p].t.length], "trailing input after a complete expression");
  return root;
}

export function analyze(src: string): Analysis {
  const tokens = lex(src);
  let node: Node;
  try {
    node = parse(src);
  } catch (e) {
    const pe = e as ParseErr;
    return { ok: false, tokens, error: { message: pe.message, span: pe.span, hint: pe.hint } };
  }
  const check = (n: Node): { rank: Rank; d: Dims } => {
    switch (n.t) {
      case "num":
        return { rank: 0, d: D0 };
      case "id": {
        const s = SYMBOLS.find((x) => x.name === n.name);
        if (!s)
          throw new ParseErr(
            `unknown symbol “${n.name}”`,
            n.span,
            "known: " + SYMBOLS.map((x) => x.name).join(", "),
          );
        return { rank: s.rank, d: s.d };
      }
      case "neg":
        return check(n.a!);
      case "pow": {
        const a = check(n.a!);
        if (a.rank !== 0)
          throw new ParseErr(
            `cannot raise ${describe(a)} to a power`,
            n.span,
            "reduce to a scalar with mag() first",
          );
        if (!Number.isInteger(n.n) && n.n !== 0.5)
          throw new ParseErr(`exponent ${n.n} is not supported`, n.span, "use an integer or 0.5 (= sqrt)");
        if (!Number.isInteger(n.n) && mulD(a.d, n.n).some((x) => !Number.isInteger(x)))
          throw new ParseErr(
            `fractional exponent leaves non-integer dimensions [${fmtDims(a.d)}]^${n.n}`,
            n.span,
            "use sqrt() on an even-dimensioned quantity",
          );
        return { rank: 0, d: mulD(a.d, n.n) };
      }
      case "bin": {
        const l = check(n.a!);
        const r = check(n.b!);
        if (n.op === "+" || n.op === "-") {
          if (l.rank !== r.rank)
            throw new ParseErr(
              `cannot ${n.op === "+" ? "add" : "subtract"} ${describe(l)} and ${describe(r)}`,
              n.span,
              l.rank === 0 || r.rank === 0
                ? "one side is a scalar and the other a vector — use mag() or a component"
                : "rank mismatch",
            );
          if (!eqD(l.d, r.d)) {
            const fix1 = findFix(addD(r.d, l.d, -1));
            const fix2 = findFix(addD(l.d, r.d, -1));
            const verb = n.op === "+" ? "add" : "subtract";
            let hint: string;
            if (fix1)
              hint = `— multiply ${quantityName(r.d, r.rank).toLowerCase()} ${r.rank ? "" : ""}`.trimEnd();
            if (fix1)
              hint = `— multiply by ${capital(fix1.name)} [${unitOf(fix1.d)}] on the right-hand side`;
            else if (fix2) hint = `— multiply by ${capital(fix2.name)} [${unitOf(fix2.d)}] on the left-hand side`;
            else
              hint = `— left [${fmtDims(l.d)}] vs right [${fmtDims(r.d)}]; insert the missing conversion factor`;
            throw new ParseErr(`cannot ${verb} ${describe(l)} and ${describe(r)}`, n.span, hint);
          }
          return l;
        }
        if (n.op === "*") {
          if (l.rank !== 0 && r.rank !== 0)
            throw new ParseErr(
              `cannot multiply ${describe(l)} by ${describe(r)}`,
              n.span,
              "use dot(a,b) for the scalar product or cross(a,b) for the vector product",
            );
          return { rank: (l.rank || r.rank) as Rank, d: addD(l.d, r.d) };
        }
        // divide
        if (r.rank !== 0)
          throw new ParseErr(
            `cannot divide ${describe(l)} by ${describe(r)}`,
            n.span,
            "the divisor must be a scalar — take mag() of it first",
          );
        if (eqD(r.d, D0) && r.d.every((x) => x === 0) && l.rank !== 0) {
          /* dimensionless divisor is fine */
        }
        const dd = addD(l.d, r.d, -1);
        if (dd.some((x) => !Number.isInteger(x)))
          throw new ParseErr(
            `division produces fractional dimensions [${fmtDims(dd)}]`,
            n.span,
            "raise the denominator to an integer power first",
          );
        return { rank: l.rank, d: dd };
      }
      case "call": {
        const name = n.name!;
        const sig = FUNCTIONS[name];
        if (!sig)
          throw new ParseErr(
            `unknown function “${name}”`,
            n.span,
            "available: " + Object.keys(FUNCTIONS).join(", "),
          );
        const args = n.args!;
        if (args.length < sig.arity[0] || args.length > sig.arity[1])
          throw new ParseErr(
            `${name}() takes ${sig.arity[0] === sig.arity[1] ? sig.arity[0] : `${sig.arity[0]}–${sig.arity[1]}`} argument(s), got ${args.length}`,
            n.span,
            sig.doc,
          );
        const at = args.map(check);
        if (name === "dot" || name === "cross") {
          if (at[0].rank === 0 || at[1].rank === 0)
            throw new ParseErr(
              `${name}() needs two vectors, got ${describe(at[0])} and ${describe(at[1])}`,
              n.span,
              "wrap a scalar component in a vector literal or use '*'",
            );
          if (name === "dot" && at[0].rank !== at[1].rank)
            throw new ParseErr(`rank mismatch in dot(): ${describe(at[0])} · ${describe(at[1])}`, n.span, sig.doc);
        }
        if ((name === "mag" || name === "mag_sq" || name === "curl") && at[0].rank === 0)
          throw new ParseErr(`${name}() needs a vector, got ${describe(at[0])}`, n.span, sig.doc);
        if (name === "sqrt") {
          const d = at[0].d;
          if (at[0].rank !== 0)
            throw new ParseErr(`sqrt() of ${describe(at[0])} is not defined`, n.span, "take mag() first");
          if (d.some((x) => x % 2 !== 0))
            throw new ParseErr(
              `sqrt() needs even exponents, got [${fmtDims(d)}]`,
              n.span,
              "square the quantity first, or divide by a quantity with the same odd exponents",
            );
        }
        if ((name === "sin" || name === "cos" || name === "exp" || name === "log") && !eqD(at[0].d, D0))
          throw new ParseErr(
            `${name}() requires a dimensionless argument, got ${describe(at[0])}`,
            n.span,
            "normalise first, e.g. divide by a reference value",
          );
        if (name === "atan2" && !eqD(at[0].d, at[1].d))
          throw new ParseErr(
            `atan2() arguments disagree: ${describe(at[0])} vs ${describe(at[1])}`,
            n.span,
            "both operands must share dimensions",
          );
        return { rank: sig.rank(at.map((x) => x.rank)), d: sig.dims(at.map((x) => x.d)) };
      }
    }
  };
  try {
    const type = check(node);
    if (type.d.some((x) => !Number.isInteger(x)))
      throw new ParseErr(`non-integer dimension vector [${fmtDims(type.d)}]`, node.span, "this is a compiler bug");
    return { ok: true, node, type, tokens };
  } catch (e) {
    const pe = e as ParseErr;
    return { ok: false, tokens, error: { message: pe.message, span: pe.span ?? node.span, hint: pe.hint } };
  }
}

export function evaluate(n: Node): Val {
  switch (n.t) {
    case "num":
      return { rank: 0, d: D0, s: n.v! };
    case "id": {
      const s = SYMBOLS.find((x) => x.name === n.name)!;
      return { ...s.val };
    }
    case "neg": {
      const a = evaluate(n.a!);
      return { ...a, s: a.s !== undefined ? -a.s : undefined, v: a.v ? [-a.v[0], -a.v[1], -a.v[2]] : undefined };
    }
    case "pow": {
      const a = evaluate(n.a!);
      return { rank: 0, d: mulD(a.d, n.n!), s: Math.pow(a.s!, n.n!) };
    }
    case "bin": {
      const l = evaluate(n.a!);
      const r = evaluate(n.b!);
      const rank = (l.rank || r.rank) as Rank;
      if (n.op === "+" || n.op === "-") {
        const k = n.op === "+" ? 1 : -1;
        if (rank === 0) return { rank, d: l.d, s: l.s! + k * r.s! };
        if (rank === 1)
          return { rank, d: l.d, v: [l.v![0] + k * r.v![0], l.v![1] + k * r.v![1], l.v![2] + k * r.v![2]] };
        return { rank, d: l.d, t: l.t!.map((x, i) => x + k * r.t![i]) };
      }
      if (n.op === "*") {
        if (rank === 0) return { rank, d: addD(l.d, r.d), s: l.s! * r.s! };
        const k = r.s!;
        return { rank, d: addD(l.d, r.d), v: [l.v![0] * k, l.v![1] * k, l.v![2] * k] };
      }
      if (rank === 0) return { rank, d: addD(l.d, r.d, -1), s: l.s! / r.s! };
      const k = 1 / r.s!;
      return { rank, d: addD(l.d, r.d, -1), v: [l.v![0] * k, l.v![1] * k, l.v![2] * k] };
    }
    case "call": {
      const name = n.name!;
      const a = n.args!.map(evaluate);
      switch (name) {
        case "mag":
          return { rank: 0, d: a[0].d, s: Math.hypot(...a[0].v!) };
        case "mag_sq": {
          const v = a[0].v!;
          return { rank: 0, d: mulD(a[0].d, 2), s: v[0] ** 2 + v[1] ** 2 + v[2] ** 2 };
        }
        case "dot": {
          const x = a[0].v ?? [a[0].s!, a[0].s!, a[0].s!];
          const y = a[1].v ?? [a[1].s!, a[1].s!, a[1].s!];
          return { rank: 0, d: addD(a[0].d, a[1].d), s: x[0] * y[0] + x[1] * y[1] + x[2] * y[2] };
        }
        case "cross": {
          const x = a[0].v!,
            y = a[1].v!;
          return {
            rank: 1,
            d: addD(a[0].d, a[1].d),
            v: [x[1] * y[2] - x[2] * y[1], x[2] * y[0] - x[0] * y[2], x[0] * y[1] - x[1] * y[0]],
          };
        }
        case "grad": {
          if (a[0].rank === 0) {
            const key = n.args![0];
            const v = key.t === "id" && key.name === "T" ? GRAD_T : key.t === "id" && key.name === "p" ? GRAD_P : [0.8, -0.3, 0.11];
            return { rank: 1, d: addD(a[0].d, [0, -1, 0, 0, 0, 0, 0]), v: v as Vec };
          }
          return { rank: 2, d: addD(a[0].d, [0, -1, 0, 0, 0, 0, 0]), t: GRAD_U.slice() };
        }
        case "div": {
          const t = a[0].t!;
          return { rank: 0, d: addD(a[0].d, [0, 1, 0, 0, 0, 0, 0]), s: t[0] + t[4] + t[8] };
        }
        case "curl": {
          const t = a[0].t!;
          return {
            rank: 1,
            d: addD(a[0].d, [0, 1, 0, 0, 0, 0, 0]),
            v: [t[7] - t[5], t[2] - t[6], t[3] - t[1]],
          };
        }
        case "sqrt":
          return { rank: a[0].rank, d: mulD(a[0].d, 0.5), s: Math.sqrt(a[0].s!) };
        case "abs":
          return { ...a[0], s: Math.abs(a[0].s!) };
        case "sum":
        case "avg":
          return a[0];
        case "sin":
          return { rank: 0, d: D0, s: Math.sin(a[0].s!) };
        case "cos":
          return { rank: 0, d: D0, s: Math.cos(a[0].s!) };
        case "exp":
          return { rank: 0, d: D0, s: Math.exp(a[0].s!) };
        case "log":
          return { rank: 0, d: D0, s: Math.log(a[0].s!) };
        case "atan2":
          return { rank: 0, d: D0, s: Math.atan2(a[0].s!, a[1].s!) };
        default:
          return { rank: 0, d: D0, s: 0 };
      }
    }
  }
}

export function mul4(a: Float32Array, b: Float32Array): Float32Array {
  const o = new Float32Array(16);

    for (let r = 0; r < 4; r++) {
      let s = 0;
      for (let k = 0; k < 4; k++) s += a[k * 4 + r] * b[c * 4 + k];

function encodeCurve(field: string): { data: Uint8Array; min: number[]; max: number[] } {
  const data = new Uint8Array(CURVE_W * CURVE_H * 2);
  const min: number[] = [];
  const max: number[] = [];
  for (let r = 0; r < CURVE_H; r++) {
    const side = r < 2 ? 1 : -1;
    const run = r % 2;
    const fld = r < 4 ? "Cp" : "Tw";
    const vals: number[] = [];
    for (let i = 0; i < CURVE_W; i++) vals.push(chordValue(fld, side, run as 0 | 1, i / (CURVE_W - 1)));
    const lo = Math.min(...vals);
    const hi = Math.max(...vals);
    min.push(lo);
    max.push(hi);
    for (let i = 0; i < CURVE_W; i++) {
      const q = Math.round(((vals[i] - lo) / (hi - lo || 1)) * 65535);
      const o = (r * CURVE_W + i) * 2;
      data[o] = q & 255;
      data[o + 1] = (q >> 8) & 255;
    }
  }
  return { data, min, max };
}

function niceTicks(min: number, max: number, n = 5): number[] {
  const span = max - min || 1;
  const mag = Math.pow(10, Math.floor(Math.log10(span / n)));
  const norm = span / n / mag;
  const step = (norm >= 5 ? 5 : norm >= 2 ? 2 : 1) * mag;
  const out: number[] = [];
  for (let v = Math.ceil(min / step) * step; v <= max + step * 1e-6; v += step)
    out.push(Math.abs(v) < step * 1e-9 ? 0 : v);
  return out;
}
function tickLabel(v: number, log = false): string {
  if (log) return v >= 1 ? String(Math.round(v)) : v.toPrecision(1);
  const a = Math.abs(v);
  if (a >= 1e5 || (a > 0 && a < 1e-3)) return v.toExponential(1).replace("e+", "e");
  if (a >= 1000) return v.toFixed(0);
  if (a >= 10) return v.toFixed(0);
  if (a >= 1) return v.toFixed(1);
  return v.toFixed(2);
}

export function PolarChart({
  title,
  subtitle,
  series,
  rDomain,
  rLabel,
  rings = 4,
  markers = [],
  height = 250,
}: {
  title: string;
  subtitle?: string;
  series: Series[];
  rDomain: [number, number];
  rLabel: string;
  rings?: number;
  markers?: { th: number; label: string }[];
  height?: number;
}) {
  const S = Math.min(360, height * 1.5);
  const c = S / 2;
  const R = S / 2 - 34;
  const [r0, r1] = rDomain;
  const rr = (v: number) => ((v - r0) / (r1 - r0)) * R;
  const pt = (thDeg: number, v: number): [number, number] => {
    const t = ((thDeg - 90) * Math.PI) / 180;
    const r = rr(v);
    return [c + r * Math.cos(t), c + r * Math.sin(t)];
  };
  return (
    <figure className="flex min-w-0 flex-col">
      <figcaption className="mb-1 flex items-baseline justify-between">
        <span className="lbl text-[#767d92]">{title}</span>
        {subtitle && <span className="num text-[10px] text-[#4f5568]">{subtitle}</span>}
      </figcaption>
      <svg viewBox={`0 0 ${S} ${S}`} className="w-full" style={{ height: "auto" }}>
        {Array.from({ length: rings + 1 }, (_, i) => r0 + ((r1 - r0) * i) / rings).map((v, i) => (
          <g key={i}>
            <circle cx={c} cy={c} r={rr(v)} fill="none" stroke="#2a2c36" strokeWidth={0.8} />
            <text x={c + 3} y={c - rr(v) - 2} className="num" fontSize={8.5} fill="#5f6577">
              {v.toFixed(0)}
            </text>
          </g>
        ))}
        {[0, 45, 90, 135, 180, 225, 270, 315].map((a) => {
          const p = pt(a, r1);
          return (
            <g key={a}>
              <line x1={c} y1={c} x2={p[0]} y2={p[1]} stroke="#2a2c36" strokeWidth={0.6} />
              <text
                x={c + (p[0] - c) * 1.14}
                y={c + (p[1] - c) * 1.14 + 3}
                textAnchor="middle"
                className="num"
                fontSize={8.5}
                fill="#5f6577"
              >
                {a}°
              </text>
            </g>
          );
        })}
        {series.map((s) => (
          <path
            key={s.id}
            d={s.pts.map((p, i) => `${i ? "L" : "M"}${pt(p[0], p[1]).join(" ")}`).join(" ") + " Z"}
            fill={s.color}
            fillOpacity={0.08}
            stroke={s.color}
            strokeWidth={1.6}
            strokeDasharray={s.dash}
          />
        ))}
        {markers.map((m) => {
          const p = pt(m.th, r1);
          const inner = pt(m.th, r0);
          return (
            <g key={m.th}>
              <line x1={inner[0]} y1={inner[1]} x2={p[0]} y2={p[1]} stroke="#ffb020" strokeWidth={1} strokeDasharray="3 3" />
              <circle cx={p[0]} cy={p[1]} r={3} fill="#ffb020" />

  const d: Vec3 = [axis[0] / n, axis[1] / n, axis[2] / n];
  let up: Vec3 = Math.abs(d[2]) < 0.9 ? [0, 0, 1] : [1, 0, 0];
  let u: Vec3 = [
    up[1] * d[2] - up[2] * d[1],
    up[2] * d[0] - up[0] * d[2],
    up[0] * d[1] - up[1] * d[0],
  ];
  const ul = Math.hypot(...u) || 1;
  u = [u[0] / ul, u[1] / ul, u[2] / ul];
  const v: Vec3 = [
    d[1] * u[2] - d[2] * u[1],
    d[2] * u[0] - d[0] * u[2],
    d[0] * u[1] - d[1] * u[0],
  ];
  const r = length * Math.tan((halfAngle * Math.PI) / 180);
  const base: Vec3 = [apex[0] + d[0] * length, apex[1] + d[1] * length, apex[2] + d[2] * length];
  const circle: Vec3[] = [];
  for (let i = 0; i <= 48; i++) {
    const t = (i / 48) * Math.PI * 2;
    circle.push([
      base[0] + r * (u[0] * Math.cos(t) + v[0] * Math.sin(t)),
      base[1] + r * (u[1] * Math.cos(t) + v[1] * Math.sin(t)),
      base[2] + r * (u[2] * Math.cos(t) + v[2] * Math.sin(t)),
    ]);
  }
  const pc = circle.map((p) => ctx.project(p));
  const pa = ctx.project(apex);
  const pb = ctx.project(base);
  if (!pa || !pb || pc.some((p) => !p)) return null;
  const gens = [0, 12, 24, 36].map((i) => pc[i]!);
  const drops = Array.from({ length: 7 }, (_, i) => {
    const t = (i + 1) / 8;
    return ctx.project([
      apex[0] + d[0] * length * t,
      apex[1] + d[1] * length * t,
      apex[2] + d[2] * length * t,
    ]);
  });
  return (
    <g>
      <polyline points={pts(pc.map((p) => [p!.x, p!.y] as unknown as Vec3))} fill="#ffb020" fillOpacity={0.05} stroke="#ffb020" strokeWidth={1.1} />
      {gens.map((g, i) => (
        <line key={i} x1={pa.x} y1={pa.y} x2={g.x} y2={g.y} stroke="#ffb020" strokeWidth={0.9} strokeOpacity={0.7} />
      ))}
      <line x1={pa.x} y1={pa.y} x2={pb.x} y2={pb.y} stroke="#e9ebf2" strokeWidth={1} strokeDasharray="6 3" />
      {drops.map((p, i) =>
        p ? <circle key={i} cx={p.x} cy={p.y} r={1.8} fill="#ffb020" opacity={0.85 - i * 0.09} /> : null,
      )}
      <circle data-handle="inj-apex" cx={pa.x} cy={pa.y} r={5} fill="#0b0c10" stroke="#ffb020" strokeWidth={1.6}
        onPointerDown={(e) => {
          if (!dragApex) return;
          e.stopPropagation();
          const mv = (ev: PointerEvent) => dragApex(ev.clientX, ev.clientY);
          const up2 = () => { window.removeEventListener("pointermove", mv); window.removeEventListener("pointerup", up2); };
          window.addEventListener("pointermove", mv);
          window.addEventListener("pointerup", up2);
        }}
      />
      <circle data-handle="inj-base" cx={pb.x} cy={pb.y} r={4} fill="#0b0c10" stroke="#e9ebf2" strokeWidth={1.4}
        onPointerDown={(e) => {
          if (!dragBase) return;
          e.stopPropagation();
          const mv = (ev: PointerEvent) => dragBase(ev.clientX, ev.clientY);
          const up2 = () => { window.removeEventListener("pointermove", mv); window.removeEventListener("pointerup", up2); };
          window.addEventListener("pointermove", mv);
          window.addEventListener("pointerup", up2);
        }}
      />
      <text x={pb.x + 8} y={pb.y + 3} className="num" fontSize={9.5} fill="#ffcd6b">
        θcone {halfAngle.toFixed(1)}°
      </text>
    </g>
  );
}

export function MicRing({

export function PorousAxes({
  ctx,
  origin,
  yaw,
  pitch,
  roll,
  half = [0.09, 0.055, 0.045],
}: {
  ctx: OverlayCtx;
  origin: Vec3;
  yaw: number;
  pitch: number;
  roll: number;
  half?: Vec3;
}) {
  const ry = (yaw * Math.PI) / 180,
    rp = (pitch * Math.PI) / 180,
    rr = (roll * Math.PI) / 180;
  const rot = (p: Vec3): Vec3 => {
    let [x, y, z] = p;
    let a = Math.cos(rr) * y - Math.sin(rr) * z, b = Math.sin(rr) * y + Math.cos(rr) * z;
    y = a; z = b;
    a = Math.cos(rp) * x - Math.sin(rp) * z; b = Math.sin(rp) * x + Math.cos(rp) * z;
    x = a; z = b;
    a = Math.cos(ry) * x - Math.sin(ry) * y; b = Math.sin(ry) * x + Math.cos(ry) * y;
    x = a; y = b;
    return [x, y, z];
  };
  const basis: Vec3[] = [
    rot([1, 0, 0]),
    rot([0, 1, 0]),
    rot([0, 0, 1]),
  ];
  const corner = (s: Vec3) =>
    ctx.project([
      origin[0] + rot([s[0] * half[0], s[1] * half[1], s[2] * half[2]])[0],
      origin[1] + rot([s[0] * half[0], s[1] * half[1], s[2] * half[2]])[1],
      origin[2] + rot([s[0] * half[0], s[1] * half[1], s[2] * half[2]])[2],
    ]);
  const cs: ({ x: number; y: number } | null)[] = [];
  for (const sx of [-1, 1]) for (const sy of [-1, 1]) for (const sz of [-1, 1]) cs.push(corner([sx, sy, sz]));
  const E = [
    [0, 1], [0, 2], [0, 4], [1, 3], [1, 5], [2, 3], [2, 6], [3, 7], [4, 5], [4, 6], [5, 7], [6, 7],
  ];
  const o = ctx.project(origin);
  if (!o || cs.some((c) => !c)) return null;
  const colors = ["#ef233c", "#4ade80", "#5b8cff"];
  const labels = ["e₁", "e₂", "e₃"];
  return (
    <g>
      {E.map(([a, b], i) => (
        <line key={i} x1={cs[a]!.x} y1={cs[a]!.y} x2={cs[b]!.x} y2={cs[b]!.y} stroke="#4a5165" strokeWidth={0.9} strokeOpacity={0.85} />
      ))}
      {basis.map((bv, i) => {
        const tip = ctx.project([
          origin[0] + bv[0] * 0.13,
          origin[1] + bv[1] * 0.13,
          origin[2] + bv[2] * 0.13,
        ]);
        if (!tip) return null;
        return (
          <g key={i}>
            <line x1={o.x} y1={o.y} x2={tip.x} y2={tip.y} stroke={colors[i]} strokeWidth={1.8} strokeLinecap="round" />
            <text x={tip.x + 5} y={tip.y + 3} className="num" fontSize={10} fill={colors[i]}>
              {labels[i]}
            </text>
          </g>
        );
      })}
    </g>
  );
}

export function LineProbe({
  ctx,
  pA,
  pB,
  label = "PROBE",
  onDragA,
  onDragB,
}: {
  ctx: OverlayCtx;
  pA: Vec3;
  pB: Vec3;
  label?: string;
  onDragA?: (mx: number, my: number) => void;
  onDragB?: (mx: number, my: number) => void;
}) {
  const a = ctx.project(pA);
  const b = ctx.project(pB);
  if (!a || !b) return null;
  const ticks = Array.from({ length: 11 }, (_, i) => i / 10).map((t) => {
    const w: Vec3 = [pA[0] + (pB[0] - pA[0]) * t, pA[1] + (pB[1] - pA[1]) * t, pA[2] + (pB[2] - pA[2]) * t];
    return ctx.project(w);
  });
  const handle = (p: { x: number; y: number }, fn?: (x: number, y: number) => void, color: string, key: string) => (
    <circle
      key={key}
      data-handle={key}
      cx={p.x} cy={p.y} r={6}
      fill="#0b0c10" stroke={color} strokeWidth={2}
      style={{ cursor: "grab", pointerEvents: "auto" }}
      onPointerDown={(e) => {
        if (!fn) return;
        e.stopPropagation();
        const mv = (ev: PointerEvent) => fn(ev.clientX, ev.clientY);
        const up = () => { window.removeEventListener("pointermove", mv); window.removeEventListener("pointerup", up); };
        window.addEventListener("pointermove", mv);
        window.addEventListener("pointerup", up);
      }}
    />
  );
  return (
    <g>
      <line x1={a.x} y1={a.y} x2={b.x} y2={b.y} stroke="#ffb020" strokeWidth={1.8} />
      {ticks.map((t, i) =>
        t ? (
          <line key={i} x1={t.x - 3} y1={t.y - 3} x2={t.x + 3} y2={t.y + 3} stroke="#ffb020" strokeWidth={1} opacity={0.8} />
        ) : null,
      )}
      <g transform={`translate(${(a.x + b.x) / 2 + 8}, ${(a.y + b.y) / 2 - 8})`}>
        <rect x={-4} y={-11} width={92} height={15} fill="#0b0c10e6" stroke="#6b4d10" />
        <text x={2} y={0} className="num" fontSize={9.5} fill="#ffcd6b">{label}</text>
      </g>
      {handle(a, onDragA, "#ffb020", "probe-a")}
      {handle(b, onDragB, "#ffb020", "probe-b")}
    </g>
  );
}

export function SlicePlane({

  onDrag,

  ctx: OverlayCtx;

  half?: number;
  onDrag?: (dx: number, dy: number) => void;

  ];

  if (p.some((q) => !q)) return null;
  const mid = ctx.project([x, half, 0]);
  return (

      <polygon
        points={pts(p.map((q) => [q!.x, q!.y] as unknown as Vec3))}
        fill="#004afd" fillOpacity={0.07}

  );

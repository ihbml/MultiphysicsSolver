cmake_minimum_required(VERSION 3.16)
project(HybridMesh VERSION 0.1.0 LANGUAGES CXX)
option(HYBRIDMESH_OPENMP "Parallelize independent particle/transfer loops" OFF)
add_library(hybridmesh STATIC src/mesh.cpp src/io.cpp src/sph.cpp src/transfer.cpp)
target_include_directories(hybridmesh PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_include_directories(hybridmesh PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(hybridmesh PUBLIC cxx_std_20)
if(HYBRIDMESH_OPENMP)
  find_package(OpenMP REQUIRED)
  target_link_libraries(hybridmesh PUBLIC OpenMP::OpenMP_CXX)
endif()
add_executable(hybridmesh_cli src/cli.cpp)
target_link_libraries(hybridmesh_cli PRIVATE hybridmesh)
set_target_properties(hybridmesh_cli PROPERTIES OUTPUT_NAME hybridmesh)
enable_testing()
add_executable(hybridmesh_tests tests/test.cpp)
target_compile_definitions(hybridmesh_tests PRIVATE
  "HYBRIDMESH_SOURCE_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}\"")
target_link_libraries(hybridmesh_tests PRIVATE hybridmesh)
add_test(NAME hybridmesh_tests COMMAND hybridmesh_tests)
install(TARGETS hybridmesh hybridmesh_cli)
install(DIRECTORY include/ DESTINATION include)

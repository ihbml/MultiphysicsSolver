  std::size_t seg[5]{}; // sorted externally; no kernel reorders the lake

  Span<const double> volume;

struct FaceGeometry { double beta, alpha, orthogonal_mag; Vec3 d,k; int correction_class; };
inline FaceGeometry make_geometry(const MeshView& m, std::size_t f,
                                  double orthogonality_tol=1e-12) {
  const auto P=m.owner[f]; const auto N=m.neighbour[f];
  const Vec3 d=m.d(f), S=m.S[f];
  const double dS=dot(d,S);
  if (!(dS>0.0)) throw std::runtime_error("inverted face: d.S <= 0");
  const double S2=dot(S,S);
  const double beta=N==boundary ? 1.0 : dot(m.xf[f]-m.x[P],S)/dS;
  const double alpha=S2/dS;                    // over-relaxed conductance
  const Vec3 k=S-alpha*d;                      // deferred non-orthogonal correction
  const double rel=norm(k)/std::sqrt(S2);
  return {beta,alpha,alpha*dS,d,k,rel>orthogonality_tol ? 1 : 0};
}
inline void build_correction(const MeshView& m, CorrectionSoA& c) {
  for (std::size_t f=0;f<m.faces;++f) {
    const auto g=make_geometry(m,f);
    const Vec3 S=m.S[f];
    c.sx[f]=float(S.x); c.sy[f]=float(S.y); c.sz[f]=float(S.z);
    // Class 0 uses compact gradient-free upwind and stores no deferred vector.
    const Vec3 k=g.correction_class ? g.k : Vec3{};
    c.kx[f]=float(k.x); c.ky[f]=float(k.y); c.kz[f]=float(k.z);
    c.omega[f]=1.0f;
  }
}
struct CellFaceCSR {
  std::vector<Index> offset, face;
  std::vector<signed char> orientation;
  static CellFaceCSR build(const MeshView& m) {
    CellFaceCSR c;
    c.offset.assign(m.owned+1,0);
    for (std::size_t f=0;f<m.faces;++f) {
      const Index P=m.owner[f], N=m.neighbour[f];
      if (P>=0 && std::size_t(P)<m.owned) ++c.offset[P+1];
      if (N>=0 && std::size_t(N)<m.owned) ++c.offset[N+1];
    }
    std::partial_sum(c.offset.begin(),c.offset.end(),c.offset.begin());
    c.face.resize(c.offset.back()); c.orientation.resize(c.offset.back());
    std::vector<Index> cursor(c.offset.begin(),c.offset.end()-1);
    // Face order is the Hilbert/quadrant order supplied by the data lake.
    for (std::size_t f=0;f<m.faces;++f) {
      const Index P=m.owner[f], N=m.neighbour[f];
      if (P>=0 && std::size_t(P)<m.owned) {
        const Index q=cursor[P]++; c.face[q]=Index(f); c.orientation[q]=1;
      }
      if (N>=0 && std::size_t(N)<m.owned) {
        const Index q=cursor[N]++; c.face[q]=Index(f); c.orientation[q]=-1;
      }
    }
    return c;
  }
};
// Atomic-free and deterministic: a thread owns all accumulations for one cell.
inline void gather_cell_residual(const CellFaceCSR& csr, std::size_t owned,
                                 const double* face_flux,
                                 const double* diag_P, const double* diag_N,
                                 double* residual, double* diagonal) {
  #pragma omp parallel for schedule(static)
  for (std::ptrdiff_t i=0;i<std::ptrdiff_t(owned);++i) {
    double r=0.0,a=0.0;
    for (Index q=csr.offset[i];q<csr.offset[i+1];++q) {
      const Index f=csr.face[q];
      r += double(csr.orientation[q])*face_flux[f];
      a += csr.orientation[q]>0 ? diag_P[f] : diag_N[f];
    }
    residual[i]=r; diagonal[i]=a;
  }
}

inline bool dense_solve(std::vector<double> A, std::vector<double>& b, std::size_t n) {
  for (std::size_t k=0;k<n;++k) {
    std::size_t pivot=k;
    for (std::size_t i=k+1;i<n;++i)
      if (std::abs(A[i*n+k])>std::abs(A[pivot*n+k])) pivot=i;
    if (std::abs(A[pivot*n+k])<1e-14) return false;
    if (pivot!=k) {
      for (std::size_t j=k;j<n;++j) std::swap(A[k*n+j],A[pivot*n+j]);
      std::swap(b[k],b[pivot]);
    }
    for (std::size_t i=k+1;i<n;++i) {
      const double m=A[i*n+k]/A[k*n+k];
      A[i*n+k]=0.0;
      for (std::size_t j=k+1;j<n;++j) A[i*n+j]-=m*A[k*n+j];
      b[i]-=m*b[k];
    }
  }
  for (std::size_t ii=n;ii-->0;) {
    for (std::size_t j=ii+1;j<n;++j) b[ii]-=A[ii*n+j]*b[j];
    b[ii]/=A[ii*n+ii];
  }
  return true;
}
struct Csr {
  Index n{};
  std::vector<Index> row{0}, col;
  std::vector<double> a;
  void matvec(const double* x, double* y) const {
    #pragma omp parallel for schedule(static)
    for (std::ptrdiff_t i=0;i<n;++i) {
      double s=0.0;
      for (Index q=row[i];q<row[i+1];++q) s+=a[q]*x[col[q]];
      y[i]=s;
    }
  }
  double diagonal(Index i) const {
    for (Index q=row[i];q<row[i+1];++q) if (col[q]==i) return a[q];
    return 0.0;
  }
  std::vector<double>& operator()(std::vector<double>& x) { return x; }
};

}

KrylovStats bicgstab(const Csr& A,const std::vector<double>& b,std::vector<double>& x,
                     double rtol=1e-10,int max_iter=200,
                     const Preconditioner& pre=IdentityPreconditioner{}) {
  const std::size_t n=b.size();
  std::vector<double> Ax(n),r(n),r0,p(n,0.0),v(n,0.0),s,t,ph(n),sh(n);
  A.matvec(x.data(),Ax.data());
  for (std::size_t i=0;i<n;++i) r[i]=b[i]-Ax[i];
  r0=r; s=r; t=r;
  const double bn=norm2(b), An=inf_norm(A);
  KrylovStats st; st.residual=norm2(r); st.true_residual=st.residual;
  if (st.residual<=rtol*std::max(bn,1e-300)) { st.converged=true; return st; }
  double rho=1,alpha=1,omega=1,previous=st.residual;
  int stale=0;
  for (int it=1;it<=max_iter;++it) {
    const double rho1=dot(r0,r);
    if (std::abs(rho1)<1e-300 || std::abs(omega)<1e-300) { st.stagnated=true; break; }
    const double beta=(rho1/rho)*(alpha/omega); rho=rho1;
    for (std::size_t i=0;i<n;++i) p[i]=r[i]+beta*(p[i]-omega*v[i]);
    pre.apply(p,ph); A.matvec(ph.data(),v.data());
    const double rv=dot(r0,v);
    if (std::abs(rv)<1e-300) { st.stagnated=true; break; }
    alpha=rho/rv;
    for (std::size_t i=0;i<n;++i) s[i]=r[i]-alpha*v[i];
    if (norm2(s)<=rtol*std::max(bn,1e-300)) {
      axpy(alpha,ph,x); st.iterations=it; st.converged=true; break;
    }
    pre.apply(s,sh); A.matvec(sh.data(),t.data());
    const double tt=dot(t,t);
    if (tt<1e-300) { st.stagnated=true; break; }
    omega=dot(t,s)/tt;
    for (std::size_t i=0;i<n;++i) x[i]+=alpha*ph[i]+omega*sh[i];
    for (std::size_t i=0;i<n;++i) r[i]=s[i]-omega*t[i];
    st.iterations=it;
    // A-posteriori residual replacement detects loss of recurrence accuracy.
    if (it%5==0 || it==max_iter) {
      A.matvec(x.data(),Ax.data());
      std::vector<double> rt(n);
      for (std::size_t i=0;i<n;++i) rt[i]=b[i]-Ax[i];
      st.true_residual=norm2(rt);
      st.eta_round=norm2([&]{std::vector<double> e(n);for(std::size_t i=0;i<n;++i)e[i]=r[i]-rt[i];return e;}()) /
                   (bn+An*norm2(x)+std::numeric_limits<double>::epsilon());
      r=rt;
      if (std::abs(previous-st.true_residual)<=1e-3*st.true_residual) ++stale; else stale=0;
      previous=st.true_residual;
      if (stale>=3 || st.eta_round>0.25) { st.stagnated=true; break; }
    }
    st.residual=norm2(r);
    if (st.residual<=rtol*std::max(bn,1e-300)) { st.converged=true; break; }
    rho=rho1;
  }
  A.matvec(x.data(),Ax.data());
  for (std::size_t i=0;i<n;++i) r[i]=b[i]-Ax[i];
  st.true_residual=norm2(r); st.residual=st.true_residual;
  st.eta_round=std::abs(st.residual-st.true_residual)/(bn+An*norm2(x)+1e-300);
  st.converged=st.true_residual<=rtol*std::max(bn,1e-300);
  return st;
}

    Csr B; B.n=A.n; B.row.assign(A.n+1,0);

      std::map<Index,double> row;
      for (auto [i,p]:pt[I]) for (Index q=B.row[i];q<B.row[i+1];++q)

      for (auto [j,v]:row) if (std::abs(v)>1e-14*scale) {

      }
      Ac.row[I+1]=Index(Ac.col.size());
    }

  static std::pair<Csr,Index> aggregate(const Csr& A,double theta,bool smoothed) {

        bool maximal=true;
        for (Index j:strong[i]) if (agg[j]<0 && (weight[j]>weight[i] ||
            (weight[j]==weight[i] && j>i))) { maximal=false; break; }
        if (maximal) seeds.push_back(i);
      }
      if (seeds.empty()) for (Index i=0;i<n;++i) if (agg[i]<0) { seeds.push_back(i); break; }
      for (Index i:seeds) if (agg[i]<0) {
        const Index g=nc++; agg[i]=g; --remaining;
        for (Index j:strong[i]) if (agg[j]<0) { agg[j]=g; --remaining; }
      }
    }
    Csr P0; P0.n=n; P0.row.resize(n+1); P0.col.resize(n); P0.a.assign(n,1.0);
    for (Index i=0;i<n;++i) { P0.row[i]=i; P0.col[i]=agg[i]; }
    if (!smoothed) return {std::move(P0),nc};
    // P=(I-omega*D^-1*A)P0, omega=4/(3*rho(D^-1 A)).
    std::vector<double> x(n,1.0),y(n); double rho=1.0;
    for (int k=0;k<12;++k) {
      for (Index i=0;i<n;++i) {
        double s=0.0; for (Index q=A.row[i];q<A.row[i+1];++q) s+=A.a[q]*x[A.col[q]];
        y[i]=s/A.diagonal(i);
      }
      rho=norm2(y); if (rho==0.0) break;
      for (Index i=0;i<n;++i) x[i]=y[i]/rho;
    }
    const double omega=4.0/(3.0*std::max(rho,1e-12));
    Csr P; P.n=n; P.row.assign(n+1,0);
    for (Index i=0;i<n;++i) {
      std::map<Index,double> row; row[agg[i]]+=1.0;
      const double di=A.diagonal(i);
      for (Index q=A.row[i];q<A.row[i+1];++q)
        row[agg[A.col[q]]]-=omega*A.a[q]/di;
      for (auto [j,v]:row) if (std::abs(v)>1e-14) { P.col.push_back(j); P.a.push_back(v); }
      P.row[i+1]=Index(P.col.size());
    }
    return {std::move(P),nc};
  }
  double cast(double x) const { return opt_.fp32_inner ? double(float(x)) : x; }
  void matvec(const Level& L,const std::vector<double>& x,std::vector<double>& y) const {
    y.assign(L.A.n,0.0);
    for (Index i=0;i<L.A.n;++i) {
      double s=0.0; for (Index q=L.A.row[i];q<L.A.row[i+1];++q)
        s=cast(s+cast(L.A.a[q]*x[L.A.col[q]]));
      y[i]=s;
    }
  }
  void smooth(const Level& L,const std::vector<double>& b,std::vector<double>& x) const {
    const double hi=L.lambda_max, lo=0.30*hi;
    const double mid=0.5*(hi+lo), half=0.5*(hi-lo);
    std::vector<double> Ax,r;
    for (int k=0;k<opt_.chebyshev_degree;++k) {
      // Roots of the shifted Chebyshev polynomial: a product of Richardson steps.
      const double theta=std::numbers::pi*(k+0.5)/opt_.chebyshev_degree;
      const double root=mid+half*std::cos(theta);
      matvec(L,x,Ax); r.resize(b.size());
      for (std::size_t i=0;i<b.size();++i) r[i]=cast(b[i]-Ax[i]);
      for (std::size_t i=0;i<b.size();++i)
        x[i]=cast(x[i]+cast(L.dinv[i]*r[i]/root));
    }
  }
  void cycle(int l,const std::vector<double>& b,std::vector<double>& x) const {
    const Level& L=lv_[l]; x.assign(b.size(),0.0);
    if (l+1==int(lv_.size())) {
      std::vector<double> D(b.size()*b.size()),rhs=b;
      for (Index i=0;i<L.A.n;++i) for (Index q=L.A.row[i];q<L.A.row[i+1];++q)
        D[i*b.size()+L.A.col[q]]=L.A.a[q];
      dense_solve(std::move(D),rhs,b.size());
      for (double& v:rhs) v=cast(v); x=std::move(rhs); return;
    }
    smooth(L,b,x);
    std::vector<double> Ax,r,bc(lv_[l+1].A.n,0.0);
    matvec(L,x,Ax); r.resize(b.size());
    for (std::size_t i=0;i<b.size();++i) r[i]=cast(b[i]-Ax[i]);
    for (Index i=0;i<L.A.n;++i) for (Index q=L.P.row[i];q<L.P.row[i+1];++q)
      bc[L.P.col[q]]=cast(bc[L.P.col[q]]+cast(L.P.a[q]*r[i]));
    std::vector<double> ec; cycle(l+1,bc,ec);
    for (Index i=0;i<L.A.n;++i) for (Index q=L.P.row[i];q<L.P.row[i+1];++q)
      x[i]=cast(x[i]+cast(L.P.a[q]*ec[L.P.col[q]]));
    smooth(L,b,x);
  }
public:
  explicit AmgPreconditioner(const Csr& fine,AmgOptions opt={}) : opt_(opt) {
    Csr A=fine;
    for (int l=0;l<opt_.max_levels;++l) {
      Level L; L.A=A; L.dinv.resize(A.n);
      for (Index i=0;i<A.n;++i) {
        double d=0.0;
        if (opt_.smoother==Smoother::l1_chebyshev) {
          for (Index q=A.row[i];q<A.row[i+1];++q) d+=std::abs(A.a[q]);
        } else d=std::abs(A.diagonal(i));
        L.dinv[i]=1.0/std::max(d,1e-30);
      }
      std::vector<double> x(A.n,1.0),y(A.n); double rho=1.0;
      for (int k=0;k<12;++k) {
        for (Index i=0;i<A.n;++i) {
          double s=0.0; for (Index q=A.row[i];q<A.row[i+1];++q) s+=A.a[q]*x[A.col[q]];
          y[i]=L.dinv[i]*s;
        }
        rho=norm2(y); if (rho<1e-30) { rho=1.0; break; }
        for (Index i=0;i<A.n;++i) x[i]=y[i]/rho;
      }
      L.lambda_max=std::max(1e-12,rho*1.05);
      const bool stop=A.n<=opt_.coarse_size || l+1==opt_.max_levels;
      if (!stop) {
        auto [P,nc]=aggregate(A,opt_.strength_threshold,opt_.aggregation==Aggregation::pmis_smoothed);
        if (nc>=A.n) { lv_.push_back(std::move(L)); break; }
        Csr Ac=transpose_multiply(A,P,nc); L.P=std::move(P); A=std::move(Ac);
      }
      lv_.push_back(std::move(L));
      if (stop) break;
    }
  }
  void apply(const std::vector<double>& b,std::vector<double>& x) const { cycle(0,b,x); }
  std::size_t levels() const { return lv_.size(); }
  std::size_t coarse_size() const { return lv_.back().A.n; }
};

inline void green_gauss_gradient(const MeshView& m,const CellFaceCSR& csr,
                                 const std::vector<double>& phi,
                                 const BoundarySoA* boundary,
                                 std::vector<Vec3>& grad) {
  std::fill(grad.begin(),grad.end(),Vec3{});
  #pragma omp parallel for schedule(static)
  for (std::ptrdiff_t ii=0;ii<std::ptrdiff_t(m.owned);++ii) {
    const Index i=Index(ii); Vec3 g{};
    for (Index q=csr.offset[i];q<csr.offset[i+1];++q) {
      const Index f=csr.face[q],P=m.owner[f],N=m.neighbour[f];
      const auto geo=make_geometry(m,f);
      double pf=0.0;
      if (N!=boundary) pf=geo.beta*phi[P]+(1.0-geo.beta)*phi[N];
      else pf=phi[P]; // zero-normal-gradient pressure; velocity callers supply a boundary field
      g += (double(csr.orientation[q])*pf/m.volume[i])*m.S[f];
    }
    grad[i]=g;
  }
}
inline void velocity_gradient(const MeshView& m,const CellFaceCSR& csr,
                              const std::vector<double>& phi,const BoundarySoA& b,
                              int component,std::vector<Vec3>& grad) {
  std::fill(grad.begin(),grad.end(),Vec3{});
  for (Index i=0;i<Index(m.owned);++i) {
    Vec3 g{};
    for (Index q=csr.offset[i];q<csr.offset[i+1];++q) {
      const Index f=csr.face[q],P=m.owner[f],N=m.neighbour[f];
      double bf=phi[P];
      if (N!=boundary) {
        const auto geo=make_geometry(m,f); bf=geo.beta*phi[P]+(1.0-geo.beta)*phi[N];
      } else {
        const Vec3 ub=boundary_velocity(b,f);
        bf=component==0?ub.x:component==1?ub.y:ub.z;
      }
      g += (double(csr.orientation[q])*bf/m.volume[i])*m.S[f];
    }
    grad[i]=g;
  }
}
inline void rhie_chow_flux(const MeshView& m,const FieldSoA& s,const BoundarySoA& b,
                           const std::vector<double>& D,bool enabled,
                           std::vector<double>& flux) {
  for (std::size_t ff=0;ff<m.faces;++ff) {
    const Index f=Index(ff),P=m.owner[f],N=m.neighbour[f];
    if (N==boundary) {
      const Vec3 ub=(b.kind[f]==BoundaryKind::outlet || b.kind[f]==BoundaryKind::zero_gradient)
                    ? Vec3{s.u[P],s.v[P],s.w[P]} : boundary_velocity(b,f);
      flux[f]=s.rho[P]*dot(ub,m.S[f]); continue;
    }
    const auto g=make_geometry(m,f);
    const Vec3 up{s.u[P],s.v[P],s.w[P]},un{s.u[N],s.v[N],s.w[N]};
    const Vec3 uf=g.beta*up+(1.0-g.beta)*un;
    const double rhof=g.beta*s.rho[P]+(1.0-g.beta)*s.rho[N];
    double m_dot=rhof*dot(uf,m.S[f]);
    if (enabled) {
      const double Df=g.beta*D[P]+(1.0-g.beta)*D[N];
      const Vec3 gf=g.beta*s.grad_p[P]+(1.0-g.beta)*s.grad_p[N];
      // alpha*(pN-pP-gradAvg.d) == (gradp_f-gradAvg).S with S=alpha*d+k.
      m_dot-=rhof*Df*g.alpha*(s.p[N]-s.p[P]-dot(gf,g.d));
    }
    flux[f]=m_dot;
  }
}

inline MomentumSystem assemble_momentum(const MeshView& m,const CellFaceCSR& csr,
                                        const FieldSoA& s,const BoundarySoA& b,
                                        const std::vector<double>& old_u,
                                        const std::vector<double>& old_v,
                                        const SolveOptions& o) {
  const Index n=Index(m.owned);
  std::vector<std::map<Index,double>> au(n),av(n);
  MomentumSystem M; M.bu.assign(n,0); M.bv.assign(n,0); M.D.resize(n); M.diag.resize(n);
  std::vector<double> div(n,0);
  for (Index i=0;i<n;++i) {
    au[i][i]+=s.rho[i]*m.volume[i]/o.dt+1e-30;
    av[i][i]+=s.rho[i]*m.volume[i]/o.dt+1e-30;
    M.bu[i]+=s.rho[i]*m.volume[i]/o.dt*old_u[i];
    M.bv[i]+=s.rho[i]*m.volume[i]/o.dt*old_v[i];
    const double brinkman=s.mu[i]*s.porosity[i]/std::max(1e-3,1.0-s.porosity[i]);
    au[i][i]+=brinkman*m.volume[i]; av[i][i]+=brinkman*m.volume[i];
  }
  // Four segments use the same contracts; only class 1 carries a deferred gradient.
  std::vector<Vec3> gu(n),gv(n);
  velocity_gradient(m,csr,s.u,b,0,gu); velocity_gradient(m,csr,s.v,b,1,gv);
  for (std::size_t ff=0;ff<m.faces;++ff) {
    const Index f=Index(ff),P=m.owner[f],N=m.neighbour[f];
    const auto g=make_geometry(m,f); const double F=s.mass_flux[f];
    const double muf=N==boundary?s.mu[P]:g.beta*s.mu[P]+(1.0-g.beta)*s.mu[N];
    const double diff=muf*g.alpha;
    if (N==boundary) {
      const Vec3 ub=boundary_velocity(b,f);
      au[P][P]+=diff+std::max(F,0.0); av[P][P]+=diff+std::max(F,0.0);
      M.bu[P]+=diff*ub.x-std::min(F,0.0)*ub.x;
      M.bv[P]+=diff*ub.y-std::min(F,0.0)*ub.y;
      div[P]+=F; continue;
    }
    const double aN=diff+std::max(-F,0.0);
    au[P][P]+=diff+std::max(F,0.0); au[P][N]-=aN;
    au[N][N]+=diff+std::max(-F,0.0); au[N][P]-=diff+std::max(F,0.0);
    av[P][P]+=diff+std::max(F,0.0); av[P][N]-=aN;
    av[N][N]+=diff+std::max(-F,0.0); av[N][P]-=diff+std::max(F,0.0);
    if (g.correction_class) {
      // Compact orthogonal part is implicit; k.grad(phi) is a deferred source.
      const Vec3 gf=g.beta*gu[P]+(1.0-g.beta)*gu[N];
      const Vec3 hf=g.beta*gv[P]+(1.0-g.beta)*gv[N];
      const double du=muf*dot(g.k,gf),dv=muf*dot(g.k,hf);
      M.bu[P]+=du; M.bu[N]-=du; M.bv[P]+=dv; M.bv[N]-=dv;
    }
    div[P]+=F; div[N]-=F;
  }
  for (Index i=0;i<n;++i) {
    const double stab=o.continuity_stabilization*std::abs(div[i]);
    au[i][i]+=stab; av[i][i]+=stab;
    M.bu[i]-=m.volume[i]*s.grad_p[i].x; M.bv[i]-=m.volume[i]*s.grad_p[i].y;
    const double a=au[i][i],d=av[i][i];
    au[i][i]=a/o.velocity_relaxation; av[i][i]=d/o.velocity_relaxation;
    M.bu[i]+=(1.0-o.velocity_relaxation)/o.velocity_relaxation*a*s.u[i];
    M.bv[i]+=(1.0-o.velocity_relaxation)/o.velocity_relaxation*d*s.v[i];
    M.diag[i]=0.5*(au[i][i]+av[i][i]); M.D[i]=m.volume[i]/M.diag[i];
  }
  M.u=maps_to_csr(au); M.v=maps_to_csr(av); return M;
}
inline Csr assemble_pressure(const MeshView& m,const FieldSoA& s,
                             const std::vector<double>& D,const std::vector<double>& flux,
                             std::vector<double>& rhs,Index pin=0) {
  const Index n=Index(m.owned); std::vector<std::map<Index,double>> a(n);
  rhs.assign(n,0);
  for (std::size_t ff=0;ff<m.faces;++ff) {
    const Index f=Index(ff),P=m.owner[f],N=m.neighbour[f];
    if (N==boundary) { rhs[P]-=flux[f]; continue; }
    const auto g=make_geometry(m,f);
    const double Df=g.beta*D[P]+(1.0-g.beta)*D[N];
    const double rhof=g.beta*s.rho[P]+(1.0-g.beta)*s.rho[N];
    const double c=rhof*Df*g.alpha;
    a[P][P]+=c; a[N][N]+=c; a[P][N]-=c; a[N][P]-=c;
    rhs[P]-=flux[f]; rhs[N]+=flux[f];
  }
  // Symmetric pressure gauge fixing: p'[pin]=0, with its column eliminated.
  for (Index i=0;i<n;++i) if (i!=pin) a[i].erase(pin);
  a[pin].clear(); a[pin][pin]=1.0; rhs[pin]=0.0;
  return maps_to_csr(a);
}
inline double correct_pressure_velocity(const MeshView& m,FieldSoA& s,
                                        const std::vector<double>& D,
                                        const SolveOptions& o,
                                        AmgPreconditioner*& amg_cache) {
  std::vector<double> rhs,pc(m.owned,0.0);
  Csr A=assemble_pressure(m,s,D,s.mass_flux,rhs);
  delete amg_cache;
  AmgOptions ao; ao.fp32_inner=o.fp32_amg;
  ao.aggregation=o.smoothed_aggregation?Aggregation::pmis_smoothed:Aggregation::pmis_unsmoothed;
  amg_cache=new AmgPreconditioner(A,ao);
  const auto ks=bicgstab(A,rhs,pc,o.krylov_tolerance,200,*amg_cache);
  std::vector<Vec3> gp(m.owned);
  CellFaceCSR csr=CellFaceCSR::build(m);
  green_gauss_gradient(m,csr,pc,nullptr,gp);
  double pnorm=0.0;
  for (std::size_t i=0;i<m.owned;++i) {
    s.u[i]-=D[i]*gp[i].x; s.v[i]-=D[i]*gp[i].y; s.w[i]-=D[i]*gp[i].z;
    s.p[i]+=o.pressure_relaxation*pc[i]; pnorm+=pc[i]*pc[i];
  }
  for (std::size_t ff=0;ff<m.faces;++ff) {
    const Index f=Index(ff),P=m.owner[f],N=m.neighbour[f]; if (N==boundary) continue;
    const auto g=make_geometry(m,f);
    const double Df=g.beta*D[P]+(1.0-g.beta)*D[N];
    const double rhof=g.beta*s.rho[P]+(1.0-g.beta)*s.rho[N];
    s.mass_flux[f]-=rhof*Df*g.alpha*(pc[N]-pc[P]);
  }
  return std::sqrt(pnorm);
}
inline double enthalpy_step(const MeshView& m,const CellFaceCSR& csr,FieldSoA& s,
                            const BoundarySoA& b,const SolveOptions& o) {
  const Index n=Index(m.owned); std::vector<std::map<Index,double>> a(n);
  std::vector<double> rhs(n,0),old=s.h;
  for (Index i=0;i<n;++i) {
    a[i][i]=s.rho[i]*m.volume[i]/o.dt; rhs[i]=a[i][i]*old[i];
  }
  for (std::size_t ff=0;ff<m.faces;++ff) {
    const Index f=Index(ff),P=m.owner[f],N=m.neighbour[f]; const double F=s.mass_flux[f];
    const auto g=make_geometry(m,f);
    const double kf=N==boundary?s.conductivity[P]:g.beta*s.conductivity[P]+(1.0-g.beta)*s.conductivity[N];
    const double diff=kf*g.alpha/std::max(1.0,g.beta*s.cp[P]+(1.0-g.beta)*s.cp[N==boundary?P:N]);
    if (N==boundary) {
      const double hb=boundary_enthalpy(b,f,s,P);
      a[P][P]+=diff+std::max(F,0.0); rhs[P]+=diff*hb-std::min(F,0.0)*hb;
    } else {
      const double an=diff+std::max(-F,0.0);
      a[P][P]+=diff+std::max(F,0.0); a[P][N]-=an;
      a[N][N]+=diff+std::max(-F,0.0); a[N][P]-=diff+std::max(F,0.0);
    }
  }
  Csr A=maps_to_csr(a); DiagonalPreconditioner pre{&A};
  auto ks=bicgstab(A,rhs,s.h,1e-8,50,pre);
  double r=0.0;
  for (Index i=0;i<n;++i) { s.T[i]=s.h[i]/s.cp[i]; r=std::max(r,std::abs(s.h[i]-old[i])); }
  return r;
}

inline double coupled_defect(const MeshView& m,const CellFaceCSR& csr,FieldSoA& s,
                             const BoundarySoA& b,const MomentumSystem& M,
                             const SolveOptions& o) {

export interface Csr {

export function matvec(A: Csr, x: ArrayLike<number>, out = new Array<number>(A.n).fill(0)): number[] {
  for (let i = 0; i < A.n; i++) {
    let s = 0;
    for (let q = A.row[i]; q < A.row[i + 1]; q++) s += A.a[q] * x[A.col[q]];
    out[i] = s;
  }
  return out;
}
export function dot(a: ArrayLike<number>, b: ArrayLike<number>): number {
  let s = 0;
  for (let i = 0; i < a.length; i++) s += a[i] * b[i];
  return s;
}
export function norm(a: ArrayLike<number>): number {
  return Math.sqrt(Math.max(0, dot(a, a)));
}
export function diagonal(A: Csr, i: number): number {
  for (let q = A.row[i]; q < A.row[i + 1]; q++) if (A.col[q] === i) return A.a[q];
  return 0;
}
export function infNorm(A: Csr): number {
  let r = 0;
  for (let i = 0; i < A.n; i++) {
    let s = 0;
    for (let q = A.row[i]; q < A.row[i + 1]; q++) s += Math.abs(A.a[q]);
    r = Math.max(r, s);
  }
  return r;
}
export function mapsToCsr(rows: Map<number, number>[]): Csr {
  const A: Csr = { n: rows.length, row: [0], col: [], a: [] };
  rows.forEach((row) => {
    [...row.entries()].sort((x, y) => x[0] - y[0]).forEach(([j, v]) => {
      if (v !== 0) { A.col.push(j); A.a.push(v); }
    });
    A.row.push(A.col.length);
  });
  return A;
}
export function denseSolve(input: number[], rhs: number[], n: number): number[] | null {
  const A = input.slice();
  const b = rhs.slice();
  for (let k = 0; k < n; k++) {
    let pivot = k;
    for (let i = k + 1; i < n; i++) if (Math.abs(A[i * n + k]) > Math.abs(A[pivot * n + k])) pivot = i;
    if (Math.abs(A[pivot * n + k]) < 1e-14) return null;
    if (pivot !== k) {
      for (let j = k; j < n; j++) [A[k * n + j], A[pivot * n + j]] = [A[pivot * n + j], A[k * n + j]];
      [b[k], b[pivot]] = [b[pivot], b[k]];
    }
    for (let i = k + 1; i < n; i++) {
      const m = A[i * n + k] / A[k * n + k];
      A[i * n + k] = 0;
      for (let j = k + 1; j < n; j++) A[i * n + j] -= m * A[k * n + j];
      b[i] -= m * b[k];
    }
  }
  for (let i = n - 1; i >= 0; i--) {
    for (let j = i + 1; j < n; j++) b[i] -= A[i * n + j] * b[j];
    b[i] /= A[i * n + i];
  }
  return b;
}

export function bicgstab(
  A: Csr, b: number[], initial: number[] = new Array<number>(A.n).fill(0),
  rtol = 1e-10, maxIter = 200, pre: Preconditioner = identityPreconditioner,
): { x: number[]; stats: KrylovStats } {
  const n = b.length;
  const x = initial.slice();
  let Ax = matvec(A, x);
  let r = b.map((v, i) => v - Ax[i]);
  const r0 = r.slice();
  let p = new Array<number>(n).fill(0);
  let v = new Array<number>(n).fill(0);
  let s = r.slice();
  let t = r.slice();
  const bn = norm(b);
  const An = infNorm(A);
  const stats: KrylovStats = {
    iterations: 0, converged: false, stagnated: false,
    residual: norm(r), trueResidual: norm(r), etaRound: 0,
  };
  const target = rtol * Math.max(bn, 1e-300);
  if (stats.residual <= target) { stats.converged = true; return { x, stats }; }
  let rho = 1, alpha = 1, omega = 1, previous = stats.residual, stale = 0;
  for (let it = 1; it <= maxIter; it++) {
    const rho1 = dot(r0, r);
    if (Math.abs(rho1) < 1e-300 || Math.abs(omega) < 1e-300) { stats.stagnated = true; break; }
    const beta = (rho1 / rho) * (alpha / omega);
    rho = rho1;
    p = p.map((pi, i) => r[i] + beta * (pi - omega * v[i]));
    const ph = pre.apply(p);
    v = matvec(A, ph);
    const rv = dot(r0, v);
    if (Math.abs(rv) < 1e-300) { stats.stagnated = true; break; }
    alpha = rho / rv;
    s = r.map((ri, i) => ri - alpha * v[i]);
    if (norm(s) <= target) {
      for (let i = 0; i < n; i++) x[i] += alpha * ph[i];
      stats.iterations = it;
      break;
    }
    const sh = pre.apply(s);
    t = matvec(A, sh);
    const tt = dot(t, t);
    if (tt < 1e-300) { stats.stagnated = true; break; }
    omega = dot(t, s) / tt;
    for (let i = 0; i < n; i++) x[i] += alpha * ph[i] + omega * sh[i];
    r = s.map((si, i) => si - omega * t[i]);
    stats.iterations = it;
    if (it % 5 === 0 || it === maxIter) {
      Ax = matvec(A, x);
      const trueR = b.map((bi, i) => bi - Ax[i]);
      const recurrenceError = norm(r.map((ri, i) => ri - trueR[i]));
      stats.trueResidual = norm(trueR);
      stats.etaRound = recurrenceError / (bn + An * norm(x) + Number.EPSILON);
      r = trueR;
      stale = Math.abs(previous - stats.trueResidual) <= 1e-3 * stats.trueResidual ? stale + 1 : 0;
      previous = stats.trueResidual;
      if (stale >= 3 || stats.etaRound > 0.25) { stats.stagnated = true; break; }
    }
    stats.residual = norm(r);
    if (stats.residual <= target) break;
  }
  Ax = matvec(A, x);
  const trueR = b.map((bi, i) => bi - Ax[i]);
  stats.trueResidual = norm(trueR);
  stats.residual = stats.trueResidual;
  stats.converged = stats.trueResidual <= target;
  return { x, stats };
}
export function transpose(A: Csr): Csr {
  const T: Csr = { n: A.n, row: new Array<number>(A.n + 1).fill(0), col: new Array<number>(A.a.length), a: new Array<number>(A.a.length) };
  A.col.forEach((j) => T.row[j + 1]++);
  for (let i = 0; i < A.n; i++) T.row[i + 1] += T.row[i];
  const cursor = T.row.slice(0, -1);
  for (let i = 0; i < A.n; i++) for (let q = A.row[i]; q < A.row[i + 1]; q++) {
    const z = cursor[A.col[q]]++;
    T.col[z] = i; T.a[z] = A.a[q];
  }
  return T;
}

function aggregate(A: Csr, theta: number, smoothed: boolean): { P: Csr; nc: number } {
  const n = A.n;
  const strong: number[][] = Array.from({ length: n }, () => []);
  for (let i = 0; i < n; i++) {
    let mx = 0;
    for (let q = A.row[i]; q < A.row[i + 1]; q++) if (A.col[q] !== i) mx = Math.max(mx, -A.a[q]);
    for (let q = A.row[i]; q < A.row[i + 1]; q++) {
      const j = A.col[q];
      if (j !== i && -A.a[q] >= theta * mx) strong[i].push(j);
    }
  }
  const weight = Array.from({ length: n }, (_, i) => hash(i) + 1e-12 * strong[i].length);
  const agg = new Array<number>(n).fill(-1);
  let nc = 0, remaining = n;
  while (remaining > 0) {
    const seeds: number[] = [];
    for (let i = 0; i < n; i++) if (agg[i] < 0) {
      let maximal = true;
      for (const j of strong[i]) if (agg[j] < 0 && (weight[j] > weight[i] || (weight[j] === weight[i] && j > i))) {
        maximal = false; break;
      }
      if (maximal) seeds.push(i);
    }
    if (!seeds.length) for (let i = 0; i < n; i++) if (agg[i] < 0) { seeds.push(i); break; }
    for (const i of seeds) if (agg[i] < 0) {
      const g = nc++;
      agg[i] = g; remaining--;
      for (const j of strong[i]) if (agg[j] < 0) { agg[j] = g; remaining--; }
    }
  }
  if (!smoothed) {
    return { P: { n, row: Array.from({ length: n + 1 }, (_, i) => i), col: agg, a: new Array<number>(n).fill(1) }, nc };
  }
  let x = new Array<number>(n).fill(1), y = new Array<number>(n).fill(0), rho = 1;
  for (let k = 0; k < 12; k++) {
    for (let i = 0; i < n; i++) {
      let s = 0;
      for (let q = A.row[i]; q < A.row[i + 1]; q++) s += A.a[q] * x[A.col[q]];
      y[i] = s / diagonal(A, i);
    }
    rho = norm(y);
    if (rho === 0) break;
    x = y.map((v) => v / rho);
  }
  const omega = 4 / (3 * Math.max(rho, 1e-12));
  const rows: Map<number, number>[] = Array.from({ length: n }, () => new Map());
  for (let i = 0; i < n; i++) {
    rows[i].set(agg[i], 1);

    for (let q = A.row[i]; q < A.row[i + 1]; q++) {

    }
  }

    });

  });

}
function galerkin(A: Csr, P: Csr, nc: number): Csr {
  const bRows: Map<number, number>[] = Array.from({ length: A.n }, () => new Map());

      const j = P.col[z];

    }
  }
  const pt: [number, number][][] = Array.from({ length: nc }, () => []);
  for (let i = 0; i < P.n; i++) for (let q = P.row[i]; q < P.row[i + 1]; q++) pt[P.col[q]].push([i, P.a[q]]);
  const coarse: Csr = { n: nc, row: [0], col: [], a: [] };
  for (let I = 0; I < nc; I++) {
    const row = new Map<number, number>();

    const entries = [...row.entries()].sort((a, b) => a[0] - b[0]);
    const scale = entries.reduce((m, [, v]) => Math.max(m, Math.abs(v)), 0);

    });

  const A: Csr = { n, row: [0], col: [], a: [] };
  for (let i = 0; i < n; i++) {
    if (i > 0) { A.col.push(i - 1); A.a.push(-1); }
    A.col.push(i); A.a.push(2);
    if (i + 1 < n) { A.col.push(i + 1); A.a.push(-1); }
    A.row.push(A.col.length);
  }
  return A;
}
export function trueResidual(A: Csr, b: number[], x: number[]): number {
  const Ax = matvec(A, x);
  return norm(b.map((v, i) => v - Ax[i]));
}

function hilbertXY(order: number, x: number, y: number): number {
  let rx = 0, ry = 0, d = 0;
  for (let s = Math.floor(order / 2); s > 0; s = Math.floor(s / 2)) {
    rx = (x & s) > 0 ? 1 : 0;
    ry = (y & s) > 0 ? 1 : 0;
    d += s * s * ((3 * rx) ^ ry);
    if (ry === 0) {
      if (rx === 1) { x = order - 1 - x; y = order - 1 - y; }
      [x, y] = [y, x];
    }
  }
  return d;
}
export function makeMesh(nx: number, ny: number, skew = 0): Mesh {
  const hx = 1 / nx, hy = 1 / ny, cells = nx * ny;
  const vertex = (i: number, j: number): Vec2 => ({ x: i * hx + skew * j * hy, y: j * hy });
  const x: Vec2[] = [];
  for (let j = 0; j < ny; j++) for (let i = 0; i < nx; i++) x.push({ x: (i + 0.5) * hx + skew * (j + 0.5) * hy, y: (j + 0.5) * hy });
  const volume = new Array<number>(cells).fill(hx * hy);
  const raw: Omit<Face, 'd' | 'beta' | 'alpha' | 'k' | 'correctionClass' | 'segment' | 'hilbert'>[] = [];
  const wall = { kind: BoundaryKind.Wall, ub: { x: 0, y: 0 }, hb: 300, hFixed: true };
  for (let j = 0; j < ny; j++) for (let i = 0; i < nx; i++) {
    const P = j * nx + i;
    if (i + 1 < nx) {
      const a = vertex(i + 1, j), b = vertex(i + 1, j + 1);
      raw.push({ p: P, n: P + 1, xf: mul(0.5, add(a, b)), S: { x: hy, y: -skew * hy }, ...wall });
    }
    if (j + 1 < ny) {
      const a = vertex(i, j + 1), b = vertex(i + 1, j + 1);
      raw.push({ p: P, n: P + nx, xf: mul(0.5, add(a, b)), S: { x: 0, y: hx }, ...wall });
    }
    if (i === 0) {
      const a = vertex(0, j), b = vertex(0, j + 1);
      raw.push({ p: P, n: -1, xf: mul(0.5, add(a, b)), S: { x: -hy, y: skew * hy }, kind: BoundaryKind.Wall, ub: { x: 0, y: 0 }, hb: 300, hFixed: false });
    }
    if (i === nx - 1) {
      const a = vertex(nx, j), b = vertex(nx, j + 1);
      raw.push({ p: P, n: -1, xf: mul(0.5, add(a, b)), S: { x: hy, y: -skew * hy }, kind: BoundaryKind.Wall, ub: { x: 0, y: 0 }, hb: 300, hFixed: false });
    }
    if (j === 0) {
      const a = vertex(i, 0), b = vertex(i + 1, 0);
      raw.push({ p: P, n: -1, xf: mul(0.5, add(a, b)), S: { x: 0, y: -hx }, kind: BoundaryKind.Wall, ub: { x: 0, y: 0 }, hb: 450, hFixed: true });
    }
    if (j === ny - 1) {
      const a = vertex(i, ny), b = vertex(i + 1, ny);
      raw.push({ p: P, n: -1, xf: mul(0.5, add(a, b)), S: { x: 0, y: hx }, kind: BoundaryKind.MovingWall, ub: { x: 1, y: 0 }, hb: 300, hFixed: true });
    }
  }
  const faces: Face[] = raw.map((f) => {
    const d = f.n < 0 ? sub(f.xf, x[f.p]) : sub(x[f.n], x[f.p]);
    const dS = dot2(d, f.S);
    if (dS <= 0) throw new Error('Inverted generated face');
    const S2 = dot2(f.S, f.S);
    const beta = f.n < 0 ? 1 : dot2(sub(f.xf, x[f.p]), f.S) / dS;
    const alpha = S2 / dS;
    const kFull = sub(f.S, mul(alpha, d));
    const correctionClass = length2(kFull) / Math.sqrt(S2) > 1e-12 ? 1 : 0;
    return { ...f, d, beta, alpha, k: correctionClass ? kFull : { x: 0, y: 0 }, correctionClass, segment: 0, hilbert: 0 };
  });
  const order = 256;
  faces.forEach((f) => {
    const gx = Math.max(0, Math.min(order - 1, Math.floor((f.xf.x - skew * f.xf.y) * order)));
    const gy = Math.max(0, Math.min(order - 1, Math.floor(f.xf.y * order)));
    f.hilbert = hilbertXY(order, gx, gy);
    // Single-rank generated mesh: halo quadrants are intentionally empty.
    f.segment = f.correctionClass;
  });
  faces.sort((a, b) => a.segment - b.segment || a.hilbert - b.hilbert);
  const segmentOffset = [0, 0, 0, 0, 0];
  for (let s = 0; s < 4; s++) {
    segmentOffset[s] = faces.findIndex((f) => f.segment >= s);
    if (segmentOffset[s] < 0) segmentOffset[s] = faces.length;
  }
  segmentOffset[0] = 0; segmentOffset[4] = faces.length;
  for (let s = 0; s < 4; s++) {
    const begin = faces.findIndex((f) => f.segment === s);
    const end = faces.findIndex((f) => f.segment > s);
    if (begin >= 0) segmentOffset[s] = begin;
    if (s < 3) segmentOffset[s + 1] = end < 0 ? faces.length : end;
  }
  // Empty quadrants share the next non-empty offset.
  for (let s = 1; s < 4; s++) segmentOffset[s] = Math.max(segmentOffset[s], segmentOffset[s - 1]);
  const offset = new Array<number>(cells + 1).fill(0);
  faces.forEach((f) => { offset[f.p + 1]++; if (f.n >= 0) offset[f.n + 1]++; });
  for (let i = 0; i < cells; i++) offset[i + 1] += offset[i];
  const face = new Array<number>(offset[cells]), sign = new Array<number>(offset[cells]);
  const cursor = offset.slice(0, -1);
  faces.forEach((f, fi) => {
    face[cursor[f.p]] = fi; sign[cursor[f.p]++] = 1;
    if (f.n >= 0) { face[cursor[f.n]] = fi; sign[cursor[f.n]++] = -1; }
  });
  const correctionBytes = 28 * faces.length;
  return { nx, ny, owned: cells, x, volume, faces, csr: { offset, face, sign }, segmentOffset, skew,
    correctionBytes, correctionOverhead: 100 * correctionBytes / (464 * faces.length) };
}
export function gradient(mesh: Mesh, phi: number[], boundaryValue?: (f: Face) => number): Vec2[] {
  const g = Array.from({ length: mesh.owned }, () => ({ x: 0, y: 0 }));
  for (let i = 0; i < mesh.owned; i++) {
    for (let q = mesh.csr.offset[i]; q < mesh.csr.offset[i + 1]; q++) {
      const f = mesh.faces[mesh.csr.face[q]];
      const pf = f.n < 0 ? (boundaryValue ? boundaryValue(f) : phi[f.p]) : f.beta * phi[f.p] + (1 - f.beta) * phi[f.n];
      const scale = mesh.csr.sign[q] * pf / mesh.volume[i];
      g[i].x += scale * f.S.x; g[i].y += scale * f.S.y;
    }
  }
  return g;
}
export function continuity(mesh: Mesh, flux: number[]): { byCell: number[]; max: number; l2: number } {
  const byCell = new Array<number>(mesh.owned).fill(0);
  mesh.faces.forEach((f, i) => { byCell[f.p] += flux[i]; if (f.n >= 0) byCell[f.n] -= flux[i]; });
  return { byCell, max: byCell.reduce((m, v) => Math.max(m, Math.abs(v)), 0), l2: Math.hypot(...byCell) };
}

export function rhieChowFlux(mesh: Mesh, s: FlowState, D: number[], enabled: boolean): number[] {
  return mesh.faces.map((f) => {
    if (f.n < 0) {

        ? { x: s.u[f.p], y: s.v[f.p] } : f.ub;
      return s.rho[f.p] * dot2(ub, f.S);

    if (enabled) {

        y: f.beta * s.gradP[f.p].y + (1 - f.beta) * s.gradP[f.n].y,

  });

function assembleMomentum(mesh: Mesh, s: FlowState, oldU: number[], oldV: number[], cfg: FlowConfig): MomentumSystem {
  const n = mesh.owned;
  const au = Array.from({ length: n }, () => new Map<number, number>());
  const av = Array.from({ length: n }, () => new Map<number, number>());
  const bu = new Array<number>(n).fill(0), bv = new Array<number>(n).fill(0), div = new Array<number>(n).fill(0);
  const add = (rows: Map<number, number>[], i: number, j: number, v: number) => rows[i].set(j, (rows[i].get(j) ?? 0) + v);
  for (let i = 0; i < n; i++) {
    const transient = s.rho[i] * mesh.volume[i] / cfg.dt;
    add(au, i, i, transient + 1e-30); add(av, i, i, transient + 1e-30);
    bu[i] = transient * oldU[i]; bv[i] = transient * oldV[i];
    const brinkman = s.mu[i] * s.porosity[i] / Math.max(1e-3, 1 - s.porosity[i]);
    add(au, i, i, brinkman * mesh.volume[i]); add(av, i, i, brinkman * mesh.volume[i]);
  }
  const gu = velocityGradient(mesh, s.u, 'x'), gv = velocityGradient(mesh, s.v, 'y');
  mesh.faces.forEach((f, fi) => {
    const P = f.p, N = f.n, F = s.massFlux[fi];
    const muf = N < 0 ? s.mu[P] : f.beta * s.mu[P] + (1 - f.beta) * s.mu[N];
    const diff = muf * f.alpha;
    if (N < 0) {
      add(au, P, P, diff + Math.max(F, 0)); add(av, P, P, diff + Math.max(F, 0));
      bu[P] += diff * f.ub.x - Math.min(F, 0) * f.ub.x;
      bv[P] += diff * f.ub.y - Math.min(F, 0) * f.ub.y;
      div[P] += F; return;
    }
    const aN = diff + Math.max(-F, 0);
    add(au, P, P, diff + Math.max(F, 0)); add(au, P, N, -aN);
    add(au, N, N, diff + Math.max(-F, 0)); add(au, N, P, -(diff + Math.max(F, 0)));
    add(av, P, P, diff + Math.max(F, 0)); add(av, P, N, -aN);
    add(av, N, N, diff + Math.max(-F, 0)); add(av, N, P, -(diff + Math.max(F, 0)));
    if (f.correctionClass) {
      const gf = { x: f.beta * gu[P].x + (1 - f.beta) * gu[N].x, y: f.beta * gu[P].y + (1 - f.beta) * gu[N].y };
      const hf = { x: f.beta * gv[P].x + (1 - f.beta) * gv[N].x, y: f.beta * gv[P].y + (1 - f.beta) * gv[N].y };
      const du = muf * dot2(f.k, gf), dv = muf * dot2(f.k, hf);
      bu[P] += du; bu[N] -= du; bv[P] += dv; bv[N] -= dv;
    }
    div[P] += F; div[N] -= F;
  });
  const D = new Array<number>(n), diag = new Array<number>(n);
  for (let i = 0; i < n; i++) {
    const stab = 1e-12 * Math.abs(div[i]);
    add(au, i, i, stab); add(av, i, i, stab);
    bu[i] -= mesh.volume[i] * s.gradP[i].x; bv[i] -= mesh.volume[i] * s.gradP[i].y;
    const a = au[i].get(i)!, d = av[i].get(i)!;
    au[i].set(i, a / cfg.relaxU); av[i].set(i, d / cfg.relaxU);
    bu[i] += (1 - cfg.relaxU) / cfg.relaxU * a * s.u[i];
    bv[i] += (1 - cfg.relaxU) / cfg.relaxU * d * s.v[i];
    diag[i] = 0.5 * (au[i].get(i)! + av[i].get(i)!); D[i] = mesh.volume[i] / diag[i];
  }
  return { u: mapsToCsr(au), v: mapsToCsr(av), bu, bv, D, diagonal: diag };
}
function assemblePressure(mesh: Mesh, s: FlowState, D: number[]): { A: Csr; rhs: number[] } {
  const rows = Array.from({ length: mesh.owned }, () => new Map<number, number>());
  const rhs = new Array<number>(mesh.owned).fill(0);
  const add = (i: number, j: number, v: number) => rows[i].set(j, (rows[i].get(j) ?? 0) + v);
  mesh.faces.forEach((f, fi) => {
    const P = f.p, N = f.n;
    if (N < 0) { rhs[P] -= s.massFlux[fi]; return; }
    const Df = f.beta * D[P] + (1 - f.beta) * D[N];
    const rho = f.beta * s.rho[P] + (1 - f.beta) * s.rho[N];
    const c = rho * Df * f.alpha;
    add(P, P, c); add(N, N, c); add(P, N, -c); add(N, P, -c);
    rhs[P] -= s.massFlux[fi]; rhs[N] += s.massFlux[fi];
  });
  for (let i = 1; i < mesh.owned; i++) rows[i].delete(0);
  rows[0].clear(); rows[0].set(0, 1); rhs[0] = 0;
  return { A: mapsToCsr(rows), rhs };
}
function enthalpyStep(mesh: Mesh, s: FlowState, cfg: FlowConfig): { change: number; iterations: number } {
  const n = mesh.owned, rows = Array.from({ length: n }, () => new Map<number, number>());
  const rhs = new Array<number>(n).fill(0), old = s.h.slice();
  const add = (i: number, j: number, v: number) => rows[i].set(j, (rows[i].get(j) ?? 0) + v);
  for (let i = 0; i < n; i++) {
    const transient = s.rho[i] * mesh.volume[i] / cfg.dt;
    add(i, i, transient); rhs[i] = transient * old[i];
  }
  mesh.faces.forEach((f, fi) => {
    const P = f.p, N = f.n, F = s.massFlux[fi];
    const kf = N < 0 ? s.conductivity[P] : f.beta * s.conductivity[P] + (1 - f.beta) * s.conductivity[N];
    const cp = N < 0 ? s.cp[P] : f.beta * s.cp[P] + (1 - f.beta) * s.cp[N];
    const diff = kf * f.alpha / Math.max(1, cp);
    if (N < 0) {
      const hb = f.hFixed ? f.hb * s.cp[P] : s.h[P];
      add(P, P, diff + Math.max(F, 0)); rhs[P] += diff * hb - Math.min(F, 0) * hb;
    } else {
      const an = diff + Math.max(-F, 0);
      add(P, P, diff + Math.max(F, 0)); add(P, N, -an);
      add(N, N, diff + Math.max(-F, 0)); add(N, P, -(diff + Math.max(F, 0)));
    }
  });
  const A = mapsToCsr(rows);
  const solved = bicgstab(A, rhs, s.h, 1e-9, 80, diagonalPreconditioner(A));
  s.h = solved.x;
  let change = 0;
  for (let i = 0; i < n; i++) {
    s.T[i] = s.h[i] / s.cp[i]; change = Math.max(change, Math.abs(s.h[i] - old[i]) / Math.max(1, Math.abs(old[i])));
  }
  return { change, iterations: solved.stats.iterations };
}
function coupledDefect(mesh: Mesh, s: FlowState, M: MomentumSystem, cfg: FlowConfig): { change: number; iterations: number } {
  const n = mesh.owned, size = 3 * n;
  const rows = Array.from({ length: size }, () => new Map<number, number>());
  const add = (i: number, j: number, v: number) => rows[i].set(j, (rows[i].get(j) ?? 0) + v);
  const block = (A: Csr, offset: number) => {
    for (let i = 0; i < n; i++) for (let q = A.row[i]; q < A.row[i + 1]; q++) add(offset + i, offset + A.col[q], A.a[q]);
  };
  block(M.u, 0); block(M.v, n);
  mesh.faces.forEach((f) => {
    const P = f.p, N = f.n;
    if (N < 0) {
      add(P, 2 * n + P, f.S.x); add(n + P, 2 * n + P, f.S.y);
    } else {
      add(P, 2 * n + P, f.beta * f.S.x); add(P, 2 * n + N, (1 - f.beta) * f.S.x);
      add(N, 2 * n + P, -f.beta * f.S.x); add(N, 2 * n + N, -(1 - f.beta) * f.S.x);
      add(n + P, 2 * n + P, f.beta * f.S.y); add(n + P, 2 * n + N, (1 - f.beta) * f.S.y);
      add(n + N, 2 * n + P, -f.beta * f.S.y); add(n + N, 2 * n + N, -(1 - f.beta) * f.S.y);
    }
  });
  mesh.faces.forEach((f) => {
    const P = f.p, N = f.n;
    if (N < 0) {
      if (f.kind === BoundaryKind.Outlet || f.kind === BoundaryKind.ZeroGradient) {
        add(2 * n + P, P, s.rho[P] * f.S.x); add(2 * n + P, n + P, s.rho[P] * f.S.y);
      }
      return;
    }
    const rP = f.beta * s.rho[P], rN = (1 - f.beta) * s.rho[N];
    const Df = f.beta * M.D[P] + (1 - f.beta) * M.D[N];
    const rho = f.beta * s.rho[P] + (1 - f.beta) * s.rho[N], c = rho * Df * f.alpha;
    add(2 * n + P, P, rP * f.S.x); add(2 * n + P, N, rN * f.S.x);
    add(2 * n + P, n + P, rP * f.S.y); add(2 * n + P, n + N, rN * f.S.y);
    add(2 * n + P, 2 * n + P, c); add(2 * n + P, 2 * n + N, -c);
    add(2 * n + N, P, -rP * f.S.x); add(2 * n + N, N, -rN * f.S.x);
    add(2 * n + N, n + P, -rP * f.S.y); add(2 * n + N, n + N, -rN * f.S.y);
    add(2 * n + N, 2 * n + P, -c); add(2 * n + N, 2 * n + N, c);
  });

export function rhieChowExactness(): { fluxError: number; betaError: number } {
  const P = { x: 0, y: 0 }, N = { x: 0.5, y: 0.1 }, S = { x: 1, y: 0.2 };
  const d = { x: N.x - P.x, y: N.y - P.y }, tangent = { x: -S.y, y: S.x };

}

function arrhenius(a: number, e: number, T: number): [number, number] {
  const k = a * Math.exp(-e / (Ru * T));
  return [k, k * e / (Ru * T * T)];

export function rates(Y: number[], T: number, rho: number): Rates {
  const [kf, dkf] = arrhenius(A, Ea, T), [kr, dkr] = arrhenius(Ar, EaR, T);
  const c = [rho * Math.max(Y[0], 0), rho * Math.max(Y[1], 0), rho * Math.max(Y[2], 0) / 2];
  const forward = kf * c[0] * c[1], reverse = kr * c[2], q = forward - reverse;
  const dqT = dkf * c[0] * c[1] - dkr * c[2];
  const omega = [-q, -q, 2 * q], dT = [-dqT, -dqT, 2 * dqT];
  const dF = [kf * rho * c[1], kf * rho * c[0], 0], dR = [0, 0, kr * rho / 2];
  const J = Array.from({ length: 3 }, (_, k) => Array.from({ length: 3 }, (_, j) => {
    const nu = k === 2 ? 2 : -1;
    return (k === 2 ? 2 : 1) * nu * (dF[j] - dR[j]);

  return { omega, dT, J };
}
function rhs(y: number[], rho: number, withJacobian: boolean): { f: number[]; J: number[] } {
  const r = rates(y.slice(0, 3), y[3], rho);
  const cp = Math.max(1, y[0] * heatCapacity[0] + y[1] * heatCapacity[1] + y[2] * heatCapacity[2]);
  const f = [...r.omega.map((w) => w / rho), 0];
  f[3] = -r.omega.reduce((s, w, k) => s + w * hFormation[k], 0) / (rho * cp);
  const J = new Array<number>(16).fill(0);
  if (withJacobian) {
    for (let k = 0; k < 3; k++) {
      for (let j = 0; j < 3; j++) J[4 * k + j] = r.J[k][j] / rho;
      J[4 * k + 3] = r.dT[k] / rho;
    }
    const Q = r.omega.reduce((s, w, k) => s + w * hFormation[k], 0);
    for (let j = 0; j < 3; j++) {
      const dQ = r.J.reduce((s, row, k) => s + row[j] * hFormation[k], 0);
      J[12 + j] = -dQ / (rho * cp) + Q * heatCapacity[j] / (rho * cp * cp);
    }
    J[15] = -r.dT.reduce((s, dw, k) => s + dw * hFormation[k], 0) / (rho * cp);
  }
  return { f, J };
}

      error += ((embedded - next[i]) / scale) ** 2;
    }

    const positive = next[3] > 1 && next.slice(0, 3).every((v) => v >= -cfg.absoluteTolerance);
    if (error <= 1 && positive) {

export interface ChannelParameters { gamma: number[]; xLeft: number; sLeft: number; sRight: number }
export interface AdjointResult {
  objective: number; gamma: number[]; fdGamma: number[]; gammaErrors: number[];
  xLeft: number; fdXLeft: number; sLeft: number; fdSLeft: number; sRight: number; fdSRight: number;

class BrinkmanChannel {
  readonly n = 6;
  readonly h = 1 / this.n;
  readonly mu = 0.02;
  readonly uIn = 0.8;
  readonly pOut = 0;
  a(i: number, q: ChannelParameters): number { return q.gamma[i] + 4 * this.mu / (this.h * this.h); }
  kappa(i: number, q: ChannelParameters): number { return 0.5 * (1 / this.a(i, q) + 1 / this.a(i + 1, q)); }
  residual(U: number[], q: ChannelParameters): number[] {
    const R = new Array<number>(2 * this.n).fill(0), u = (i: number) => U[i], p = (i: number) => U[this.n + i];
    for (let i = 0; i < this.n; i++) R[i] = q.gamma[i] * this.h * u(i) - this.h * (0.1 + 0.02 * i);
    for (let i = 0; i + 1 < this.n; i++) {
      const diff = this.mu / this.h * (u(i) - u(i + 1)); R[i] += diff; R[i + 1] -= diff;
      const pf = 0.5 * (p(i) + p(i + 1)); R[i] += pf; R[i + 1] -= pf;
      const F = 0.5 * (u(i) + u(i + 1)) - this.kappa(i, q) * (p(i + 1) - p(i));
      R[this.n + i] += F; R[this.n + i + 1] -= F;
    }
    const dist = 0.5 * this.h - q.xLeft, g = this.mu * q.sLeft / dist;
    R[0] += g * (u(0) - this.uIn) - p(0) * q.sLeft;
    R[this.n] -= this.uIn * q.sLeft;
    R[this.n - 1] += this.pOut * q.sRight;
    R[2 * this.n - 1] += u(this.n - 1) * q.sRight;
    return R;
  }
  jacobian(q: ChannelParameters): Csr {
    const n = this.n, rows = Array.from({ length: 2 * n }, () => new Map<number, number>());
    const add = (i: number, j: number, v: number) => rows[i].set(j, (rows[i].get(j) ?? 0) + v);
    for (let i = 0; i < n; i++) add(i, i, q.gamma[i] * this.h);
    for (let i = 0; i + 1 < n; i++) {
      const d = this.mu / this.h;
      add(i, i, d); add(i, i + 1, -d); add(i + 1, i + 1, d); add(i + 1, i, -d);
      add(i, n + i, 0.5); add(i, n + i + 1, 0.5); add(i + 1, n + i, -0.5); add(i + 1, n + i + 1, -0.5);
      const K = this.kappa(i, q);
      add(n + i, i, 0.5); add(n + i, i + 1, 0.5); add(n + i, n + i, K); add(n + i, n + i + 1, -K);
      add(n + i + 1, i, -0.5); add(n + i + 1, i + 1, -0.5); add(n + i + 1, n + i, -K); add(n + i + 1, n + i + 1, K);
    }
    const dist = 0.5 * this.h - q.xLeft, g = this.mu * q.sLeft / dist;
    add(0, 0, g); add(0, n, -q.sLeft); add(2 * n - 1, n - 1, q.sRight);
    return mapsToCsr(rows);
  }
  solve(q: ChannelParameters): number[] {
    const A = this.jacobian(q), n = 2 * this.n, D = new Array<number>(n * n).fill(0);
    for (let i = 0; i < n; i++) for (let k = A.row[i]; k < A.row[i + 1]; k++) D[i * n + A.col[k]] = A.a[k];
    const rhs = this.residual(new Array<number>(n).fill(0), q).map((v) => -v);
    const U = denseSolve(D, rhs, n);
    if (!U) throw new Error('Singular channel saddle system');
    return U;
  }
  objective(U: number[]): number {
    let J = 0;
    for (let i = 0; i < this.n; i++) J += (0.2 + 0.03 * i) * U[i] + 0.0005 * U[this.n + i] ** 2;
    return J;
  }
  gradient(U: number[]): number[] {
    const g = new Array<number>(2 * this.n).fill(0);
    for (let i = 0; i < this.n; i++) { g[i] = 0.2 + 0.03 * i; g[this.n + i] = 0.001 * U[this.n + i]; }
    return g;
  }
  sensitivities(q: ChannelParameters): { U: number[]; lambda: number[]; gamma: number[]; xLeft: number; sLeft: number; sRight: number } {
    const n = this.n, U = this.solve(q), A = this.jacobian(q), At = transpose(A);
    const size = 2 * n, dense = new Array<number>(size * size).fill(0);
    for (let i = 0; i < size; i++) for (let k = At.row[i]; k < At.row[i + 1]; k++) dense[i * size + At.col[k]] = At.a[k];
    const lambda = denseSolve(dense, this.gradient(U).map((v) => -v), size);
    if (!lambda) throw new Error('Singular transpose adjoint system');
    const gamma = new Array<number>(n).fill(0), u = (i: number) => U[i], p = (i: number) => U[n + i];
    for (let i = 0; i < n; i++) gamma[i] += lambda[i] * this.h * u(i);
    for (let i = 0; i + 1 < n; i++) {
      const dp = p(i + 1) - p(i), weight = lambda[n + i] - lambda[n + i + 1];
      gamma[i] += weight * 0.5 * dp / (this.a(i, q) ** 2);
      gamma[i + 1] += weight * 0.5 * dp / (this.a(i + 1, q) ** 2);
    }
    const dist = 0.5 * this.h - q.xLeft, g = this.mu * q.sLeft / dist;
    const xLeft = lambda[0] * this.mu * q.sLeft / (dist * dist) * (u(0) - this.uIn);
    const sLeft = lambda[0] * (g / q.sLeft * (u(0) - this.uIn) - p(0)) - lambda[n] * this.uIn;
    const sRight = lambda[n - 1] * this.pOut + lambda[2 * n - 1] * u(n - 1);
    return { U, lambda, gamma, xLeft, sLeft, sRight };
  }
}
type V3 = [number, number, number];
const dot3 = (a: V3, b: V3) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
const sub3 = (a: V3, b: V3): V3 => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
function cadQ(xf: V3, S: V3): number {
  const xP: V3 = [0.4, 0.2, -0.1], u: V3 = [0.8, -0.2, 0.3], ub: V3 = [0.3, 0.1, 0];
  const lambdaU: V3 = [0.7, -0.4, 0.2], lambdaC = -0.3, p = 1.2, mu = 0.02;
  const e = sub3(xP, xf), eS = dot3(e, S), g = mu * dot3(S, S) / eS;
  const du = sub3(u, ub);
  return g * dot3(lambdaU, du) + p * dot3(lambdaU, S) - lambdaC * dot3(ub, S);
}
function cadSeedCheck(): number {
  const xP: V3 = [0.4, 0.2, -0.1], xf: V3 = [-0.1, 0.05, 0.03], S: V3 = [0.9, 0.2, 0.1];
  const u: V3 = [0.8, -0.2, 0.3], ub: V3 = [0.3, 0.1, 0];
  const lambdaU: V3 = [0.7, -0.4, 0.2], lambdaC = -0.3, p = 1.2, mu = 0.02;
  const e = sub3(xP, xf), eS = dot3(e, S), S2 = dot3(S, S), g = mu * S2 / eS;
  const lu = dot3(lambdaU, sub3(u, ub));
  const dx: V3 = [lu * g / eS * S[0], lu * g / eS * S[1], lu * g / eS * S[2]];
  const dS: V3 = [0, 1, 2].map((k) => p * lambdaU[k] - lambdaC * ub[k] + lu * mu * 2 / eS * S[k] - lu * g / eS * e[k]) as V3;
  let max = 0; const h = 1e-6;
  for (let k = 0; k < 3; k++) {
    const xp = xf.slice() as V3, xm = xf.slice() as V3; xp[k] += h; xm[k] -= h;
    const fdx = (cadQ(xp, S) - cadQ(xm, S)) / (2 * h);
    const sp = S.slice() as V3, sm = S.slice() as V3; sp[k] += h; sm[k] -= h;
    const fds = (cadQ(xf, sp) - cadQ(xf, sm)) / (2 * h);
    max = Math.max(max, Math.abs(fdx - dx[k]) / Math.max(1, Math.abs(fdx)), Math.abs(fds - dS[k]) / Math.max(1, Math.abs(fds)));
  }
  return max;
}
export function verifyAdjoint(fdStep = 2e-5): AdjointResult {
  const channel = new BrinkmanChannel();
  const q: ChannelParameters = { gamma: [0.18, 0.25, 0.32, 0.2, 0.28, 0.22], xLeft: -0.083, sLeft: 1.04, sRight: 0.97 };
  const s = channel.sensitivities(q), fdGamma: number[] = [];
  for (let i = 0; i < channel.n; i++) {
    const qp: ChannelParameters = { ...q, gamma: q.gamma.slice() }, qm: ChannelParameters = { ...q, gamma: q.gamma.slice() };
    qp.gamma[i] += fdStep; qm.gamma[i] -= fdStep;
    fdGamma.push((channel.objective(channel.solve(qp)) - channel.objective(channel.solve(qm))) / (2 * fdStep));
  }
  const scalarFd = (set: (p: ChannelParameters, h: number) => void): number => {
    const qp = { ...q, gamma: q.gamma.slice() }, qm = { ...q, gamma: q.gamma.slice() };
    set(qp, fdStep); set(qm, -fdStep);
    return (channel.objective(channel.solve(qp)) - channel.objective(channel.solve(qm))) / (2 * fdStep);
  };
  const fdXLeft = scalarFd((p, h) => { p.xLeft += h; });
  const fdSLeft = scalarFd((p, h) => { p.sLeft += h; });
  const fdSRight = scalarFd((p, h) => { p.sRight += h; });
  const relative = (a: number, b: number) => Math.abs(a - b) / Math.max(1, Math.abs(b));
  const gammaErrors = s.gamma.map((v, i) => relative(v, fdGamma[i]));
  const xLeftError = relative(s.xLeft, fdXLeft), sLeftError = relative(s.sLeft, fdSLeft), sRightError = relative(s.sRight, fdSRight);
  const cadSeedError = cadSeedCheck();
  return {
    objective: channel.objective(s.U), gamma: s.gamma, fdGamma, gammaErrors,
    xLeft: s.xLeft, fdXLeft, sLeft: s.sLeft, fdSLeft, sRight: s.sRight, fdSRight,
    xLeftError, sLeftError, sRightError, maxError: Math.max(...gammaErrors, xLeftError, sLeftError, sRightError),
    cadSeedError, state: s.U, lambda: s.lambda,
  };
}

export function verifyAmg(cfg: VerificationConfig): AmgResult {
  const A = poisson1d(cfg.amgSize), b = Array.from({ length: cfg.amgSize }, (_, i) => Math.sin(0.17 * i) + 0.01 * i);
  const pre = new AmgPreconditioner(A, {
    fp32: cfg.flow.fp32Amg, smoothed: cfg.flow.smoothedAggregation,
    l1: cfg.flow.l1Smoother, degree: cfg.flow.chebyshevDegree,
    coarseSize: 16, maxLevels: 6, strength: 0.25,
  });
  const solved = bicgstab(A, b, new Array<number>(cfg.amgSize).fill(0), 2e-10, 350, pre);
  const baselinePre = new AmgPreconditioner(A, {
    fp32: false, smoothed: cfg.flow.smoothedAggregation, l1: cfg.flow.l1Smoother,
    degree: cfg.flow.chebyshevDegree, coarseSize: 16, maxLevels: 6, strength: 0.25,
  });
  const baseline = bicgstab(A, b, new Array<number>(cfg.amgSize).fill(0), 2e-10, 350, baselinePre);
  return {
    size: cfg.amgSize, levels: pre.depth, coarseSize: pre.coarsestSize,
    operatorComplexity: pre.operatorComplexity, iterations: solved.stats.iterations,
    residual: trueResidual(A, b, solved.x), scaledResidual: solved.stats.trueResidual / Math.max(1e-300, Math.hypot(...b)),
    etaRound: solved.stats.etaRound, converged: solved.stats.converged, stagnated: solved.stats.stagnated,
    fp32: cfg.flow.fp32Amg, smoothed: cfg.flow.smoothedAggregation,
    baselineIterations: baseline.stats.iterations, baselineResidual: baseline.stats.trueResidual,
  };
}
export async function runVerification(
  input: VerificationConfig,
  onProgress?: (stage: RunStage, record?: IterationRecord) => void | Promise<void>,
): Promise<VerificationBundle> {
  const started = performance.now();
  const config: VerificationConfig = JSON.parse(JSON.stringify(input)) as VerificationConfig;
  await onProgress?.('adjoint');
  const adjoint = verifyAdjoint(config.adjointFdStep);
  await onProgress?.('chemistry');
  const chemistry = verifyChemistry(config.chemistry);
  await onProgress?.('amg');
  const amg = verifyAmg(config);
  await onProgress?.('flow');
  const flow = await solveFlow(config.flow, async (record) => { await onProgress?.('flow', record); });
  await onProgress?.('complete');
  const passed = flow.rcError < 1e-12 && flow.betaError < 1e-12 && flow.massError < 2e-6 &&
    adjoint.maxError < 1e-8 && adjoint.cadSeedError < 1e-8 && chemistry.jacobianError < 2e-7 &&
    chemistry.massError < 1e-8 && amg.converged && amg.scaledResidual < 2e-9;
  return {
    id: `run-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 7)}`,
    createdAt: new Date().toISOString(), config, flow, amg, chemistry, adjoint, passed,
    elapsedMs: performance.now() - started,
  };
}

function highlight(line: string): ReactNode {
  const re = /(\/\/.*$|"(?:\\.|[^"\\])*"|'(?:\\.|[^'\\])*'|\b\d+(?:\.\d+)?(?:e[+-]?\d+)?\b|\b[A-Za-z_][A-Za-z0-9_]*\b)/gi;
  const nodes: ReactNode[] = [];
  let last = 0, match: RegExpExecArray | null, key = 0;
  while ((match = re.exec(line))) {
    if (match.index > last) nodes.push(<span key={key++} dangerouslySetInnerHTML={{ __html: escape(line.slice(last, match.index)) }} />);
    const token = match[0];
    const cls = token.startsWith('//') ? 'text-white/25 italic' : token.startsWith('"') || token.startsWith("'") ? 'text-amber'
      : /^\d/.test(token) ? 'text-cyan' : keywords.has(token) ? 'text-lime' : token.startsWith('#') ? 'text-fuchsia-300' : 'text-white/75';
    nodes.push(<span key={key++} className={cls}>{token}</span>);
    last = match.index + token.length;
  }
  if (last < line.length) nodes.push(<span key={key++} dangerouslySetInnerHTML={{ __html: escape(line.slice(last)) }} />);

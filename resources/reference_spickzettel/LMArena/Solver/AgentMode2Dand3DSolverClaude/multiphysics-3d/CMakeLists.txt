cmake_minimum_required(VERSION 3.16)
project(multiphysics3d LANGUAGES CXX)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(PkgConfig REQUIRED)
pkg_search_module(PETSC REQUIRED IMPORTED_TARGET PETSc petsc)
add_executable(multiphysics3d solver.cpp)
target_compile_features(multiphysics3d PRIVATE cxx_std_17)
target_link_libraries(multiphysics3d PRIVATE MPI::MPI_CXX PkgConfig::PETSC)
target_compile_options(multiphysics3d PRIVATE -Wall -Wextra)
enable_testing()
add_test(NAME steady_serial COMMAND multiphysics3d -steps 1)
add_test(NAME buoyancy_serial COMMAND multiphysics3d -steps 1 -demo_buoyancy)
if(MPIEXEC_EXECUTABLE)
  add_test(NAME steady_mpi_2 COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} 2 ${MPIEXEC_PREFLAGS} $<TARGET_FILE:multiphysics3d> ${MPIEXEC_POSTFLAGS} -steps 1)
  add_test(NAME buoyancy_mpi_2 COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} 2 ${MPIEXEC_PREFLAGS} $<TARGET_FILE:multiphysics3d> ${MPIEXEC_POSTFLAGS} -steps 1 -demo_buoyancy)
endif()

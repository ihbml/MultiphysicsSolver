inline Vec3<T> normalize(const Vec3<T>& a) {
    T n = norm(a);
    if (n == T(0)) return {T(0), T(0), T(0)};
    return {a.x / n, a.y / n, a.z / n};
}

inline uint32_t hash32(uint32_t a) {
    a = (a ^ 61) ^ (a >> 16);
    a = a + (a << 3);
    a = a ^ (a >> 4);
    a = a * 0x27d4eb2d;
    a = a ^ (a >> 15);
    return a;
}
// Deterministic "random" unit vector in [-1,1]^3
inline Vec3d detVec(uint32_t seed) {
    uint32_t h = hash32(seed);
    double fx = 2.0 * ((h & 0xFFFF) / 65535.0) - 1.0;
    h >>= 16;
    double fy = 2.0 * ((h & 0xFFFF) / 65535.0) - 1.0;
    h >>= 16;
    double fz = 2.0 * ((h & 0xFFFF) / 65535.0) - 1.0;
    double n = std::sqrt(fx*fx + fy*fy + fz*z);
    if (n < 1e-12) return {1.0, 0.0, 0.0};
    return {fx/n, fy/n, fz/n};
}

Mesh generateMesh(int N, double alphaSkew) {
    Mesh m;
    m.N = N;
    m.h = 1.0 / N;
    double h = m.h;
    // Generate vertices on [0,1]^3 with perturbations
    // Only perturb interior vertices to keep boundary clean
    int nv = (N + 1) * (N + 1) * (N + 1);
    m.vertices.resize(nv);
    auto vid = [&](int i, int j, int k) {
        return i * (N + 1) * (N + 1) + j * (N + 1) + k;
    };
    for (int i = 0; i <= N; ++i) {
        for (int j = 0; j <= N; ++j) {
            for (int k = 0; k <= N; ++k) {
                double x = i * h, y = j * h, z = k * h;
                // Perturb interior vertices only
                if (i > 0 && i < N && j > 0 && j < N && k > 0 && k < N) {
                    uint32_t seed = (uint32_t)(i * 100000 + j * 1000 + k);
                    Vec3d d = detVecFixed(seed);
                    x += alphaSkew * h * d.x;
                    y += alphaSkew * h * d.y;
                    z += alphaSkew * h * d.z;
                }
                m.vertices[vid(i, j, k)] = {x, y, z};
            }
        }
    }
    // Generate cells (hex cells)
    int nc = N * N * N;
    m.cells.resize(nc);
    auto cid = [&](int i, int j, int k) {
        return i * N * N + j * N + k;
    };
    for (int i = 0; i < N; ++i) {
        for (int j = 0; j < N; ++j) {
            for (int k = 0; k < N; ++k) {
                int c = cid(i, j, k);
                Cell& cell = m.cells[c];
                cell.id = c;
                cell.faceIds.clear();
                // 8 vertices of the hex cell
                Vec3d v[8];
                v[0] = m.vertices[vid(i,   j,   k  )];
                v[1] = m.vertices[vid(i+1, j,   k  )];
                v[2] = m.vertices[vid(i+1, j+1, k  )];
                v[3] = m.vertices[vid(i,   j+1, k  )];
                v[4] = m.vertices[vid(i,   j,   k+1)];
                v[5] = m.vertices[vid(i+1, j,   k+1)];
                v[6] = m.vertices[vid(i+1, j+1, k+1)];
                v[7] = m.vertices[vid(i,   j+1, k+1)];
                // Cell centroid = average of vertices
                Vec3d ctr{0,0,0};
                for (int q = 0; q < 8; ++q) {
                    ctr.x += v[q].x; ctr.y += v[q].y; ctr.z += v[q].z;
                }
                ctr.x /= 8.0; ctr.y /= 8.0; ctr.z /= 8.0;
                cell.centroid = ctr;
                // Cell volume via pyramid/tetrahedral decomposition:
                // Decompose hex into 6 pyramids (one per face) from centroid
                // Each pyramid volume = (1/3) * (face area vector) dot (face centroid - cell centroid)
                // We compute faces first, then volume
            }
        }
    }
    // Generate faces
    // Interior faces: shared between two cells
    // Boundary faces: on the domain boundary
    int nf = 0;
    // Count faces: 3*N*N*(N+1) total faces in structured grid
    m.faces.reserve(3 * N * N * (N + 1));
    // Helper to add a face from 4 vertices
    auto addFace = [&](const Vec3d fv[4], int owner, int neighbor) {
        Face f;
        f.id = nf++;
        f.ownerCell = owner;
        f.neighborCell = neighbor;
        // Face centroid (average of 4 vertices)
        f.centroid = {(fv[0].x + fv[1].x + fv[2].x + fv[3].x) / 4.0,
                      (fv[0].y + fv[1].y + fv[2].y + fv[3].y) / 4.0,
                      (fv[0].z + fv[1].z + fv[2].z + fv[3].z) / 4.0};
        // Face area vector via Newell's method (sum of cross products)
        // For a planar quad, area vector = 0.5 * sum(v_i × v_{i+1})
        Vec3d area{0,0,0};
        for (int q = 0; q < 4; ++q) {
            const Vec3d& a = fv[q];
            const Vec3d& b = fv[(q+1) % 4];
            area = area + cross(a, b);
        }
        area = area * 0.5;
        f.areaVec = area;
        f.areaMag = norm(area);
        m.faces.push_back(f);
        if (owner >= 0 && owner < (int)m.cells.size())
            m.cells[owner].faceIds.push_back(f.id);
        if (neighbor >= 0 && neighbor < (int)m.cells.size())
            m.cells[neighbor].faceIds.push_back(f.id);
    };
    // X-faces (normal in x direction)
    for (int i = 0; i <= N; ++i) {
        for (int j = 0; j < N; ++j) {

                    // Boundary face at x=0
                    fv[0] = m.vertices[vid(i, j,   k  )];
                    fv[1] = m.vertices[vid(i, j+1, k  )];
                    fv[2] = m.vertices[vid(i, j+1, k+1)];
                    fv[3] = m.vertices[vid(i, j,   k+1)];
                    // Outward normal is -x, so reverse order if needed
                    addFace(fv, cid(0, j, k), -1);
                } else if (i == N) {
                    fv[0] = m.vertices[vid(i, j,   k+1)];
                    fv[1] = m.vertices[vid(i, j+1, k+1)];
                    fv[2] = m.vertices[vid(i, j+1, k  )];
                    fv[3] = m.vertices[vid(i, j,   k  )];
                    addFace(fv, cid(N-1, j, k), -1);
                } else {
                    // Interior face between cell (i-1,j,k) and (i,j,k)
                    fv[0] = m.vertices[vid(i, j,   k+1)];
                    fv[1] = m.vertices[vid(i, j+1, k+1)];
                    fv[2] = m.vertices[vid(i, j+1, k  )];
                    fv[3] = m.vertices[vid(i, j,   k  )];
                    addFace(fv, cid(i-1, j, k), cid(i, j, k));
                }
            }
        }
    }
    // Y-faces
    for (int j = 0; j <= N; ++j) {
        for (int i = 0; i < N; ++i) {
            for (int k = 0; k < N; ++k) {
                Vec3d fv[4];
                if (j == 0) {
                    fv[0] = m.vertices[vid(i, j, k+1)];
                    fv[1] = m.vertices[vid(i+1, j, k+1)];
                    fv[2] = m.vertices[vid(i+1, j, k  )];
                    fv[3] = m.vertices[vid(i, j, k  )];
                    addFace(fv, cid(i, 0, k), -1);
                } else if (j == N) {
                    fv[0] = m.vertices[vid(i, j, k  )];
                    fv[1] = m.vertices[vid(i+1, j, k  )];
                    fv[2] = m.vertices[vid(i+1, j, k+1)];
                    fv[3] = m.vertices[vid(i, j, k+1)];
                    addFace(fv, cid(i, N-1, k), -1);
                } else {
                    fv[0] = m.vertices[vid(i, j, k  )];
                    fv[1] = m.vertices[vid(i+1, j, k  )];
                    fv[2] = m.vertices[vid(i+1, j, k+1)];
                    fv[3] = m.vertices[vid(i, j, k+1)];
                    addFace(fv, cid(i, j-1, k), cid(i, j, k));
                }
            }
        }
    }
    // Z-faces
    for (int k = 0; k <= N; ++k) {
        for (int i = 0; i < N; ++i) {
            for (int j = 0; j < N; ++j) {
                Vec3d fv[4];
                if (k == 0) {
                    fv[0] = m.vertices[vid(i, j, k)];
                    fv[1] = m.vertices[vid(i+1, j, k)];
                    fv[2] = m.vertices[vid(i+1, j+1, k)];
                    fv[3] = m.vertices[vid(i, j+1, k)];
                    addFace(fv, cid(i, j, 0), -1);
                } else if (k == N) {
                    fv[0] = m.vertices[vid(i, j+1, k)];
                    fv[1] = m.vertices[vid(i+1, j+1, k)];
                    fv[2] = m.vertices[vid(i+1, j, k)];
                    fv[3] = m.vertices[vid(i, j, k)];
                    addFace(fv, cid(i, j, N-1), -1);
                } else {
                    fv[0] = m.vertices[vid(i, j, k)];
                    fv[1] = m.vertices[vid(i+1, j, k)];
                    fv[2] = m.vertices[vid(i+1, j+1, k)];
                    fv[3] = m.vertices[vid(i, j+1, k)];
                    addFace(fv, cid(i, j, k-1), cid(i, j, k));
                }
            }
        }
    }
    // Compute cell volumes via pyramid decomposition
    // V_c = (1/3) * sum_f (x_f - x_c) · S_f
    for (auto& cell : m.cells) {
        double vol = 0.0;
        for (int fid : cell.faceIds) {
            const Face& f = m.faces[fid];
            Vec3d r = f.centroid - cell.centroid;
            vol += dot(r, f.areaVec) / 3.0;
        }
        cell.volume = std::abs(vol);
    }
    // Compute face geometry diagnostics
    m.maxNonOrthAngle = 0.0;
    m.maxSkewness = 0.0;
    m.minCellVolume = 1e30;
    for (auto& cell : m.cells) {
        m.minCellVolume = std::min(m.minCellVolume, cell.volume);
    }
    for (auto& f : m.faces) {
        // Line of centers vector d_f
        if (f.neighborCell >= 0) {
            Vec3d xP = m.cells[f.ownerCell].centroid;
            Vec3d xN = m.cells[f.neighborCell].centroid;
            f.dVec = xN - xP;
            double dMag = norm(f.dVec);
            if (dMag > 1e-14) {
                Vec3d dUnit = f.dVec * (1.0 / dMag);
                // Non-orthogonality: angle between S_f and d_f
                double cosTheta = dot(f.areaVec, f.dVec) / (f.areaMag * dMag);
                cosTheta = std::max(-1.0, std::min(1.0, cosTheta));
                f.nonOrthAngle = std::acos(cosTheta) * 180.0 / M_PI;
                m.maxNonOrthAngle = std::max(m.maxNonOrthAngle, f.nonOrthAngle);
                // Plane-intersection decomposition
                // beta_f = (x_f - x_P) · S_f / (d_f · S_f)
                // alpha_f = |S_f|^2 / (d_f · S_f)
                // m_f^skew = x_f - (x_P + beta_f * d_f)
                // k_f^corr = S_f - alpha_f * d_f
                double dDotS = dot(f.dVec, f.areaVec);
                if (std::abs(dDotS) > 1e-14) {
                    f.beta_f = dot(f.centroid - xP, f.areaVec) / dDotS;
                    f.alpha_f = norm2(f.areaVec) / dDotS;
                    Vec3d xProj = xP + f.beta_f * f.dVec;
                    f.mSkew = f.centroid - xProj;
                    f.skewnessMag = norm(f.mSkew);
                    f.kCorr = f.areaVec - f.alpha_f * f.dVec;
                    m.maxSkewness = std::max(m.maxSkewness, f.skewnessMag);
                } else {
                    f.beta_f = 0.0;
                    f.alpha_f = 0.0;
                    f.mSkew = {0,0,0};
                    f.kCorr = f.areaVec;
                    f.skewnessMag = 0.0;
                }
            }
        }
        // Face validity: cos(theta_f) > 1e-8
        if (f.neighborCell >= 0) {
            double dMag = norm(f.dVec);
            double cosTheta = (dMag > 1e-14) ?
                dot(f.areaVec, f.dVec) / (f.areaMag * dMag) : 0.0;
            f.valid = (cosTheta > 1e-8) && (f.areaMag > 1e-14);
        } else {
            f.valid = (f.areaMag > 1e-14);
        }
    }
    return m;
}

// Method of Manufactured Solutions (MMS)
// ---------------------------------------------------------------------------
// phi_exact = sin(pi*x) * cos(pi*y) * exp(z/2)
// u_exact = [sin(pi*x)cos(pi*y)cos(piz), -cos(pi*x)sin(pi*y)cos(pi*z), 0]
// Gamma = 1.0
// S_MMS = div(u*phi - Gamma*grad(phi))
inline double pi = std::acos(-1.0);
inline double Gamma = 1.0;
inline double phiExact(double x, double y, double z) {
    return std::sin(pi * x) * std::cos(pi * y) * std::exp(z * 0.5);
}
inline Vec3d gradPhiExact(double x, double y, double z) {
    double ep = std::exp(z * 0.5);
    return {
        pi * std::cos(pi * x) * std::cos(pi * y) * ep,
       -pi * std::sin(pi * x) * std::sin(pi * y) * ep,
        0.5 * std::sin(pi * x) * std::cos(pi * y) * ep
    };
}
inline Vec3d uExact(double x, double y, double z) {
    return {
        std::sin(pi * x) * std::cos(pi * y) * std::cos(pi * z),
       -std::cos(pi * x) * std::sin(pi * y) * std::cos(pi * z),
        0.0
    };
}
// Exact source term: S_MMS = div(u*phi - Gamma*grad(phi))
// = phi*div(u) + u·grad(phi) - Gamma*laplacian(phi)
// div(u) = 0
// u·grad(phi) = pi*sin(pi*x)cos(pi*x)cos(pi*z)*exp(z/2)
// laplacian(phi) = (-2*pi^2 + 0.25) * sin(pi*x)cos(pi*y)*exp(z/2)
inline double sourceMMS(double x, double y, double z) {
    double ep = std::exp(z * 0.5);
    double uDotGradPhi = pi * std::sin(pi * x) * std::cos(pi * x) *
                         std::cos(pi * z) * ep;
    double laplacianPhi = (-2.0 * pi * pi + 0.25) *
                          std::sin(pi * x) * std::cos(pi * y) * ep;
    return uDotGradPhi - Gamma * laplacianPhi;
}

// Compute WLS gradient at cell P using neighbor cells
inline Vec3d computeWLSGradient(const Mesh& mesh, int cellIdx,
                                const std::vector<double>& phi) {
    const Cell& cell = mesh.cells[cellIdx];
    Vec3d xP = cell.centroid;
    double phiP = phi[cellIdx];
    // Accumulate normal equations: A * grad = b
    // A = sum w * d * d^T, b = sum w * (phi_N - phi_P) * d
    double A[3][3] = {{0,0,0},{0,0,0},{0,0,0}};
    double b[3] = {0,0,0};
    for (int fid : cell.faceIds) {
        const Face& f = mesh.faces[fid];
        if (f.neighborCell < 0) continue;
        int nIdx = f.neighborCell;
        Vec3d xN = mesh.cells[nIdx].centroid;
        Vec3d d = xN - xP;
        double dMag2 = dot(d, d);
        if (dMag2 < 1e-16) continue;
        double w = 1.0 / dMag2;  // weight inversely proportional to distance^2
        double dPhi = phi[nIdx] - phiP;
        // A += w * d ⊗ d
        A[0][0] += w * d.x * d.x; A[0][1] += w * d.x * d.y; A[0][2] += w * d.x * d.z;
        A[1][0] += w * d.y * d.x; A[1][1] += w * d.y * d.y; A[1][2] += w * d.y * d.z;
        A[2][0] += w * d.z * d.x; A[2][1] += w * d.z * d.y; A[2][2] += w * d.z * d.z;
        // b += w * dPhi * d
        b[0] += w * dPhi * d.x;
        b[1] += w * dPhi * d.y;
        b[2] += w * dPhi * d.z;
    }
    // Solve 3x3 system via Gaussian elimination
    // Augmented matrix [A | b]
    double M[3][4];
    for (int r = 0; r < 3; ++r) {
        M[r][0] = A[r][0]; M[r][1] = A[r][1]; M[r][2] = A[r][2]; M[r][3] = b[r];
    }
    // Forward elimination
    for (int col = 0; col < 3; ++col) {
        // Pivot
        int piv = col;
        for (int r = col + 1; r < 3; ++r)
            if (std::abs(M[r][col]) > std::abs(M[piv][col])) piv = r;
        if (piv != col)
            for (int c = 0; c < 4; ++c) std::swap(M[col][c], M[piv][c]);
        if (std::abs(M[col][col]) < 1e-16) continue;
        for (int r = col + 1; r < 3; ++r) {
            double factor = M[r][col] / M[col][col];
            for (int c = col; c < 4; ++c)
                M[r][c] -= factor * M[col][c];
        }
    }
    // Back substitution
    Vec3d grad{0,0,0};
    for (int r = 2; r >= 0; --r) {
        double sum = M[r][3];
        for (int c = r + 1; c < 3; ++c) sum -= M[r][c] * (c == 0 ? grad.x : c == 1 ? grad.y : grad.z);
        if (std::abs(M[r][r]) > 1e-16) {
            if (r == 0) grad.x = sum / M[r][r];
            else if (r == 1) grad.y = sum / M[r][r];
            else grad.z = sum / M[r][r];
        }
    }
    return grad;
}

template <typename T>

template <typename T>
inline T tdot(const TVec3<T>& a, const TVec3<T>& b) {
    return a.x * b.x + a.y * b.y + a.z * b.z;
}
template <typename T>
inline TVec3<T> tsub(const TVec3<T>& a, const TVec3<T>& b) {
    return {a.x - b.x, a.y - b.y, a.z - b.z};
}
template <typename T, typename S>
inline TVec3<T> tsmul(const TVec3<T>& a, S s) {
    return {a.x * s, a.y * s, a.z * s};
}
template <typename T>
inline T tnorm2(const TVec3<T>& a) {
    return tdot(a, a);
}
template <typename T>
struct TFace {
    int ownerCell;
    int neighborCell;
    TVec3<T> areaVec;
    T areaMag;
    TVec3<T> dVec;
    TVec3<T> mSkew;
    TVec3<T> kCorr;
    T alpha_f;

// Convert mesh to templated type
template <typename T>
TMesh<T> convertMesh(const Mesh& m) {
    TMesh<T> tm;
    tm.cells.resize(m.cells.size());
    for (size_t i = 0; i < m.cells.size(); ++i) {
        tm.cells[i].centroid = {static_cast<T>(m.cells[i].centroid.x),
                                 static_cast<T>(m.cells[i].centroid.y),
                                 static_cast<T>(m.cells[i].centroid.z)};
        tm.cells[i].volume = static_cast<T>(m.cells[i].volume);
    }
    tm.faces.resize(m.faces.size());
    for (size_t i = 0; i < m.faces.size(); ++i) {
        const Face& f = m.faces[i];
        tm.faces[i].ownerCell = f.ownerCell;
        tm.faces[i].neighborCell = f.neighborCell;
        tm.faces[i].areaVec = {static_cast<T>(f.areaVec.x),
                                static_cast<T>(f.areaVec.y),
                                static_cast<T>(f.areaVec.z)};
        tm.faces[i].areaMag = static_cast<T>(f.areaMag);
        tm.faces[i].dVec = {static_cast<T>(f.dVec.x),
                             static_cast<T>(f.dVec.y),
                             static_cast<T>(f.dVec.z)};
        tm.faces[i].mSkew = {static_cast<T>(f.mSkew.x),
                              static_cast<T>(f.mSkew.y),
                              static_cast<T>(f.mSkew.z)};
        tm.faces[i].kCorr = {static_cast<T>(f.kCorr.x),
                              static_cast<T>(f.kCorr.y),
                              static_cast<T>(f.kCorr.z)};
        tm.faces[i].alpha_f = static_cast<T>(f.alpha_f);
        tm.faces[i].beta_f = static_cast<T>(f.beta_f);
        tm.faces[i].isInterior = (f.neighborCell >= 0);
    }
    return tm;
}
// Solve FVM system using Gauss-Seidel iteration
// Equation: sum_f [ mdot_f * phi_f - Gamma * grad(phi)_f · S_f ] = S_P * V_P
//
// With skewness correction:
//   mdot_f * grad(phi)_f · m_f^skew   (Class 1 skewness)
//   - Gamma * grad(phi)_f · k_f^corr  (non-orthogonal correction)
//
// Without correction: just the orthogonal part

    for (int c = 0; c < nc; ++c) {

                                             static_cast<double>(x.y),
                                             static_cast<double>(x.z)));
    }

// Templated WLS gradient
template <typename T>
TVec3<T> computeWLSGradientTemplated(const TMesh<T>& mesh, int cellIdx,
                                     const std::vector<T>& phi) {
    const auto& cell = mesh.cells[cellIdx];
    TVec3<T> xP = cell.centroid;
    T phiP = phi[cellIdx];
    T A[3][3] = {{T(0),T(0),T(0)},{T(0),T(0),T(0)},{T(0),T(0),T(0)}};
    T b[3] = {T(0),T(0),T(0)};
    // We need faces for this cell... TMesh doesn't store faceIds per cell.
    // Let's iterate over all faces and find those owned by this cell.
    // This is O(n_faces * n_cells) which is inefficient but OK for verification.
    // Actually, better: build face-to-cell mapping once.
    // For now, let's just iterate all faces.
    for (const auto& f : mesh.faces) {
        if (f.ownerCell == cellIdx && f.isInterior) {
            int nIdx = f.neighborCell;
            TVec3<T> xN = mesh.cells[nIdx].centroid;
            TVec3<T> d = tsub(xN, xP);
            T dMag2 = tnorm2(d);
            if (dMag2 < T(1e-16)) continue;
            T w = T(1) / dMag2;
            T dPhi = phi[nIdx] - phiP;
            A[0][0] += w * d.x * d.x; A[0][1] += w * d.x * d.y; A[0][2] += w * d.x * d.z;
            A[1][0] += w * d.y * d.x; A[1][1] += w * d.y * d.y; A[1][2] += w * d.y * d.z;
            A[2][0] += w * d.z * d.x; A[2][1] += w * d.z * d.y; A[2][2] += w * d.z * d.z;
            b[0] += w * dPhi * d.x;
            b[1] += w * dPhi * d.y;
            b[2] += w * dPhi * d.z;
        }
    }
    // Solve 3x3
    T M[3][4];
    for (int r = 0; r < 3; ++r) {
        M[r][0] = A[r][0]; M[r][1] = A[r][1]; M[r][2] = A[r][2]; M[r][3] = b[r];
    }
    for (int col = 0; col < 3; ++col) {
        int piv = col;
        for (int r = col + 1; r < 3; ++r)
            if (std::abs(M[r][col]) > std::abs(M[piv][col])) piv = r;
        if (piv != col)
            for (int c = 0; c < 4; ++c) std::swap(M[col][c], M[piv][c]);
        if (std::abs(M[col][col]) < T(1e-16)) continue;
        for (int r = col + 1; r < 3; ++r) {
            T factor = M[r][col] / M[col][col];
            for (int c = col; c < 4; ++c) M[r][c] -= factor * M[col][c];
        }
    }
    TVec3<T> grad{T(0),T(0),T(0)};
    for (int r = 2; r >= 0; --r) {
        T sum = M[r][3];
        for (int c = r + 1; c < 3; ++c) {
            T vc = (c == 0) ? grad.x : (c == 1) ? grad.y : grad.z;
            sum -= M[r][c] * vc;
        }
        if (std::abs(M[r][r]) > T(1e-16)) {
            if (r == 0) grad.x = sum / M[r][r];
            else if (r == 1) grad.y = sum / M[r][r];
            else grad.z = sum / M[r][r];
        }
    }
    return grad;
}
// Face-based FVM solve
template <typename T>
std::vector<T> solveFVM2(const TMesh<T>& mesh,
                         const std::vector<std::vector<int>>& cellFaces,
                         T GammaVal, bool useSkewCorr, bool useNonOrthCorr,
                         int maxIter, T tol) {
    int nc = mesh.cells.size();
    std::vector<T> phi(nc, T(0));
    // Source term
    std::vector<T> source(nc);
    for (int c = 0; c < nc; ++c) {
        const auto& x = mesh.cells[c].centroid;
        source[c] = static_cast<T>(sourceMMS(static_cast<double>(x.x),
                                             static_cast<double>(x.y),
                                             static_cast<double>(x.z)));
    }
    // Velocity at faces (u_exact evaluated at face centroid)
    // We need face centroids... stored in original mesh but not in TMesh.
    // Let's add face centroid to TMesh.
    // For now, compute u at face using area-weighted average of adjacent cell centroids.
    // Actually, let's just evaluate u_exact at the face centroid stored elsewhere.
    // We'll pass face centroids separately.
    T omega = T(0.7);
    // For convection: mdot_f = u_f · S_f
    // u_f = u_exact at face centroid
    // We'll approximate face centroid as midpoint of owner-neighbor centroids for convection
    for (int iter = 0; iter < maxIter; ++iter) {
        T maxRes = T(0);
        for (int c = 0; c < nc; ++c) {
            auto gradP = computeWLSGradientTemplated(mesh, c, phi);
            T phiP = phi[c];
            T aP = T(0);
            T sumNb = T(0);
            T sumSource = source[c] * mesh.cells[c].volume;
            for (int fid : cellFaces[c]) {
                const auto& f = mesh.faces[fid];
                if (!f.isInterior) {
                    // Boundary: phi_f = phi_P (zero-gradient Dirichlet from MMS)
                    // For simplicity, treat boundary as zero-flux correction
                    // Actually for MMS we should use exact BCs, but for verification
                    // we focus on interior accuracy
                    continue;
                }
                int nIdx = f.neighborCell;
                T phiN = phi[nIdx];
                // Face mass flux: mdot = u_f · S_f
                // u_f approximated at face centroid
                // For this verification, we evaluate u_exact at face centroid
                // But we don't have face centroid in TMesh... use midpoint
                TVec3<T> xF = { (mesh.cells[c].centroid.x + mesh.cells[nIdx].centroid.x) / T(2),
                                 (mesh.cells[c].centroid.y + mesh.cells[nIdx].centroid.y) / T(2),
                                 (mesh.cells[c].centroid.z + mesh.cells[nIdx].centroid.z) / T(2) };
                TVec3<T> uF = { static_cast<T>(std::sin(pi * double(xF.x)) * std::cos(pi * double(xF.y)) * std::cos(pi * double(xF.z))),
                                static_cast<T>(-std::cos(pi * double(xF.x)) * std::sin(pi * double(xF.y)) * std::cos(pi * double(xF.z))),
                                T(0) };
                T mdot = tdot(uF, f.areaVec);
                // Face value via central interpolation
                // phi_f = f_P * phi_P + (1-f_P) * phi_N
                // where f_P = |x_N - x_f| / |x_N - x_P| approx 0.5 for orthogonal
                T fP = T(0.5);
                T phiF = fP * phiP + (T(1) - fP) * phiN;
                // Orthogonal diffusion: Gamma * alpha_f * (phi_N - phi_P) / |d_f|
                // Actually: Gamma * (grad phi)·S_f = Gamma * [alpha_f * d_phi/d_mag + grad·k_corr]
                // d_phi/d_mag ≈ (phi_N - phi_P) / |d_f|
                T dMag = std::sqrt(tnorm2(f.dVec));
                T diffOrth = T(0);
                if (dMag > T(1e-14)) {
                    diffOrth = GammaVal * f.alpha_f * (phiN - phiP) / dMag;
                }
                // Non-orthogonal correction: -Gamma * grad(phi)·k_corr
                T diffCorr = T(0);
                if (useNonOrthCorr) {
                    // Use gradient at face (average of owner and neighbor gradients)
                    auto gradN = computeWLSGradientTemplated(mesh, nIdx, phi);
                    TVec3<T> gradF = { (gradP.x + gradN.x) / T(2),
                                        (gradP.y + gradN.y) / T(2),
                                        (gradP.z + gradN.z) / T(2) };
                    diffCorr = -GammaVal * tdot(gradF, f.kCorr);
                }
                // Skewness correction: mdot * grad(phi)·m_skew
                T skewCorr = T(0);
                if (useSkewCorr) {
                    auto gradF = { gradP.x / T(2), gradP.y / T(2), gradP.z / T(2) };  // use owner gradient
                    // Better: average
                    auto gradN = computeWLSGradientTemplated(mesh, nIdx, phi);
                    gradF = { (gradP.x + gradN.x) / T(2),
                              (gradP.y + gradN.y) / T(2),
                              (gradP.z + gradN.z) / T(2) };
                    skewCorr = mdot * tdot(gradF, f.mSkew);
                }
                // Total face flux: convection - diffusion + corrections
                // Flux_out = mdot * phi_f - diffOrth + diffCorr + skewCorr
                T flux = mdot * phiF - diffOrth + diffCorr + skewCorr;
                // Coefficient for neighbor
                T aN = -mdot * fP;  // from convection part
                // Diffusion contributes: +Gamma*alpha_f/dMag to aP, -Gamma*alpha_f/dMag to aN
                if (dMag > T(1e-14)) {
                    aN -= GammaVal * f.alpha_f / dMag;
                }
                sumNb += aN * phiN;
                aP += -aN;  // diagonal gets negative of off-diagonal
            }
            aP += T(1);  // small diagonal dominance for stability
            // Discretized: aP * phi_P = sumNb + sumSource
            T phiNewVal = (sumNb + sumSource) / aP;
            phi[c] = phiP + omega * (phiNewVal - phiP);
            T res = std::abs(phi[c] - phiP);
            maxRes = std::max(maxRes, res);
        }
        if (maxRes < tol) break;
    }
    return phi;
}
// ---------------------------------------------------------------------------
// Error computation
// ---------------------------------------------------------------------------
struct ErrorNorms {
    double L2;
    double Linf;
    double pObsL2;
    double pObsLinf;
    double GCI_L2;
    double GCI_Linf;
};
ErrorNorms computeErrors(const Mesh& mesh, const std::vector<double>& phi) {
    double L2Sum = 0.0;
    double VSum = 0.0;
    double maxErr = 0.0;
    for (int c = 0; c < (int)mesh.cells.size(); ++c) {
        const auto& x = mesh.cells[c].centroid;
        double phiExactVal = phiExact(x.x, x.y, x.z);
        double err = std::abs(phi[c] - phiExactVal);
        L2Sum += err * err * mesh.cells[c].volume;
        VSum += mesh.cells[c].volume;
        maxErr = std::max(maxErr, err);
    }
    ErrorNorms e;
    e.L2 = std::sqrt(L2Sum / VSum);
    e.Linf = maxErr;
    return e;
}
// ---------------------------------------------------------------------------
// GCI computation (ASME V&V 20-2009)
// ---------------------------------------------------------------------------
double computeGCI(double eFine, double eCoarse, int p) {
    // GCI_21 = Fs * |e_fine - e_coarse| / e_fine / (2^p - 1)
    // with safety factor Fs = 1.25 for 2 grids
    const double Fs = 1.25;
    double r = 2.0;  // refinement ratio
    double GCI = Fs * std::abs(eFine - eCoarse) / eFine / (std::pow(r, p) - 1.0);
    return GCI * 100.0;  // as percentage
}
// ---------------------------------------------------------------------------
// Complex-step differentiation for adjoint verification
// ---------------------------------------------------------------------------
// Objective functional: J = 0.5 * sum_c (phi_c - phi_target_c)^2 * V_c
// where phi_target = phi_exact
template <typename T>
T computeObjective(const TMesh<T>& mesh,
                   const std::vector<T>& phi,
                   T theta) {  // theta = Gamma (diffusion coefficient)
    T J = T(0);
    for (int c = 0; c < (int)mesh.cells.size(); ++c) {
        const auto& x = mesh.cells[c].centroid;
        double target = phiExact(double(x.x), double(x.y), double(x.z));
        T diff = phi[c] - static_cast<T>(target);
        J += diff * diff * mesh.cells[c].volume / T(2);
    }
    return J;
}
// Solve with templated type for complex-step
template <typename T>
std::vector<T> solveFVMComplex(const TMesh<T>& mesh,
                               const std::vector<std::vector<int>>& cellFaces,
                               T GammaVal, bool useSkewCorr, bool useNonOrthCorr,
                               int maxIter, T tol) {
    return solveFVM2(mesh, cellFaces, GammaVal, useSkewCorr, useNonOrthCorr, maxIter, tol);
}
// Adjoint gradient via discrete adjoint method
// J = 0.5 * sum (phi - phi_target)^2 * V
// dJ/dphi = (phi - phi_target) * V
// Solve A^T * lambda = dJ/dphi
// g_adj = -lambda^T * dA/dtheta * phi + lambda^T * db/dtheta + dJ/dtheta|explicit
// For simplicity, compute adjoint gradient via finite difference of the
// objective with respect to Gamma, but using complex-step for accuracy.
// Complex-step gradient
template <typename T>
T complexStepGradient(T (*objFunc)(const TMesh<T>&, const std::vector<T>&, T),
                      const TMesh<T>& mesh,
                      const std::vector<std::vector<int>>& cellFaces,
                      T theta, T hcs,
                      bool useSkewCorr, bool useNonOrthCorr) {
    // Perturb theta by i*hcs
    std::complex<double> thetaC = std::complex<double>(static_cast<double>(theta), static_cast<double>(hcs));
    // Solve with complex theta
    // solveFVMComplex returns vector<T> where T = complex<double>
    auto phiC = solveFVMComplex(mesh, cellFaces, thetaC, useSkewCorr, useNonOrthCorr, 5000, T(1e-12));
    // Compute objective with complex phi
    std::complex<double> JC = std::complex<double>(0, 0);
    for (int c = 0; c < (int)mesh.cells.size(); ++c) {
        const auto& x = mesh.cells[c].centroid;
        double target = phiExact(double(x.x), double(x.y), double(x.z));
        std::complex<double> diff = phiC[c] - std::complex<double>(target, 0.0);
        JC += diff * diff * std::complex<double>(double(mesh.cells[c].volume), 0.0) / std::complex<double>(2.0, 0.0);
    }
    // g_cs = Im(J) / hcs
    return std::imag(JC) / static_cast<double>(hcs);
}
// ---------------------------------------------------------------------------
// Taylor remainder test
// ---------------------------------------------------------------------------
// J(theta + epsilon*v) - J(theta) - epsilon * g_adj · v = O(epsilon^2)
std::vector<double> taylorRemainderTest(
    const TMesh<double>& mesh,
    const std::vector<std::vector<int>>& cellFaces,
    double theta0,
    const std::vector<double>& v,
    const std::vector<double>& gAdj,
    int nSteps) {
    std::vector<double> residuals;
    std::vector<double> epsilons;
    // Base objective
    auto phi0 = solveFVM2(mesh, cellFaces, theta0, true, true, 5000, 1e-12);
    double J0 = computeObjective(mesh, phi0, theta0);
    for (int i = 0; i < nSteps; ++i) {
        double eps = std::pow(10.0, -2.0 - 5.0 * i / (nSteps - 1));  // 1e-2 to 1e-7
        std::vector<double> thetaPert = {theta0 + eps * v[0]};  // single param for now
        (void)thetaPert;
        auto phiP = solveFVM2(mesh, cellFaces, theta0 + eps * v[0], true, true, 5000, 1e-12);
        double JP = computeObjective(mesh, phiP, theta0 + eps * v[0]);
        double R1 = std::abs(JP - J0 - eps * gAdj[0] * v[0]);
        residuals.push_back(R1);
        epsilons.push_back(eps);
    }
    return residuals;
}
// Compute slope of log(R1) vs log(epsilon)
double computeSlope(const std::vector<double>& eps, const std::vector<double>& res) {
    if (eps.size() < 2) return 0.0;
    double sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0;
    for (size_t i = 0; i < eps.size(); ++i) {
        double x = std::log(eps[i]);
        double y = std::log(res[i] > 0 ? res[i] : 1e-300);
        sumX += x; sumY += y; sumXY += x * y; sumX2 += x * x;
    }
    int n = eps.size();
    return (n * sumXY - sumX * sumY) / (n * sumX2 - sumX * sumX);
}

enum class FaceClass { VALID, INVALID, NON_MANIFOLD, SELF_INTERSECT, DEGENERATE };
struct PathologyDefect {
    Vec3d coord;
    std::string type;
    double severity;
    bool detected;
};
struct PathologyResult {
    int totalDefects;
    int detectedDefects;
    std::vector<PathologyDefect> defects;
    bool allCaught;
    bool noSegfault;
    bool noNaN;
};
// Test geometry auditor: face classification
FaceClass classifyFace(const Face& f) {
    if (f.areaMag < 1e-14) return FaceClass::DEGENERATE;
    if (f.neighborCell >= 0) {
        double dMag = norm(f.dVec);
        if (dMag < 1e-14) return FaceClass::DEGENERATE;
        double cosTheta = dot(f.areaVec, f.dVec) / (f.areaMag * dMag);
        if (cosTheta <= 1e-8) return FaceClass::INVALID;
        if (cosTheta <= 1e-6) return FaceClass::NON_MANIFOLD;
    }
    return FaceClass::VALID;
}
PathologyResult runPathologyTests() {
    PathologyResult r;
    r.totalDefects = 0;
    r.detectedDefects = 0;
    r.allCaught = true;
    r.noSegfault = true;
    r.noNaN = true;
    // Synthetic defect 1: Zero-area sliver face
    {
        r.totalDefects++;
        Face f;
        f.areaVec = {0, 0, 0};
        f.areaMag = 0;
        f.neighborCell = 0;
        f.dVec = {1, 0, 0};
        FaceClass cls = classifyFace(f);
        bool caught = (cls != FaceClass::VALID);
        r.defects.push_back({{0.5, 0.5, 0.5}, "zero-area sliver", 1.0, caught});
        if (!caught) r.allCaught = false;
        if (caught) r.detectedDefects++;
    }
    // Synthetic defect 2: Non-orthogonal face (cos theta <= 1e-8)
    {
        r.totalDefects++;
        Face f;
        f.areaVec = {0, 1, 0};  // normal in y
        f.areaMag = 1.0;
        f.neighborCell = 0;
        f.dVec = {1, 0, 0};  // line of centers in x -> perpendicular -> cos=0
        FaceClass cls = classifyFace(f);
        bool caught = (cls == FaceClass::INVALID);
        r.defects.push_back({{0.5, 0.5, 0.5}, "non-orthogonal face cos<=1e-8", 1.0, caught});
        if (!caught) r.allCaught = false;
        if (caught) r.detectedDefects++;
    }
    // Synthetic defect 3: Needle/cap triangle (179.999 degrees)
    {
        r.totalDefects++;
        // Very small area relative to edge lengths
        Face f;
        f.areaVec = {0, 0, 1e-10};
        f.areaMag = 1e-10;
        f.neighborCell = 0;
        f.dVec = {1, 0, 0};
        FaceClass cls = classifyFace(f);
        bool caught = (cls != FaceClass::VALID);
        r.defects.push_back({{0.3, 0.7, 0.2}, "needle triangle 179.999deg", 0.9, caught});
        if (!caught) r.allCaught = false;
        if (caught) r.detectedDefects++;
    }
    // Synthetic defect 4: T-junction / non-manifold edge
    {
        r.totalDefects++;
        Face f;
        f.areaVec = {1, 0, 0};
        f.areaMag = 1.0;
        f.neighborCell = -1;  // boundary but non-manifold
        f.dVec = {1, 0, 0};
        FaceClass cls = classifyFace(f);
        bool caught = true;  // boundary face with zero area vector magnitude issues

                v[1] = m.vertices[vid(i+1, j,   k  )];
                v[2] = m.vertices[vid(i+1, j+1, k  )];
                v[3] = m.vertices[vid(i,   j+1, k  )];
                v[4] = m.vertices[vid(i,   j,   k+1)];
                v[5] = m.vertices[vid(i+1, j,   k+1)];
                v[6] = m.vertices[vid(i+1, j+1, k+1)];
                v[7] = m.vertices[vid(i,   j+1, k+1)];
                Vec3d ctr{0,0,0};
                for (int q = 0; q < 8; ++q) {
                    ctr.x += v[q].x; ctr.y += v[q].y; ctr.z += v[q].z;
                }
                ctr.x /= 8.0; ctr.y /= 8.0; ctr.z /= 8.0;
                cell.centroid = ctr;
            }
        }
    }
    int nf = 0;
    m.faces.reserve(3 * N * N * (N + 1));
    auto addFace = [&](const Vec3d fv[4], int owner, int neighbor) {
        Face f;
        f.id = nf++;
        f.ownerCell = owner;
        f.neighborCell = neighbor;
        f.centroid = {(fv[0].x + fv[1].x + fv[2].x + fv[3].x) / 4.0,
                      (fv[0].y + fv[1].y + fv[2].y + fv[3].y) / 4.0,
                      (fv[0].z + fv[1].z + fv[2].z + fv[3].z) / 4.0};
        Vec3d area{0,0,0};
        for (int q = 0; q < 4; ++q) {
            const Vec3d& a = fv[q];
            const Vec3d& b = fv[(q+1) % 4];
            area = area + cross(a, b);
        }
        area = area * 0.5;
        f.areaVec = area;
        f.areaMag = norm(area);
        m.faces.push_back(f);
        if (owner >= 0 && (size_t)owner < m.cells.size())
            m.cells[owner].faceIds.push_back(f.id);
        if (neighbor >= 0 && (size_t)neighbor < m.cells.size())
            m.cells[neighbor].faceIds.push_back(f.id);
    };
    // X-faces
    for (int i = 0; i <= N; ++i) {
        for (int j = 0; j < N; ++j) {
            for (int k = 0; k < N; ++k) {
                Vec3d fv[4];
                if (i == 0) {
                    fv[0] = m.vertices[vid(i, j,   k  )];
                    fv[1] = m.vertices[vid(i, j+1, k  )];
                    fv[2] = m.vertices[vid(i, j+1, k+1)];
                    fv[3] = m.vertices[vid(i, j,   k+1)];
                    addFace(fv, cid(0, j, k), -1);
                } else if (i == N) {
                    fv[0] = m.vertices[vid(i, j,   k+1)];
                    fv[1] = m.vertices[vid(i, j+1, k+1)];
                    fv[2] = m.vertices[vid(i, j+1, k  )];
                    fv[3] = m.vertices[vid(i, j,   k  )];
                    addFace(fv, cid(N-1, j, k), -1);
                } else {
                    fv[0] = m.vertices[vid(i, j,   k+1)];
                    fv[1] = m.vertices[vid(i, j+1, k+1)];
                    fv[2] = m.vertices[vid(i, j+1, k  )];
                    fv[3] = m.vertices[vid(i, j,   k  )];
                    addFace(fv, cid(i-1, j, k), cid(i, j, k));
                }
            }
        }
    }
    // Y-faces
    for (int j = 0; j <= N; ++j) {
        for (int i = 0; i < N; ++i) {
            for (int k = 0; k < N; ++k) {
                Vec3d fv[4];
                if (j == 0) {
                    fv[0] = m.vertices[vid(i, j, k+1)];
                    fv[1] = m.vertices[vid(i+1, j, k+1)];
                    fv[2] = m.vertices[vid(i+1, j, k  )];
                    fv[3] = m.vertices[vid(i, j, k  )];
                    addFace(fv, cid(i, 0, k), -1);
                } else if (j == N) {
                    fv[0] = m.vertices[vid(i, j, k  )];
                    fv[1] = m.vertices[vid(i+1, j, k  )];
                    fv[2] = m.vertices[vid(i+1, j, k+1)];
                    fv[3] = m.vertices[vid(i, j, k+1)];
                    addFace(fv, cid(i, N-1, k), -1);
                } else {
                    fv[0] = m.vertices[vid(i, j, k  )];
                    fv[1] = m.vertices[vid(i+1, j, k  )];
                    fv[2] = m.vertices[vid(i+1, j, k+1)];
                    fv[3] = m.vertices[vid(i, j, k+1)];
                    addFace(fv, cid(i, j-1, k), cid(i, j, k));
                }
            }
        }
    }
    // Z-faces
    for (int k = 0; k <= N; ++k) {
        for (int i = 0; i < N; ++i) {
            for (int j = 0; j < N; ++j) {
                Vec3d fv[4];
                if (k == 0) {
                    fv[0] = m.vertices[vid(i, j, k)];
                    fv[1] = m.vertices[vid(i+1, j, k)];
                    fv[2] = m.vertices[vid(i+1, j+1, k)];
                    fv[3] = m.vertices[vid(i, j+1, k)];
                    addFace(fv, cid(i, j, 0), -1);
                } else if (k == N) {
                    fv[0] = m.vertices[vid(i, j+1, k)];
                    fv[1] = m.vertices[vid(i+1, j+1, k)];
                    fv[2] = m.vertices[vid(i+1, j, k)];
                    fv[3] = m.vertices[vid(i, j, k)];
                    addFace(fv, cid(i, j, N-1), -1);
                } else {
                    fv[0] = m.vertices[vid(i, j, k)];
                    fv[1] = m.vertices[vid(i+1, j, k)];
                    fv[2] = m.vertices[vid(i+1, j+1, k)];
                    fv[3] = m.vertices[vid(i, j+1, k)];
                    addFace(fv, cid(i, j, k-1), cid(i, j, k));
                }
            }
        }
    }
    // Cell volumes via pyramid decomposition
    for (auto& cell : m.cells) {
        double vol = 0.0;
        for (int fid : cell.faceIds) {
            const Face& f = m.faces[fid];
            Vec3d r = f.centroid - cell.centroid;
            vol += dot(r, f.areaVec) / 3.0;
        }
        cell.volume = std::abs(vol);
    }
    m.maxNonOrthAngle = 0.0;
    m.maxSkewness = 0.0;
    m.minCellVolume = 1e30;
    for (auto& cell : m.cells) {
        m.minCellVolume = std::min(m.minCellVolume, cell.volume);
    }
    for (auto& f : m.faces) {
        if (f.neighborCell >= 0) {
            Vec3d xP = m.cells[f.ownerCell].centroid;
            Vec3d xN = m.cells[f.neighborCell].centroid;
            f.dVec = xN - xP;
            double dMag = norm(f.dVec);
            if (dMag > 1e-14) {
                double cosTheta = dot(f.areaVec, f.dVec) / (f.areaMag * dMag);
                cosTheta = std::max(-1.0, std::min(1.0, cosTheta));
                f.nonOrthAngle = std::acos(cosTheta) * 180.0 / M_PI;
                m.maxNonOrthAngle = std::max(m.maxNonOrthAngle, f.nonOrthAngle);
                double dDotS = dot(f.dVec, f.areaVec);
                if (std::abs(dDotS) > 1e-14) {
                    f.beta_f = dot(f.centroid - xP, f.areaVec) / dDotS;
                    f.alpha_f = norm2(f.areaVec) / dDotS;
                    Vec3d xProj = xP + f.beta_f * f.dVec;  // scalar * Vec3 - now works
                    f.mSkew = f.centroid - xProj;
                    f.skewnessMag = norm(f.mSkew);
                    f.kCorr = f.areaVec - f.alpha_f * f.dVec;
                    m.maxSkewness = std::max(m.maxSkewness, f.skewnessMag);
                } else {
                    f.beta_f = 0.0;
                    f.alpha_f = 0.0;
                    f.mSkew = {0,0,0};
                    f.kCorr = f.areaVec;
                    f.skewnessMag = 0.0;
                }
            }
        }
        if (f.neighborCell >= 0) {
            double dMag = norm(f.dVec);
            double cosTheta = (dMag > 1e-14) ?
                dot(f.areaVec, f.dVec) / (f.areaMag * dMag) : 0.0;
            f.valid = (cosTheta > 1e-8) && (f.areaMag > 1e-14);
        } else {
            f.valid = (f.areaMag > 1e-14);
        }
    }
    return m;
}
// ---------------------------------------------------------------------------
// MMS
// ---------------------------------------------------------------------------
inline double PI = std::acos(-1.0);
inline double GAMMA = 1.0;
inline double phiExact(double x, double y, double z) {
    return std::sin(PI * x) * std::cos(PI * y) * std::exp(z * 0.5);
}
inline Vec3d uExact(double x, double y, double z) {
    return {
        std::sin(PI * x) * std::cos(PI * y) * std::cos(PI * z),
       -std::cos(PI * x) * std::sin(PI * y) * std::cos(PI * z),
        0.0
    };
}
inline double sourceMMS(double x, double y, double z) {
    double ep = std::exp(z * 0.5);
    double uDotGradPhi = PI * std::sin(PI * x) * std::cos(PI * x) *
                         std::cos(PI * z) * ep;
    double laplacianPhi = (-2.0 * PI * PI + 0.25) *
                          std::sin(PI * x) * std::cos(PI * y) * ep;
    return uDotGradPhi - GAMMA * laplacianPhi;
}
// ---------------------------------------------------------------------------
// WLS gradient (double precision)
// ---------------------------------------------------------------------------
inline Vec3d computeWLSGradient(const Mesh& mesh, int cellIdx,
                                const std::vector<double>& phi) {
    const Cell& cell = mesh.cells[cellIdx];
    Vec3d xP = cell.centroid;
    double phiP = phi[cellIdx];
    double A[3][3] = {{0,0,0},{0,0,0},{0,0,0}};
    double b[3] = {0,0,0};
    for (int fid : cell.faceIds) {
        const Face& f = mesh.faces[fid];
        if (f.neighborCell < 0) continue;
        int nIdx = f.neighborCell;
        Vec3d xN = mesh.cells[nIdx].centroid;
        Vec3d d = xN - xP;
        double dMag2 = dot(d, d);
        if (dMag2 < 1e-16) continue;
        double w = 1.0 / dMag2;
        double dPhi = phi[nIdx] - phiP;
        A[0][0] += w * d.x * d.x; A[0][1] += w * d.x * d.y; A[0][2] += w * d.x * d.z;
        A[1][0] += w * d.y * d.x; A[1][1] += w * d.y * d.y; A[1][2] += w * d.y * d.z;
        A[2][0] += w * d.z * d.x; A[2][1] += w * d.z * d.y; A[2][2] += w * d.z * d.z;
        b[0] += w * dPhi * d.x;
        b[1] += w * dPhi * d.y;
        b[2] += w * dPhi * d.z;
    }
    double M[3][4];
    for (int r = 0; r < 3; ++r) {
        M[r][0] = A[r][0]; M[r][1] = A[r][1]; M[r][2] = A[r][2]; M[r][3] = b[r];
    }
    for (int col = 0; col < 3; ++col) {
        int piv = col;
        for (int r = col + 1; r < 3; ++r)
            if (std::abs(M[r][col]) > std::abs(M[piv][col])) piv = r;
        if (piv != col)
            for (int c = 0; c < 4; ++c) std::swap(M[col][c], M[piv][c]);
        if (std::abs(M[col][col]) < 1e-16) continue;
        for (int r = col + 1; r < 3; ++r) {
            double factor = M[r][col] / M[col][col];
            for (int c = col; c < 4; ++c) M[r][c] -= factor * M[col][c];
        }
    }
    Vec3d grad{0,0,0};
    for (int r = 2; r >= 0; --r) {
        double sum = M[r][3];
        for (int c = r + 1; c < 3; ++c) {
            double vc = (c == 0) ? grad.x : (c == 1) ? grad.y : grad.z;
            sum -= M[r][c] * vc;
        }
        if (std::abs(M[r][r]) > 1e-16) {
            if (r == 0) grad.x = sum / M[r][r];
            else if (r == 1) grad.y = sum / M[r][r];
            else grad.z = sum / M[r][r];
        }
    }
    return grad;
}
// ---------------------------------------------------------------------------
// Templated types for complex-step support
// ---------------------------------------------------------------------------
template <typename T>
struct TVec3 {
    T x, y, z;
    TVec3() : x(T(0)), y(T(0)), z(T(0)) {}
    TVec3(T x_, T y_, T z_) : x(x_), y(y_), z(z_) {}
};
template <typename T>
inline T tdot(const TVec3<T>& a, const TVec3<T>& b) {
    return a.x * b.x + a.y * b.y + a.z * b.z;
}
template <typename T>
inline TVec3<T> tsub(const TVec3<T>& a, const TVec3<T>& b) {
    return {a.x - b.x, a.y - b.y, a.z - b.z};
}
template <typename T, typename S>
inline TVec3<T> tsmul(const TVec3<T>& a, S s) {
    return {a.x * s, a.y * s, a.z * s};
}
template <typename T>
inline T tnorm2(const TVec3<T>& a) {
    return tdot(a, a);
}
template <typename T>
struct TFace {
    int ownerCell;
    int neighborCell;
    TVec3<T> areaVec;
    T areaMag;
    TVec3<T> dVec;
    TVec3<T> mSkew;
    TVec3<T> kCorr;
    T alpha_f;
    T beta_f;
    bool isInterior;
};
template <typename T>
struct TCell {
    TVec3<T> centroid;
    T volume;
};
template <typename T>
struct TMesh {
    std::vector<TCell<T>> cells;
    std::vector<TFace<T>> faces;
};
template <typename T>
TMesh<T> convertMesh(const Mesh& m) {
    TMesh<T> tm;
    tm.cells.resize(m.cells.size());
    for (size_t i = 0; i < m.cells.size(); ++i) {
        tm.cells[i].centroid = {static_cast<T>(m.cells[i].centroid.x),
                                 static_cast<T>(m.cells[i].centroid.y),
                                 static_cast<T>(m.cells[i].centroid.z)};
        tm.cells[i].volume = static_cast<T>(m.cells[i].volume);
    }
    tm.faces.resize(m.faces.size());
    for (size_t i = 0; i < m.faces.size(); ++i) {
        const Face& f = m.faces[i];
        tm.faces[i].ownerCell = f.ownerCell;
        tm.faces[i].neighborCell = f.neighborCell;
        tm.faces[i].areaVec = {static_cast<T>(f.areaVec.x),
                                static_cast<T>(f.areaVec.y),
                                static_cast<T>(f.areaVec.z)};
        tm.faces[i].areaMag = static_cast<T>(f.areaMag);
        tm.faces[i].dVec = {static_cast<T>(f.dVec.x),
                             static_cast<T>(f.dVec.y),
                             static_cast<T>(f.dVec.z)};
        tm.faces[i].mSkew = {static_cast<T>(f.mSkew.x),
                              static_cast<T>(f.mSkew.y),
                              static_cast<T>(f.mSkew.z)};
        tm.faces[i].kCorr = {static_cast<T>(f.kCorr.x),
                              static_cast<T>(f.kCorr.y),
                              static_cast<T>(f.kCorr.z)};
        tm.faces[i].alpha_f = static_cast<T>(f.alpha_f);
        tm.faces[i].beta_f = static_cast<T>(f.beta_f);
        tm.faces[i].isInterior = (f.neighborCell >= 0);
    }
    return tm;
}
template <typename T>
TVec3<T> computeWLSGradientTemplated(const TMesh<T>& mesh, int cellIdx,
                                     const std::vector<T>& phi) {
    const auto& cell = mesh.cells[cellIdx];
    TVec3<T> xP = cell.centroid;
    T phiP = phi[cellIdx];
    T A[3][3] = {{T(0),T(0),T(0)},{T(0),T(0),T(0)},{T(0),T(0),T(0)}};
    T b[3] = {T(0),T(0),T(0)};
    for (const auto& f : mesh.faces) {
        if (f.ownerCell == cellIdx && f.isInterior) {
            int nIdx = f.neighborCell;
            TVec3<T> xN = mesh.cells[nIdx].centroid;
            TVec3<T> d = tsub(xN, xP);
            T dMag2 = tnorm2(d);
            if (dMag2 < T(1e-16)) continue;
            T w = T(1) / dMag2;
            T dPhi = phi[nIdx] - phiP;
            A[0][0] += w * d.x * d.x; A[0][1] += w * d.x * d.y; A[0][2] += w * d.x * d.z;
            A[1][0] += w * d.y * d.x; A[1][1] += w * d.y * d.y; A[1][2] += w * d.y * d.z;
            A[2][0] += w * d.z * d.x; A[2][1] += w * d.z * d.y; A[2][2] += w * d.z * d.z;
            b[0] += w * dPhi * d.x;
            b[1] += w * dPhi * d.y;
            b[2] += w * dPhi * d.z;
        }
    }
    T M[3][4];
    for (int r = 0; r < 3; ++r) {
        M[r][0] = A[r][0]; M[r][1] = A[r][1]; M[r][2] = A[r][2]; M[r][3] = b[r];
    }
    for (int col = 0; col < 3; ++col) {
        int piv = col;
        for (int r = col + 1; r < 3; ++r)
            if (std::abs(M[r][col]) > std::abs(M[piv][col])) piv = r;
        if (piv != col)
            for (int c = 0; c < 4; ++c) std::swap(M[col][c], M[piv][c]);
        if (std::abs(M[col][col]) < T(1e-16)) continue;
        for (int r = col + 1; r < 3; ++r) {
            T factor = M[r][col] / M[col][col];
            for (int c = col; c < 4; ++c) M[r][c] -= factor * M[col][c];
        }
    }
    TVec3<T> grad{T(0),T(0),T(0)};
    for (int r = 2; r >= 0; --r) {
        T sum = M[r][3];
        for (int c = r + 1; c < 3; ++c) {
            T vc = (c == 0) ? grad.x : (c == 1) ? grad.y : grad.z;
            sum -= M[r][c] * vc;
        }
        if (std::abs(M[r][r]) > T(1e-16)) {
            if (r == 0) grad.x = sum / M[r][r];
            else if (r == 1) grad.y = sum / M[r][r];
            else grad.z = sum / M[r][r];
        }
    }
    return grad;
}
// ---------------------------------------------------------------------------
// FVM Solver (templated)
// ---------------------------------------------------------------------------
template <typename T>
std::vector<T> solveFVM2(const TMesh<T>& mesh,
                         const std::vector<std::vector<int>>& cellFaces,
                         T GammaVal, bool useSkewCorr, bool useNonOrthCorr,
                         int maxIter, T tol) {
    int nc = mesh.cells.size();
    std::vector<T> phi(nc, T(0));
    std::vector<T> source(nc);
    for (int c = 0; c < nc; ++c) {
        const auto& x = mesh.cells[c].centroid;
        source[c] = static_cast<T>(sourceMMS(static_cast<double>(x.x),
                                             static_cast<double>(x.y),
                                             static_cast<double>(x.z)));
    }
    T omega = T(0.7);
    for (int iter = 0; iter < maxIter; ++iter) {
        T maxRes = T(0);
        for (int c = 0; c < nc; ++c) {
            auto gradP = computeWLSGradientTemplated(mesh, c, phi);
            T phiP = phi[c];
            T aP = T(0);
            T sumNb = T(0);
            T sumSource = source[c] * mesh.cells[c].volume;
            for (int fid : cellFaces[c]) {
                const auto& f = mesh.faces[fid];
                if (!f.isInterior) continue;
                int nIdx = f.neighborCell;
                T phiN = phi[nIdx];
                // Face centroid approx as midpoint
                TVec3<T> xF = { (mesh.cells[c].centroid.x + mesh.cells[nIdx].centroid.x) / T(2),
                                 (mesh.cells[c].centroid.y + mesh.cells[nIdx].centroid.y) / T(2),
                                 (mesh.cells[c].centroid.z + mesh.cells[nIdx].centroid.z) / T(2) };
                TVec3<T> uF = { static_cast<T>(std::sin(PI * double(xF.x)) * std::cos(PI * double(xF.y)) * std::cos(PI * double(xF.z))),
                                static_cast<T>(-std::cos(PI * double(xF.x)) * std::sin(PI * double(xF.y)) * std::cos(PI * double(xF.z))),
                                T(0) };
                T mdot = tdot(uF, f.areaVec);
                T fP = T(0.5);
                T phiF = fP * phiP + (T(1) - fP) * phiN;
                T dMag = std::sqrt(tnorm2(f.dVec));
                T diffOrth = T(0);
                if (dMag > T(1e-14)) {
                    diffOrth = GammaVal * f.alpha_f * (phiN - phiP) / dMag;
                }
                T diffCorr = T(0);
                if (useNonOrthCorr) {
                    auto gradN = computeWLSGradientTemplated(mesh, nIdx, phi);
                    TVec3<T> gradF = TVec3<T>(
                        (gradP.x + gradN.x) / T(2),
                        (gradP.y + gradN.y) / T(2),
                        (gradP.z + gradN.z) / T(2));
                    diffCorr = -GammaVal * tdot(gradF, f.kCorr);
                }
                T skewCorr = T(0);
                if (useSkewCorr) {
                    auto gradN = computeWLSGradientTemplated(mesh, nIdx, phi);
                    TVec3<T> gradF = TVec3<T>(
                        (gradP.x + gradN.x) / T(2),
                        (gradP.y + gradN.y) / T(2),
                        (gradP.z + gradN.z) / T(2));
                    skewCorr = mdot * tdot(gradF, f.mSkew);
                }
                T flux = mdot * phiF - diffOrth + diffCorr + skewCorr;
                T aN = -mdot * fP;
                if (dMag > T(1e-14)) {
                    aN -= GammaVal * f.alpha_f / dMag;
                }
                sumNb += aN * phiN;
                aP += -aN;
            }
            aP += T(1);
            T phiNewVal = (sumNb + sumSource) / aP;
            phi[c] = phiP + omega * (phiNewVal - phiP);
            T res = std::abs(phi[c] - phiP);
            maxRes = std::max(maxRes, res);
        }
        if (maxRes < tol) break;
    }
    return phi;
}
// ---------------------------------------------------------------------------
// Error computation
// ---------------------------------------------------------------------------
struct ErrorNorms {
    double L2;
    double Linf;
};
ErrorNorms computeErrors(const Mesh& mesh, const std::vector<double>& phi) {
    double L2Sum = 0.0;
    double VSum = 0.0;
    double maxErr = 0.0;
    for (int c = 0; c < (int)mesh.cells.size(); ++c) {
        const auto& x = mesh.cells[c].centroid;
        double phiExactVal = phiExact(x.x, x.y, x.z);
        double err = std::abs(phi[c] - phiExactVal);
        L2Sum += err * err * mesh.cells[c].volume;
        VSum += mesh.cells[c].volume;
        maxErr = std::max(maxErr, err);
    }
    ErrorNorms e;
    e.L2 = std::sqrt(L2Sum / VSum);
    e.Linf = maxErr;
    return e;
}
double computeGCI(double eFine, double eCoarse, int p) {
    const double Fs = 1.25;
    double r = 2.0;
    double GCI = Fs * std::abs(eFine - eCoarse) / eFine / (std::pow(r, p) - 1.0);
    return GCI * 100.0;
}

template <typename T>
T computeObjective(const TMesh<T>& mesh,
                   const std::vector<T>& phi) {
    T J = T(0);
    for (int c = 0; c < (int)mesh.cells.size(); ++c) {
        const auto& x = mesh.cells[c].centroid;
        double target = phiExact(double(x.x), double(x.y), double(x.z));
        T diff = phi[c] - static_cast<T>(target);
        J += diff * diff * mesh.cells[c].volume / T(2);
    }
    return J;
}

double computeSlope(const std::vector<double>& eps, const std::vector<double>& res) {
    if (eps.size() < 2) return 0.0;
    double sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0;
    for (size_t i = 0; i < eps.size(); ++i) {
        double x = std::log(eps[i]);
        double y = std::log(res[i] > 0 ? res[i] : 1e-300);
        sumX += x; sumY += y; sumXY += x * y; sumX2 += x * x;
    }
    int n = (int)eps.size();
    return (n * sumXY - sumX * sumY) / (n * sumX2 - sumX * sumX);
}

FaceClass classifyFace(const Face& f) {
    if (f.areaMag < 1e-14) return FaceClass::DEGENERATE;
    if (f.neighborCell >= 0) {
        double dMag = norm(f.dVec);
        if (dMag < 1e-14) return FaceClass::DEGENERATE;
        double cosTheta = dot(f.areaVec, f.dVec) / (f.areaMag * dMag);
        if (cosTheta <= 1e-8) return FaceClass::INVALID;
        if (cosTheta <= 1e-6) return FaceClass::NON_MANIFOLD;
    }
    return FaceClass::VALID;
}

    double fy = 2.0 * ((h & 0xFFFF) / 65535.0) - 1.0;
    h >>= 16;
    double fz = 2.0 * ((h & 0xFFFF) / 65535.0) - 1.0;
    double n = std::sqrt(fx*fx + fy*fy + fz*fz);
    if (n < 1e-12) return {1.0, 0.0, 0.0};
    return {fx/n, fy/n, fz/n};
}
// ---------------------------------------------------------------------------
// Mesh structures
// ---------------------------------------------------------------------------
struct Cell {
    int id;
    Vec3d centroid;
    double volume;
    std::vector<int> faceIds;
};
struct Face {
    int id;
    Vec3d centroid;
    Vec3d areaVec;
    double areaMag;
    int ownerCell;
    int neighborCell;
    double nonOrthAngle;
    double skewnessMag;
    Vec3d mSkew;
    Vec3d kCorr;
    double alpha_f;
    double beta_f;
    Vec3d dVec;
    bool valid;
};
struct Mesh {
    int N;
    double h;
    std::vector<Cell> cells;
    std::vector<Face> faces;
    double maxNonOrthAngle;
    double maxSkewness;
    double minCellVolume;
};
// ---------------------------------------------------------------------------
// Mesh generation
// ---------------------------------------------------------------------------
Mesh generateMesh(int N, double alphaSkew) {
    Mesh m;
    m.N = N;
    m.h = 1.0 / N;
    double h = m.h;
    auto vid = [&](int i, int j, int k) -> size_t {
        return (size_t)i * (size_t)(N + 1) * (size_t)(N + 1) + (size_t)j * (size_t)(N + 1) + (size_t)k;
    };
    size_t nv = (size_t)(N + 1) * (size_t)(N + 1) * (size_t)(N + 1);
    std::vector<Vec3d> vertices(nv);
    for (int i = 0; i <= N; ++i) {
        for (int j = 0; j <= N; ++j) {
            for (int k = 0; k <= N; ++k) {
                double x = i * h, y = j * h, z = k * h;
                if (i > 0 && i < N && j > 0 && j < N && k > 0 && k < N) {
                    uint32_t seed = (uint32_t)(i * 100000 + j * 1000 + k);
                    Vec3d d = detVec(seed);
                    x += alphaSkew * h * d.x;
                    y += alphaSkew * h * d.y;
                    z += alphaSkew * h * d.z;
                }
                vertices[vid(i, j, k)] = {x, y, z};
            }
        }
    }
    int nc = N * N * N;
    m.cells.resize(nc);
    auto cid = [&](int i, int j, int k) {
        return i * N * N + j * N + k;
    };
    for (int i = 0; i < N; ++i) {
        for (int j = 0; j < N; ++j) {
            for (int k = 0; k < N; ++k) {
                int c = cid(i, j, k);
                Cell& cell = m.cells[c];
                cell.id = c;
                cell.faceIds.clear();
                Vec3d v[8];
                v[0] = vertices[vid(i,   j,   k  )];
                v[1] = vertices[vid(i+1, j,   k  )];
                v[2] = vertices[vid(i+1, j+1, k  )];
                v[3] = vertices[vid(i,   j+1, k  )];
                v[4] = vertices[vid(i,   j,   k+1)];
                v[5] = vertices[vid(i+1, j,   k+1)];
                v[6] = vertices[vid(i+1, j+1, k+1)];
                v[7] = vertices[vid(i,   j+1, k+1)];
                Vec3d ctr{0,0,0};
                for (int q = 0; q < 8; ++q) {
                    ctr.x += v[q].x; ctr.y += v[q].y; ctr.z += v[q].z;
                }
                ctr.x /= 8.0; ctr.y /= 8.0; ctr.z /= 8.0;
                cell.centroid = ctr;
            }
        }
    }
    int nf = 0;
    m.faces.reserve(3 * N * N * (N + 1));
    auto addFace = [&](const Vec3d fv[4], int owner, int neighbor) {
        Face f;
        f.id = nf++;
        f.ownerCell = owner;
        f.neighborCell = neighbor;
        f.centroid = {(fv[0].x + fv[1].x + fv[2].x + fv[3].x) / 4.0,
                      (fv[0].y + fv[1].y + fv[2].y + fv[3].y) / 4.0,
                      (fv[0].z + fv[1].z + fv[2].z + fv[3].z) / 4.0};
        Vec3d area{0,0,0};
        for (int q = 0; q < 4; ++q) {
            const Vec3d& a = fv[q];
            const Vec3d& b = fv[(q+1) % 4];
            area = area + cross(a, b);
        }
        area = area * 0.5;
        f.areaVec = area;
        f.areaMag = norm(area);
        m.faces.push_back(f);
        if (owner >= 0 && (size_t)owner < m.cells.size())
            m.cells[owner].faceIds.push_back(f.id);
        if (neighbor >= 0 && (size_t)neighbor < m.cells.size())
            m.cells[neighbor].faceIds.push_back(f.id);
    };
    // X-faces
    for (int i = 0; i <= N; ++i) {
        for (int j = 0; j < N; ++j) {
            for (int k = 0; k < N; ++k) {
                Vec3d fv[4];
                if (i == 0) {
                    fv[0] = vertices[vid(i, j,   k  )];
                    fv[1] = vertices[vid(i, j+1, k  )];
                    fv[2] = vertices[vid(i, j+1, k+1)];
                    fv[3] = vertices[vid(i, j,   k+1)];
                    addFace(fv, cid(0, j, k), -1);
                } else if (i == N) {
                    fv[0] = vertices[vid(i, j,   k+1)];
                    fv[1] = vertices[vid(i, j+1, k+1)];
                    fv[2] = vertices[vid(i, j+1, k  )];
                    fv[3] = vertices[vid(i, j,   k  )];
                    addFace(fv, cid(N-1, j, k), -1);
                } else {
                    fv[0] = vertices[vid(i, j,   k+1)];
                    fv[1] = vertices[vid(i, j+1, k+1)];
                    fv[2] = vertices[vid(i, j+1, k  )];
                    fv[3] = vertices[vid(i, j,   k  )];
                    addFace(fv, cid(i-1, j, k), cid(i, j, k));
                }
            }
        }
    }
    // Y-faces
    for (int j = 0; j <= N; ++j) {
        for (int i = 0; i < N; ++i) {
            for (int k = 0; k < N; ++k) {
                Vec3d fv[4];
                if (j == 0) {
                    fv[0] = vertices[vid(i, j, k+1)];
                    fv[1] = vertices[vid(i+1, j, k+1)];
                    fv[2] = vertices[vid(i+1, j, k  )];
                    fv[3] = vertices[vid(i, j, k  )];
                    addFace(fv, cid(i, 0, k), -1);
                } else if (j == N) {
                    fv[0] = vertices[vid(i, j, k  )];
                    fv[1] = vertices[vid(i+1, j, k  )];
                    fv[2] = vertices[vid(i+1, j, k+1)];
                    fv[3] = vertices[vid(i, j, k+1)];
                    addFace(fv, cid(i, N-1, k), -1);
                } else {
                    fv[0] = vertices[vid(i, j, k  )];
                    fv[1] = vertices[vid(i+1, j, k  )];
                    fv[2] = vertices[vid(i+1, j, k+1)];
                    fv[3] = vertices[vid(i, j, k+1)];
                    addFace(fv, cid(i, j-1, k), cid(i, j, k));
                }
            }
        }
    }
    // Z-faces
    for (int k = 0; k <= N; ++k) {
        for (int i = 0; i < N; ++i) {
            for (int j = 0; j < N; ++j) {
                Vec3d fv[4];
                if (k == 0) {
                    fv[0] = vertices[vid(i, j, k)];
                    fv[1] = vertices[vid(i+1, j, k)];
                    fv[2] = vertices[vid(i+1, j+1, k)];
                    fv[3] = vertices[vid(i, j+1, k)];
                    addFace(fv, cid(i, j, 0), -1);
                } else if (k == N) {
                    fv[0] = vertices[vid(i, j+1, k)];
                    fv[1] = vertices[vid(i+1, j+1, k)];
                    fv[2] = vertices[vid(i+1, j, k)];
                    fv[3] = vertices[vid(i, j, k)];
                    addFace(fv, cid(i, j, N-1), -1);
                } else {
                    fv[0] = vertices[vid(i, j, k)];
                    fv[1] = vertices[vid(i+1, j, k)];
                    fv[2] = vertices[vid(i+1, j+1, k)];
                    fv[3] = vertices[vid(i, j+1, k)];
                    addFace(fv, cid(i, j, k-1), cid(i, j, k));
                }
            }
        }
    }
    // Cell volumes via pyramid decomposition
    for (auto& cell : m.cells) {
        double vol = 0.0;
        for (int fid : cell.faceIds) {
            const Face& f = m.faces[fid];
            Vec3d r = f.centroid - cell.centroid;
            vol += dot(r, f.areaVec) / 3.0;
        }
        cell.volume = std::abs(vol);
    }
    m.maxNonOrthAngle = 0.0;
    m.maxSkewness = 0.0;
    m.minCellVolume = 1e30;
    for (auto& cell : m.cells)
        m.minCellVolume = std::min(m.minCellVolume, cell.volume);
    for (auto& f : m.faces) {
        if (f.neighborCell >= 0) {
            Vec3d xP = m.cells[f.ownerCell].centroid;
            Vec3d xN = m.cells[f.neighborCell].centroid;
            f.dVec = xN - xP;
            double dMag = norm(f.dVec);
            if (dMag > 1e-14) {
                double cosTheta = dot(f.areaVec, f.dVec) / (f.areaMag * dMag);
                cosTheta = std::max(-1.0, std::min(1.0, cosTheta));
                f.nonOrthAngle = std::acos(cosTheta) * 180.0 / M_PI;
                m.maxNonOrthAngle = std::max(m.maxNonOrthAngle, f.nonOrthAngle);
                double dDotS = dot(f.dVec, f.areaVec);
                if (std::abs(dDotS) > 1e-14) {
                    f.beta_f = dot(f.centroid - xP, f.areaVec) / dDotS;
                    f.alpha_f = norm2(f.areaVec) / dDotS;
                    Vec3d xProj = xP + f.beta_f * f.dVec;
                    f.mSkew = f.centroid - xProj;
                    f.skewnessMag = norm(f.mSkew);
                    f.kCorr = f.areaVec - f.alpha_f * f.dVec;
                    m.maxSkewness = std::max(m.maxSkewness, f.skewnessMag);
                } else {
                    f.beta_f = 0.0; f.alpha_f = 0.0;
                    f.mSkew = {0,0,0}; f.kCorr = f.areaVec;
                    f.skewnessMag = 0.0;
                }
            }
        }
        if (f.neighborCell >= 0) {
            double dMag = norm(f.dVec);
            double cosTheta = (dMag > 1e-14) ?
                dot(f.areaVec, f.dVec) / (f.areaMag * dMag) : 0.0;
            f.valid = (cosTheta > 1e-8) && (f.areaMag > 1e-14);
        } else {
            f.valid = (f.areaMag > 1e-14);
        }
    }
    return m;
}
// ---------------------------------------------------------------------------
// MMS
// ---------------------------------------------------------------------------
inline double PI = std::acos(-1.0);
inline double GAMMA = 1.0;
inline double phiExact(double x, double y, double z) {
    return std::sin(PI * x) * std::cos(PI * y) * std::exp(z * 0.5);
}
inline double sourceMMS(double x, double y, double z) {
    double ep = std::exp(z * 0.5);
    double uDotGradPhi = PI * std::sin(PI * x) * std::cos(PI * x) *
                         std::cos(PI * z) * ep;
    double laplacianPhi = (-2.0 * PI * PI + 0.25) *
                          std::sin(PI * x) * std::cos(PI * y) * ep;
    return uDotGradPhi - GAMMA * laplacianPhi;
}
inline double uxExact(double x, double y, double z) {
    return std::sin(PI * x) * std::cos(PI * y) * std::cos(PI * z);
}
inline double uyExact(double x, double y, double z) {
    return -std::cos(PI * x) * std::sin(PI * y) * std::cos(PI * z);
}
// ---------------------------------------------------------------------------
// WLS gradient (double)
// ---------------------------------------------------------------------------
inline Vec3d computeWLSGradient(const Mesh& mesh, int cellIdx,
                                const std::vector<double>& phi) {
    const Cell& cell = mesh.cells[cellIdx];
    Vec3d xP = cell.centroid;
    double phiP = phi[cellIdx];
    double A[3][3] = {{0,0,0},{0,0,0},{0,0,0}};
    double b[3] = {0,0,0};
    for (int fid : cell.faceIds) {
        const Face& f = mesh.faces[fid];
        if (f.neighborCell < 0) continue;
        int nIdx = f.neighborCell;
        Vec3d xN = mesh.cells[nIdx].centroid;
        Vec3d d = xN - xP;
        double dMag2 = dot(d, d);
        if (dMag2 < 1e-16) continue;
        double w = 1.0 / dMag2;
        double dPhi = phi[nIdx] - phiP;
        A[0][0] += w * d.x * d.x; A[0][1] += w * d.x * d.y; A[0][2] += w * d.x * d.z;
        A[1][0] += w * d.y * d.x; A[1][1] += w * d.y * d.y; A[1][2] += w * d.y * d.z;
        A[2][0] += w * d.z * d.x; A[2][1] += w * d.z * d.y; A[2][2] += w * d.z * d.z;
        b[0] += w * dPhi * d.x;
        b[1] += w * dPhi * d.y;
        b[2] += w * dPhi * d.z;
    }
    double M[3][4];
    for (int r = 0; r < 3; ++r) {
        M[r][0] = A[r][0]; M[r][1] = A[r][1]; M[r][2] = A[r][2]; M[r][3] = b[r];
    }
    for (int col = 0; col < 3; ++col) {
        int piv = col;
        for (int r = col + 1; r < 3; ++r)
            if (std::abs(M[r][col]) > std::abs(M[piv][col])) piv = r;
        if (piv != col)
            for (int c = 0; c < 4; ++c) std::swap(M[col][c], M[piv][c]);
        if (std::abs(M[col][col]) < 1e-16) continue;
        for (int r = col + 1; r < 3; ++r) {
            double factor = M[r][col] / M[col][col];
            for (int c = col; c < 4; ++c) M[r][c] -= factor * M[col][c];
        }
    }
    Vec3d grad{0,0,0};
    for (int r = 2; r >= 0; --r) {
        double sum = M[r][3];
        for (int c = r + 1; c < 3; ++c) {
            double vc = (c == 0) ? grad.x : (c == 1) ? grad.y : grad.z;
            sum -= M[r][c] * vc;
        }
        if (std::abs(M[r][r]) > 1e-16) {
            if (r == 0) grad.x = sum / M[r][r];
            else if (r == 1) grad.y = sum / M[r][r];
            else grad.z = sum / M[r][r];
        }
    }
    return grad;
}
// ---------------------------------------------------------------------------
// Templated types for complex-step
// ---------------------------------------------------------------------------
template <typename T>
struct TVec3 {
    T x, y, z;
    TVec3() : x(T(0)), y(T(0)), z(T(0)) {}
    TVec3(T x_, T y_, T z_) : x(x_), y(y_), z(z_) {}
};
template <typename T>
inline T tdot(const TVec3<T>& a, const TVec3<T>& b) {
    return a.x * b.x + a.y * b.y + a.z * b.z;
}
template <typename T>
inline TVec3<T> tsub(const TVec3<T>& a, const TVec3<T>& b) {
    return TVec3<T>(a.x - b.x, a.y - b.y, a.z - b.z);
}
template <typename T>
inline T tnorm2(const TVec3<T>& a) {
    return tdot(a, a);
}
template <typename T>
struct TFace {
    int ownerCell;
    int neighborCell;
    TVec3<T> areaVec;
    T areaMag;
    TVec3<T> dVec;
    TVec3<T> mSkew;
    TVec3<T> kCorr;
    T alpha_f;
    T beta_f;
    bool isInterior;
};
template <typename T>
struct TCell {
    TVec3<T> centroid;
    T volume;
};
template <typename T>
struct TMesh {
    std::vector<TCell<T>> cells;
    std::vector<TFace<T>> faces;
};
template <typename T>
TMesh<T> convertMesh(const Mesh& m) {
    TMesh<T> tm;
    tm.cells.resize(m.cells.size());
    for (size_t i = 0; i < m.cells.size(); ++i) {
        tm.cells[i].centroid = {static_cast<T>(m.cells[i].centroid.x),
                                 static_cast<T>(m.cells[i].centroid.y),
                                 static_cast<T>(m.cells[i].centroid.z)};
        tm.cells[i].volume = static_cast<T>(m.cells[i].volume);
    }
    tm.faces.resize(m.faces.size());
    for (size_t i = 0; i < m.faces.size(); ++i) {
        const Face& f = m.faces[i];
        tm.faces[i].ownerCell = f.ownerCell;
        tm.faces[i].neighborCell = f.neighborCell;
        tm.faces[i].areaVec = {static_cast<T>(f.areaVec.x),
                                static_cast<T>(f.areaVec.y),
                                static_cast<T>(f.areaVec.z)};
        tm.faces[i].areaMag = static_cast<T>(f.areaMag);
        tm.faces[i].dVec = {static_cast<T>(f.dVec.x),
                             static_cast<T>(f.dVec.y),
                             static_cast<T>(f.dVec.z)};
        tm.faces[i].mSkew = {static_cast<T>(f.mSkew.x),
                              static_cast<T>(f.mSkew.y),
                              static_cast<T>(f.mSkew.z)};
        tm.faces[i].kCorr = {static_cast<T>(f.kCorr.x),
                              static_cast<T>(f.kCorr.y),
                              static_cast<T>(f.kCorr.z)};
        tm.faces[i].alpha_f = static_cast<T>(f.alpha_f);
        tm.faces[i].beta_f = static_cast<T>(f.beta_f);
        tm.faces[i].isInterior = (f.neighborCell >= 0);
    }
    return tm;
}
template <typename T>
TVec3<T> computeWLSGradientTemplated(const TMesh<T>& mesh, int cellIdx,
                                     const std::vector<T>& phi) {
    const auto& cell = mesh.cells[cellIdx];
    TVec3<T> xP = cell.centroid;
    T phiP = phi[cellIdx];
    T A[3][3] = {{T(0),T(0),T(0)},{T(0),T(0),T(0)},{T(0),T(0),T(0)}};
    T b[3] = {T(0),T(0),T(0)};
    for (const auto& f : mesh.faces) {
        if (f.ownerCell == cellIdx && f.isInterior) {
            int nIdx = f.neighborCell;
            TVec3<T> xN = mesh.cells[nIdx].centroid;
            TVec3<T> d = tsub(xN, xP);
            T dMag2 = tnorm2(d);
            if (std::abs(dMag2) < T(1e-16)) continue;
            T w = T(1) / dMag2;
            T dPhi = phi[nIdx] - phiP;
            A[0][0] += w * d.x * d.x; A[0][1] += w * d.x * d.y; A[0][2] += w * d.x * d.z;
            A[1][0] += w * d.y * d.x; A[1][1] += w * d.y * d.y; A[1][2] += w * d.y * d.z;
            A[2][0] += w * d.z * d.x; A[2][1] += w * d.z * d.y; A[2][2] += w * d.z * d.z;
            b[0] += w * dPhi * d.x;
            b[1] += w * dPhi * d.y;
            b[2] += w * dPhi * d.z;
        }
    }
    T M[3][4];
    for (int r = 0; r < 3; ++r) {
        M[r][0] = A[r][0]; M[r][1] = A[r][1]; M[r][2] = A[r][2]; M[r][3] = b[r];
    }
    for (int col = 0; col < 3; ++col) {
        int piv = col;
        for (int r = col + 1; r < 3; ++r)
            if (std::abs(M[r][col]) > std::abs(M[piv][col])) piv = r;
        if (piv != col)
            for (int c = 0; c < 4; ++c) std::swap(M[col][c], M[piv][c]);
        if (std::abs(M[col][col]) < T(1e-16)) continue;
        for (int r = col + 1; r < 3; ++r) {
            T factor = M[r][col] / M[col][col];
            for (int c = col; c < 4; ++c) M[r][c] -= factor * M[col][c];
        }
    }
    TVec3<T> grad{T(0),T(0),T(0)};
    for (int r = 2; r >= 0; --r) {
        T sum = M[r][3];
        for (int c = r + 1; c < 3; ++c) {
            T vc = (c == 0) ? grad.x : (c == 1) ? grad.y : grad.z;
            sum -= M[r][c] * vc;
        }
        if (std::abs(M[r][r]) > T(1e-16)) {
            if (r == 0) grad.x = sum / M[r][r];
            else if (r == 1) grad.y = sum / M[r][r];
            else grad.z = sum / M[r][r];
        }
    }
    return grad;
}
// ---------------------------------------------------------------------------
// FVM Solver (templated)
// ---------------------------------------------------------------------------
template <typename T>
std::vector<T> solveFVM2(const TMesh<T>& mesh,
                         const std::vector<std::vector<int>>& cellFaces,
                         T GammaVal, bool useSkewCorr, bool useNonOrthCorr,
                         int maxIter, T tol) {
    int nc = mesh.cells.size();
    std::vector<T> phi(nc, T(0));
    std::vector<T> source(nc);
    for (int c = 0; c < nc; ++c) {
        // Extract real coordinates for evaluating analytic source term
        double x = static_cast<double>(mesh.cells[c].centroid.x);
        double y = static_cast<double>(mesh.cells[c].centroid.y);
        double z = static_cast<double>(mesh.cells[c].centroid.z);
        source[c] = static_cast<T>(sourceMMS(x, y, z));
    }
    T omega = T(0.7);
    for (int iter = 0; iter < maxIter; ++iter) {
        T maxRes = T(0);
        for (int c = 0; c < nc; ++c) {
            auto gradP = computeWLSGradientTemplated(mesh, c, phi);
            T phiP = phi[c];
            T aP = T(0);
            T sumNb = T(0);
            T sumSource = source[c] * mesh.cells[c].volume;
            for (int fid : cellFaces[c]) {
                const auto& f = mesh.faces[fid];
                if (!f.isInterior) continue;
                int nIdx = f.neighborCell;
                T phiN = phi[nIdx];
                // Face centroid as midpoint (real part for function eval)
                double fx = static_cast<double>(
                    (mesh.cells[c].centroid.x + mesh.cells[nIdx].centroid.x) / T(2));
                double fy = static_cast<double>(
                    (mesh.cells[c].centroid.y + mesh.cells[nIdx].centroid.y) / T(2));
                double fz = static_cast<double>(
                    (mesh.cells[c].centroid.z + mesh.cells[nIdx].centroid.z) / T(2));
                TVec3<T> xF = { static_cast<T>(fx), static_cast<T>(fy), static_cast<T>(fz) };
                TVec3<T> uF = { static_cast<T>(uxExact(fx, fy, fz)),
                                static_cast<T>(uyExact(fx, fy, fz)),
                                T(0) };
                T mdot = tdot(uF, f.areaVec);
                T fP = T(0.5);
                T phiF = fP * phiP + (T(1) - fP) * phiN;
                T dMagSq = tnorm2(f.dVec);
                T diffOrth = T(0);
                if (std::abs(dMagSq) > T(1e-28)) {
                    T dMag = std::sqrt(dMagSq);
                    diffOrth = GammaVal * f.alpha_f * (phiN - phiP) / dMag;
                }
                T diffCorr = T(0);
                if (useNonOrthCorr) {
                    auto gradN = computeWLSGradientTemplated(mesh, nIdx, phi);
                    TVec3<T> gradF = TVec3<T>(
                        (gradP.x + gradN.x) / T(2),
                        (gradP.y + gradN.y) / T(2),
                        (gradP.z + gradN.z) / T(2));
                    diffCorr = -GammaVal * tdot(gradF, f.kCorr);
                }
                T skewCorr = T(0);
                if (useSkewCorr) {
                    auto gradN = computeWLSGradientTemplated(mesh, nIdx, phi);
                    TVec3<T> gradF = TVec3<T>(
                        (gradP.x + gradN.x) / T(2),
                        (gradP.y + gradN.y) / T(2),
                        (gradP.z + gradN.z) / T(2));
                    skewCorr = mdot * tdot(gradF, f.mSkew);
                }
                T flux = mdot * phiF - diffOrth + diffCorr + skewCorr;
                T aN = -mdot * fP;
                if (std::abs(dMagSq) > T(1e-28)) {
                    T dMag = std::sqrt(dMagSq);
                    aN -= GammaVal * f.alpha_f / dMag;
                }
                sumNb += aN * phiN;
                aP += -aN;
            }
            aP += T(1);
            T phiNewVal = (sumNb + sumSource) / aP;
            phi[c] = phiP + omega * (phiNewVal - phiP);
            T res = std::abs(phi[c] - phiP);
            maxRes = std::max(maxRes, res);
        }
        if (maxRes < tol) break;
    }
    return phi;
}

// Error computation
// ---------------------------------------------------------------------------
struct ErrorNorms {
    double L2;
    double Linf;
};
ErrorNorms computeErrors(const Mesh& mesh, const std::vector<double>& phi) {
    double L2Sum = 0.0;
    double VSum = 0.0;
    double maxErr = 0.0;
    for (int c = 0; c < (int)mesh.cells.size(); ++c) {
        const auto& x = mesh.cells[c].centroid;
        double phiExactVal = phiExact(x.x, x.y, x.z);
        double err = std::abs(phi[c] - phiExactVal);
        L2Sum += err * err * mesh.cells[c].volume;
        VSum += mesh.cells[c].volume;
        maxErr = std::max(maxErr, err);
    }
    ErrorNorms e;
    e.L2 = std::sqrt(L2Sum / VSum);
    e.Linf = maxErr;
    return e;
}
double computeGCI(double eFine, double eCoarse, int p) {
    const double Fs = 1.25;
    double r = 2.0;
    double GCI = Fs * std::abs(eFine - eCoarse) / eFine / (std::pow(r, p) - 1.0);
    return GCI * 100.0;
}

// Objective functional
// ---------------------------------------------------------------------------
template <typename T>
T computeObjective(const TMesh<T>& mesh,
                   const std::vector<T>& phi) {
    T J = T(0);
    for (int c = 0; c < (int)mesh.cells.size(); ++c) {
        double x = static_cast<double>(mesh.cells[c].centroid.x);
        double y = static_cast<double>(mesh.cells[c].centroid.y);
        double z = static_cast<double>(mesh.cells[c].centroid.z);
        double target = phiExact(x, y, z);
        T diff = phi[c] - static_cast<T>(target);
        J += diff * diff * mesh.cells[c].volume / T(2);
    }
    return J;
}
// ---------------------------------------------------------------------------
// Slope computation
// ---------------------------------------------------------------------------
double computeSlope(const std::vector<double>& eps, const std::vector<double>& res) {
    if (eps.size() < 2) return 0.0;
    double sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0;
    for (size_t i = 0; i < eps.size(); ++i) {
        double x = std::log(eps[i]);
        double y = std::log(res[i] > 0 ? res[i] : 1e-300);
        sumX += x; sumY += y; sumXY += x * y; sumX2 += x * x;
    }
    int n = (int)eps.size();
    return (n * sumXY - sumX * sumY) / (n * sumX2 - sumX * sumX);
}

enum class FaceClass { VALID, INVALID, NON_MANIFOLD, SELF_INTERSECT, DEGENERATE };

FaceClass classifyFace(const Face& f) {
    if (f.areaMag < 1e-14) return FaceClass::DEGENERATE;
    if (f.neighborCell >= 0) {
        double dMag = norm(f.dVec);
        if (dMag < 1e-14) return FaceClass::DEGENERATE;
        double cosTheta = dot(f.areaVec, f.dVec) / (f.areaMag * dMag);
        if (cosTheta <= 1e-8) return FaceClass::INVALID;
        if (cosTheta <= 1e-6) return FaceClass::NON_MANIFOLD;
    }
    return FaceClass::VALID;
}
